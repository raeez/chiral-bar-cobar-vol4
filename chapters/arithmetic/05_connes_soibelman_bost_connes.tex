\chapter{Bost--Connes versus \(\mathcal{V}^{\mathrm{prim}}\): \(C^{*}\)-Algebra versus Trace Class}
\label{v4-arith:connes-soibelman}

The Bost--Connes algebra is a \(C^*\)-dynamical system. The object
\(\mathcal V^{\mathrm{prim}}\) is a Hochschild trace class. Their
common arithmetic character is visible only after passing through the
finite Heisenberg covers and the KMS partition function.

\section{The Bost--Connes \(C^{*}\)-Algebra Explicit}
\label{v4-arith:cs:bost-connes-explicit}

\begin{definition}[Bost--Connes 1995, \S1]
\label{v4-arith:def:bost-connes-algebra}
The Bost--Connes Q-lattice \(C^{*}\)-algebra \(C_{\mathbb{Q}}^{*}\) has
generators
\begin{align}
\{\mu_{n}\}_{n\in\mathbb{N}^{\times}},\quad
\{\mu_{n}^{*}\}_{n\in\mathbb{N}^{\times}},\quad
\{e(\gamma)\}_{\gamma\in\mathbb{Q}/\mathbb{Z}},
\end{align}
with relations
\begin{align}
\mu_{n}\mu_{m}\;&=\;\mu_{nm}, \\
\mu_{n}^{*}\mu_{n}\;&=\;1, \\
\mu_{n}\mu_{n}^{*}\;&=\;e_{n}\;=\;\frac{1}{n}\sum_{\gamma:n\gamma=0}e(\gamma), \\
e(\gamma_{1})e(\gamma_{2})\;&=\;e(\gamma_{1}+\gamma_{2}), \\
\mu_{n}e(\gamma)\mu_{n}^{*}\;&=\;\rho_{n}(e(\gamma))\;=\;\frac{1}{n}\sum_{\delta:n\delta=\gamma}e(\delta).
\end{align}
The \(e(\gamma)\) generate the group \(C^{*}\)-algebra
\(C^{*}(\mathbb{Q}/\mathbb{Z})\); the \(\mu_{n}\) furnish an endomorphism
semigroup action; \(\mu_{n}^{*}\mu_{n}=1\), \(\mu_{n}\mu_{n}^{*}=e_{n}\)
makes \(\mu_{n}\) a Toeplitz isometry. The crossed product
\(C^{*}(\mathbb{Q}/\mathbb{Z})\rtimes\mathbb{N}^{\times}\) yields
\(C_{\mathbb{Q}}^{*}\).
\end{definition}

\begin{theorem}[KMS trace, Bost--Connes~Prop.~3.3 and Thm~5.4]
\label{v4-arith:thm:BC-KMS}
\ClaimStatusProvedElsewhere. The KMS\({}_{\beta}\) state \(\varphi_{\beta}\)
on \(C_{\mathbb{Q}}^{*}\) is unique for \(0<\beta\leq1\). For
\(\beta>1\) the KMS\({}_{\beta}\) states form a simplex with extremal
points indexed by the symmetry-broken arithmetic boundary. In the
standard Gibbs representation the partition function is
\begin{equation}
Z_{\mathrm{BC}}(\beta)\;=\;\mathrm{tr}(e^{-\beta H})\;=\;\sum_{n\geq 1}n^{-\beta}\;=\;\zeta(\beta),
\end{equation}
where \(H=\log N\) generates \(\sigma_{t}(\mu_{n})=n^{it}\mu_{n}\). The
\(\zeta\)-partition function is explicitly motivated by Julia's
primon gas (\emph{Statistical theory of numbers}, Springer
Proc.~Phys.~47, 1990).
\end{theorem}

\section{The Structural Type Mismatch}
\label{v4-arith:cs:type-mismatch}

\begin{observation}
\label{v4-arith:obs:CCR-vs-Cuntz-Toeplitz}
The Heisenberg CCR \([a_{m}^{(p)},a_{n}^{(q)}]=m\delta_{m+n,0}\delta_{pq}\)
is a relation on unbounded self-adjoint operators on a Fock space
with countable spectrum; \(\mu_{n}\mu_{n}^{*}=e_{n}\) is a
Toeplitz--Cuntz relation on bounded partial isometries with
spectrum \(\{1/n\}\cup\{0\}\). These are distinct operator-algebraic
structures.
\end{observation}

\begin{corollary}[Bost--Connes representation factors through the zero-mode lattice]
\label{v4-arith:cor:fock-projections-determine-rep}
\ClaimStatusProvedHere. The Heisenberg CCR and Toeplitz--Cuntz
operator-algebraic structures are incompatible. The finite
Heisenberg covers of \(\mathcal{V}^{\mathrm{prim}}\) carry local Fock
representations, while \(\mathcal{V}^{\mathrm{prim}}\) itself is the
Hochschild trace class. The faithful Bost--Connes action is the
zero-mode lattice action
\(L \hookrightarrow \mathcal{V}^{\mathrm{prim}}\) of
\Cref{v4-arith:def:zero-mode-lattice}, on which it coincides with the
standard Bost--Connes representation on \(\ell^{2}(\mathbb{N}^{\times})\).
\end{corollary}

\section{Action on the Zero-Mode Lattice}
\label{v4-arith:cs:zero-mode-action}

\begin{definition}[zero-mode lattice]
\label{v4-arith:def:zero-mode-lattice}
Let \(L\subset\mathcal{V}^{\mathrm{prim}}\) be the lattice
\begin{equation}
L\;=\;\Bigl\{\prod_{p}|n_{p}\rangle_{p}\colon n=\prod_{p}p^{n_{p}}\in\mathbb{N}^{\times}\Bigr\},
\end{equation}
where \(|n_{p}\rangle_{p}=(a^{(p)}_{-1})^{n_{p}}|0\rangle_{p}/\sqrt{n_{p}!}\)
is the \(n_{p}\)-th Fock number state. Then \(L\cong\ell^{2}(\mathbb{N}^{\times})\)
as a Hilbert space, with standard orthonormal basis
\(\{\delta_{n}\}_{n\in\mathbb{N}^{\times}}\) indexed by positive integers,
where \(\delta_{n}=\prod_{p}|n_{p}\rangle_{p}\) for
\(n=\prod_{p}p^{n_{p}}\).
\end{definition}

\begin{theorem}[Bost--Connes acts on \(L\)]
\label{v4-arith:thm:BC-acts-on-L}
\ClaimStatusProvedHere. \(C_{\mathbb{Q}}^{*}\) acts faithfully on
the zero-mode lattice \(L\subset\mathcal{V}^{\mathrm{prim}}\) via:
\begin{itemize}
\item \(\mu_{n}\delta_{m}=\delta_{nm}\),
\item \(\mu_{n}^{*}\delta_{m}=\delta_{m/n}\) if \(n\mid m\), else zero,
\item \(e(\gamma)\delta_{m}=\zeta_{m}(\gamma)\delta_{m}\) with
\(\zeta_{m}\) a primitive \(m\)-th root of unity.
\end{itemize}
This is the Bost--Connes representation from 1995~\S1.2, Prop.~1,
transported to the zero-mode lattice of \(\mathcal{V}^{\mathrm{prim}}\).
\end{theorem}

\begin{observation}[maximality of the zero-mode action]
\label{v4-arith:obs:zero-mode-action-maximal}
The Bost--Connes action on \(L\) is maximal in \(\mathcal{V}^{\mathrm{prim}}\):
the off-lattice states (coherent states, descendants carrying
higher Heisenberg modes \(a^{(p)}_{-n}\) for \(n \geq 2\), and their
superpositions) lie outside the Bost--Connes orbit, so \(L\) is
the largest \(C^{*}_{\mathbb{Q}}\)-stable subspace of
\(\mathcal{V}^{\mathrm{prim}}\).
\end{observation}

\section{The Tensor Decomposition}
\label{v4-arith:cs:tensor-decomposition}

\begin{theorem}[tensor-product placement of \(\mathcal{V}^{\mathrm{prim}}\)]
\label{v4-arith:thm:BC-tensor-placement}
\ClaimStatusProvedHere. For each finite \(N\), let
\(\mathcal{V}^{\mathrm{prim}}_{N}=\bigotimes_{p\leq N}\mathcal{H}^{(p)}\)
be the rank-\(\pi(N)\) truncation. Then as graded vector spaces,
\begin{equation}
\mathcal{V}^{\mathrm{prim}}_{N}\;\cong\;
\underbrace{C_{\mathbb{Q},N}^{*}\bigr|_{\mathrm{lattice}}}_{\text{arithmetic}}
\;\otimes\;
\underbrace{\mathcal{H}^{\otimes\pi(N)}_{\mathrm{descendant}}}_{\text{universal}}
\end{equation}
where the left factor is the Bost--Connes action on the rank-\(\pi(N)\)
zero-mode lattice, and the right factor is the universal
Heisenberg-descendant tower of rank \(\pi(N)\) carrying all
higher-mode data \(a^{(p)}_{-n}\) for \(n\geq 2\).
\end{theorem}

\begin{corollary}
\label{v4-arith:cor:arithmetic-content-on-lattice}
\ClaimStatusProvedHere. All arithmetic content of
\(\mathcal{V}^{\mathrm{prim}}\) lives in the Bost--Connes factor
(left-hand side). The Heisenberg descendant tower depends only on
the rank \(\pi(N)\), not on the arithmetic nature of the primes; it
is a universal rank-\(\pi(N)\) Heisenberg dressing.
\end{corollary}

\section{Kesten--McKay = Plancherel of Unramified \(GL_{2}(\mathbb{Q}_{p})\)}
\label{v4-arith:cs:kesten-mckay-plancherel}

\begin{theorem}[Kesten 1959; McKay 1981; Lubotzky 1994~Thm~7.4.5]
\label{v4-arith:thm:kesten-mckay-plancherel}
\ClaimStatusProvedElsewhere. The Kesten--McKay distribution on
\([-2\sqrt{p},2\sqrt{p}]\) for \(q=p\) prime,
\begin{equation}
d\nu_{p}(x)\;=\;\frac{(p+1)\sqrt{4p-x^{2}}}{2\pi((p+1)^{2}-x^{2})}\,dx,
\end{equation}
is the spectral measure of the adjacency operator on the
\((p+1)\)-regular tree, equivalently the \(p\)-adic Bruhat--Tits tree
of \(\mathrm{PGL}_{2}(\mathbb{Q}_{p})\), equivalently the Plancherel
measure of the spherical-function algebra on the tree.
\end{theorem}

\begin{corollary}
\label{v4-arith:cor:kesten-mckay-is-local-hecke}
\ClaimStatusProvedHere. The Kesten--McKay averaging is the
Plancherel decomposition of the \(GL_{2}(\mathbb{Q}_{p})\) spherical
Hecke algebra, not a spectral theorem for the \(GL_{1}\) Bost--Connes
semigroup algebra. Any map from the Julia--Bost--Connes partition
function to \(\nu_{p}\) must pass through a specified \(GL_{1}\to GL_{2}\)
correspondence.
\end{corollary}

\begin{remark}[arithmetic averaging as Plancherel decomposition]
\label{v4-arith:rem:arithmetic-averaging-plancherel}
The arithmetic averaging map \(\mathrm{av}^{\mathrm{arith}}\) is the
Plancherel decomposition of an automorphic representation, while the
operadic averaging \(\mathfrak{g}^{E_{1}}\to\mathfrak{g}^{\mathrm{mod}}\)
of Vol~I projects onto a symmetric sector weighted by a measure on
chiral cohomology. Both have the formal shape of a projection onto a
symmetric sector, but the operadic averaging acts on chiral cohomology
and the Plancherel averaging acts on spherical representations of
\(GL_2(\mathbb{Q}_p)\): distinct objects sharing a formal pattern.
\end{remark}

\section{The Soibelman Cohomological Hall Algebra Reading}
\label{v4-arith:cs:soibelman-coha}

\begin{theorem}[Kontsevich--Soibelman 1006.2706 \S2.4]
\label{v4-arith:thm:KS-CoHA-adelic}
\ClaimStatusProvedElsewhere. The Kesten--McKay spectral measure
\(\nu_{p}\) arises as the Behrend-weighted motivic Euler
characteristic of the moduli stack of rank-\(n\) representations of
the Jordan quiver (one vertex, one loop, trivial potential) at the
\(p\)-adic coefficient. This is a statement of the cohomological Hall
algebra (CoHA) of Kontsevich--Soibelman at the adelic coefficient.
\end{theorem}

\begin{proposition}[three-way comparison obstruction]
\label{v4-arith:prop:HY-BC-Soibelman-dictionary}
\ClaimStatusProvedHere. The composed arrow
\begin{equation*}
\text{Julia primon gas}
\xrightarrow{\text{Plancherel}}
\mathrm{Bost\text{-}Connes\,Hecke\text{-}Plancherel}

is not defined by the preceding theorems. The Bost--Connes algebra is

\ClaimStatusProvedElsewhere.
\begin{align}
HH^{0}(C_{\mathbb{Q}}^{*})\;&=\;\mathbb{C}\quad(\text{centre: scalars}), \\

\begin{observation}
\label{v4-arith:obs:CM-stricter-than-H-arith}

\mathcal{V}^{\mathrm{prim}}_{N}\;=\;\bigl(\text{Bost--Connes module at rank }\pi(N)\bigr)

All arithmetic content (Hecke algebra, KMS states,

\begin{remark}[module structure]
\(\mathcal{V}^{\mathrm{prim}}\) is a trace object with a Bost--Connes

\section{Structural Conclusions}
\label{v4-arith:cs:summary}

\begin{enumerate}
\item Bost--Connes \(C^{*}_{\mathbb{Q}}\) carries \(\zeta(\beta)\) as
KMS partition function (\Cref{v4-arith:thm:BC-KMS}).
\item The Heisenberg CCR and Toeplitz--Cuntz relations are
structurally distinct operator algebras; the Fock-space
projections determine the representation, and the Bost--Connes
action on \(\mathcal{V}^{\mathrm{prim}}\) factors through the
zero-mode lattice (\Cref{v4-arith:cor:fock-projections-determine-rep}).
\item Bost--Connes acts faithfully on the zero-mode lattice
\(L \subset \mathcal{V}^{\mathrm{prim}}\); the off-lattice states
lie outside the Bost--Connes orbit
(\Cref{v4-arith:thm:BC-acts-on-L},
\Cref{v4-arith:obs:zero-mode-action-maximal}).
\item \(\mathcal{V}^{\mathrm{prim}}_{N} \cong
(\text{Bost--Connes module}) \otimes (\text{universal Heisenberg})\):
arithmetic content on the left, Heisenberg-cover structure on the right
(\Cref{v4-arith:thm:BC-tensor-placement}).
\item Kesten--McKay spectral measure \(= \) Plancherel of unramified
\(GL_{2}(\mathbb{Q}_{p}) = \) Bost--Connes local Hecke spectrum \(= \)
Soibelman CoHA on the Jordan quiver
(\Cref{v4-arith:cor:kesten-mckay-is-local-hecke},
\Cref{v4-arith:prop:HY-BC-Soibelman-dictionary}).
\item Connes--Marcolli \(HH^{\bullet}(C_{\mathbb{Q}}^{*})\) concentrates
in degrees \(\{0,1\}\), sharpening Theorem~H\(^{\mathrm{arith}}\)
(\Cref{v4-arith:thm:CM-HH-BC}).
\item The finite Heisenberg covers lie in archetype \(\mathsf{G}\)
with \(\kappa=\pi(N)\), \(r_{\max}=2\); the \(\mathsf{B}\)-row threshold
requires Borcherds-product encoding of Hecke data
(\Cref{v4-arith:cor:placement-G-archetype}).
\end{enumerate}

The vertex algebra structure is universal and non-arithmetic because
it belongs only to the finite Heisenberg covers. The arithmetic content
is carried by the Bost--Connes \(C^{*}\)-algebra factor and by the
Hochschild trace class. A genuine VA-arithmetic object encoding Hecke
eigenvalues in the stress tensor would be a new object, not
\(\mathcal{V}^{\mathrm{prim}}\).
