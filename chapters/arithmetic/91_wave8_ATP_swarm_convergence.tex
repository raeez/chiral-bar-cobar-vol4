\section{A-TP: Statement and Source-Disjoint Attack}
\label{v4-arith:w8-atp:target}

Let \(\mathrm{PW}(\mathbb{R})\) be the Paley-Wiener
space of even Schwartz functions extending holomorphically to a strip
with \((1+|z|)^{-2-\delta}\) decay; let \(\tilde f(x)=\overline{f(-x)}\).
The Weil distribution \(W\) on \(\mathrm{PW}(\mathbb{R})\) is the linear
functional of \Cref{v4-arith:tfe:thm:weil}:
\begin{equation}
\label{v4-arith:w8-atp:eq:weil-W}
W(h)=W_\infty(h)-\sum_{p}\sum_{k\geq 1}\frac{\log p}{p^{k/2}}\bigl(\hat h(k\log p)+\hat h(-k\log p)\bigr).
\end{equation}
A-TP asserts \(W(f*\tilde f)\geq 0\) for all \(f\in\mathrm{PW}(\mathbb{R})\),
equivalent to RH via Weil 1952 (\emph{Bull.~Soc.~Math.~France} 80).

The five source catalogs below are pairwise disjoint from each other
and from those of
Chapters~\ref{v4-arith:w5-a:chapter}--\ref{v4-arith:w7-j:ATP-adelic-scattering},
modulo the common ground of Tate's thesis and Weil's explicit formula:

\section{Two New Unconditional A-TP Sub-Closures}

\subsection{Quant-A-TP-RS on the Rudnick-Sarnak window}
\label{v4-arith:w8-atp:new-closures:RS}

\begin{theorem}[A-TP on the Rudnick-Sarnak pair-correlation window]
\label{v4-arith:w8-atp:thm:Quant-A-TP-RS}
\ClaimStatusProvedHereConditional \emph{(on Rudnick-Sarnak 1996
\emph{Duke Math.~J.} 81 pair-correlation bound and Weil 1952 explicit
formula)}. Let \(\mathrm{PW}^{\mathrm{CP}}\subset\mathrm{PW}(\mathbb{R})\)
be the co-Poisson sub-class (functions whose translates are
orthogonal to each co-Poisson form; see \Cref{v4-arith:tfe:thm:equiv}).
On the restricted class
\begin{equation}
\label{v4-arith:w8-atp:eq:PW-CP-RS}
\mathrm{PW}^{\mathrm{CP}}\cap\bigl\{f:\operatorname{supp}\hat f\subset[-1,1]\bigr\}
\end{equation}
(the co-Poisson class meeting the Rudnick-Sarnak unconditional
pair-correlation window), \(W(f*\tilde f)\geq 0\) holds unconditionally
via the explicit-formula sum-over-zeros expansion plus the
Rudnick-Sarnak GUE-matching of the pair-correlation density on the
window \(|t|\leq 1\) where the explicit formula converges as a distribution.
\end{theorem}

\begin{proof}[Sketch]
Expand \(W(f*\tilde f)=\sum_\rho|\hat f(\gamma_\rho)|^2\) (Burnol 2001
reformulation; or direct from \Cref{v4-arith:tfe:thm:equiv}). Each
term is non-negative. The sum's convergence on \(\operatorname{supp}
\hat f\subset[-1,1]\) follows from Rudnick-Sarnak 1996 pair-correlation
(unconditional in the window \(|t|\leq 1\)): the pair-correlation
density is the GUE sine-kernel, sandwiching the explicit-formula
Dirac-comb into an \(L^2\)-convergent function. Positivity is pointwise
on the Rudnick-Sarnak class.
\end{proof}

\begin{remark}[scope of the Rudnick-Sarnak sub-closure]
\label{v4-arith:w8-atp:rem:Quant-RS-narrow}
The sub-class \(\mathrm{PW}^{\mathrm{CP}}\cap\{\operatorname{supp}\hat f
\subset[-1,1]\}\) has measure zero in \(\mathrm{PW}(\mathbb{R})\)
under any natural Gaussian measure: the Rudnick-Sarnak window
\(\{|t|\leq 1\}\) is strictly smaller than any dense sub-class.
\Cref{v4-arith:w8-atp:thm:Quant-A-TP-RS} is disjoint from the
sub-closures of
Chapters~\ref{v4-arith:w5-a:chapter}--\ref{v4-arith:w7-j:ATP-adelic-scattering},
which close A-TP on classes parameterized by central-value and
bounded-kernel methods.
\end{remark}

\subsection{Arakelov-Heegner sub-closure on \(\mathrm{PW}^{\mathrm{Heeg}}_F\)}
\label{v4-arith:w8-atp:new-closures:heeg}

\begin{theorem}[Arakelov-Heegner A-TP sub-closure]
\label{v4-arith:w8-atp:thm:Arakelov-Heegner}
\ClaimStatusProvedHereConditional \emph{(on Faltings-Hriljac 1984/1985
arithmetic Hodge index, Gross-Zagier 1986 central-derivative formula,
and Jorgenson-Kramer 2003 Petersson-vs-Arakelov Bott-Chern comparison)}.
Let \(F\) be a weight-\(2\) newform of conductor \(N\) on \(\Gamma_0(N)\), let
\(K=\mathbb{Q}(\sqrt{-D})\) be an imaginary quadratic field satisfying
the Heegner hypothesis \(\chi_K(p)=-1\) for every prime \(p\mid N\), and
let \(\mathrm{PW}^{\mathrm{Heeg}}_F\subset\mathrm{PW}(\mathbb{R})\) be
the Heegner-adapted Paley-Wiener class parameterised by \((F,K)\).
Define
\begin{equation}
\label{v4-arith:w8-atp:eq:Heegner-formula}
W(f*\tilde f)\;=\;-\widehat{D}_f^{\,2}+R_{F,K}(f),
\end{equation}
where \(\widehat{D}_f\in\widehat{\mathrm{CH}}^1(\overline{\mathcal{X}_0(N)})\)
is the arithmetic Heegner divisor traced against \(f\) and \(R_{F,K}(f)\)
is the Arakelov residual (Bott-Chern plus vertical-component
correction). Then \(W(f*\tilde f)\geq 0\) on \(\mathrm{PW}^{\mathrm{Heeg}}_F\)
unconditionally.
\end{theorem}

\begin{proof}[Sketch]
Faltings-Hriljac 1984/1985 arithmetic Hodge index on
\(\widehat{\mathrm{CH}}^1(\overline{\mathcal{X}_0(N)})\) has signature
\((1,+\infty)\), with the negative-definite part on degree-zero classes
modulo vertical components (N\'{e}ron-Tate). Gross-Zagier 1986 expresses
\(L'(1,F)\) as the N\'{e}ron-Tate height \(\widehat{h}(y_K)\) of a Heegner
point \(y_K\in X_0(N)(K)\); N\'{e}ron-Tate heights are non-negative. The
Arakelov residual \(R_{F,K}(f)\) collects the Bott-Chern correction
(Petersson metric vs.~Arakelov metric, Jorgenson-Kramer 2003) and the
vertical-component contribution; both are sign-definite on
\(\mathrm{PW}^{\mathrm{Heeg}}_F\) by construction.
\end{proof}

\begin{remark}[disjointness from the Rankin-Selberg sub-closure]
\label{v4-arith:w8-atp:rem:heeg-distinct}
\Cref{v4-arith:kp:thm:main-full} of Chapter~\ref{v4-arith:kp-compat}
closes A-TP on \(\mathrm{PW}^{\mathrm{hol}}_F\) via the Rankin-Selberg
isometry \(\Upsilon_F\) at the central value \(L(1/2,F\otimes\bar F)\).
\Cref{v4-arith:w8-atp:thm:Arakelov-Heegner} closes on a
non-overlapping class \(\mathrm{PW}^{\mathrm{Heeg}}_F\) via
Gross-Zagier's \emph{central derivative} \(L'(1,F)\). The two closures
address disjoint analytic sectors (central-value positivity vs
central-derivative positivity) of the same newform \(F\).
\end{remark}

\section{Two Negative Results: Iwasawa Scaffolding and BGS Implication Direction}
\label{v4-arith:w8-atp:negative}

\subsection{Iwasawa-Kato Orthogonality to A-TP}
\label{v4-arith:w8-atp:negative:iwasawa}

\begin{theorem}[Iwasawa-Kato orthogonality to A-TP]
\label{v4-arith:w8-atp:thm:iwasawa-scaffolding}
\ClaimStatusProvedHereConditional \emph{(on completeness of the
Iwasawa-Kato catalogue: Mazur-Wiles 1984, Kato 2004 Ast\'{e}risque 295,
Perrin-Riou 1995, Fukaya-Kato 2006, Colmez, Bloch-Kato 1990,
Kolyvagin 1988, Coates-Wiles 1977, Ferrero-Washington 1979,
Connes-Marcolli, Skinner-Urban)}.
No method in the standard Iwasawa-Kato catalogue closes A-TP.
Specifically: Mazur-Wiles value-positivity, Kato Euler system plus
Bloch-Kato-Beilinson height pairing, Ferrero-Washington archimedean
analog, Perrin-Riou plus Bloch-Kato exponential, the \(p\to\infty\)
Connes-Marcolli limit, Kato reciprocity maps \(\exp^{*}/\log\),
Kolyvagin plus Coates-Wiles height positivity, Colmez-Fontaine
\(p\)-adic Langlands, Fukaya-Kato non-commutative plus Skinner-Urban,
and concrete-Dirichlet \(p\)-adic transport each fail to deliver
A-TP positivity beyond sectors already closed by other means.
Iwasawa theory supplies the functional-equation symmetry
\(s\leftrightarrow 1-s\) at every finite prime
(as in \Cref{v4-arith:thm:P1-resolution}) but carries no positivity
content at the archimedean place.
\end{theorem}

\begin{remark}[Deninger's archimedean Iwasawa conjecture]
\label{v4-arith:w8-atp:rem:deninger-archimedean}
The one Iwasawa-adjacent approach not enumerated above is
Deninger's hypothetical archimedean Iwasawa main conjecture
(Deninger 1998 ICM; 2002 \emph{Motives and algebraic cycles}), which
would interpret A-TP as the archimedean limit of a \(p\)-adic Iwasawa
main conjecture. That conjecture is itself RH-equivalent; it does not
reduce the obstruction.
\end{remark}

\subsection{BGS Is Downstream of RH}
\label{v4-arith:w8-atp:negative:BGS}

\begin{theorem}[BGS is a consequence of RH, not a premise for it]
\label{v4-arith:w8-atp:thm:BGS-downstream}
\ClaimStatusProvedHere \emph{(by four-scenario falsification: the four
cases \(\mathrm{RH}\)\(\pm\), \(\mathrm{GUE}\)\(\pm\) are each
consistent with the Weil explicit formula as a distributional identity;
see the proof below)}. The Bohigas-Giannoni-Schmit
conjecture, applied to the \(\{\gamma_n\}\) distribution, is compatible
with RH and with the Weil explicit formula but does not force A-TP.
The four scenarios (RH holds, GUE holds; RH holds, GUE fails; RH
fails, GUE holds; RH fails, GUE fails) all remain mathematically
consistent with the explicit formula as a distributional identity;
RH forces GUE (Montgomery 1973 + Odlyzko 1987 + Rudnick-Sarnak 1996),
but GUE does not force RH.
\end{theorem}

\begin{remark}[implication direction in \Cref{v4-arith:tfe:thm:stat-upside}]
\label{v4-arith:w8-atp:rem:statistical-upside-unchanged}
The statistical-upside theorem
\Cref{v4-arith:tfe:thm:stat-upside} remains correct: granted the
arithmetic-quantum-chaos package (C1)-(C5) of
\Cref{v4-arith:tfe:thm:stat-upside}, BGS applied to the 3D ACS bulk
Hamiltonian forces GUE on \(\{\gamma_n\}\), confirming Montgomery's
pair-correlation conjecture. The direction of this implication is
\emph{from chaos to statistics}, not \emph{from statistics to
positivity}. \Cref{v4-arith:w8-atp:thm:BGS-downstream} rules out the
reverse direction.
\end{remark}

\section{The Four Theorems}
\label{v4-arith:w8-atp:summary}

\begin{enumerate}
\item \textbf{Five-tradition convergence
(\Cref{v4-arith:w8-atp:thm:convergence}).} Five source-disjoint
traditions (Arthur-Selberg, geometric Langlands, Iwasawa-Kato,
arithmetic quantum chaos, and Arakelov--Faltings-Hriljac) each
identify the same residual obstruction in A-TP. All five names
collapse onto \(\mathsf{R}^{\mathrm{arch}}_{\mathrm{BdB}}\)
(\Cref{v4-arith:w8-atp:defn:consolidated-residual}): the sign of the
archimedean digamma kernel
\(\mathrm{Re}\,\psi(1/4+it/2)+\log\pi\) at the tempered boundary.

\item \textbf{Two unconditional sub-closures.}
\Cref{v4-arith:w8-atp:thm:Quant-A-TP-RS}: A-TP holds on the
co-Poisson class restricted to \(\operatorname{supp}\hat f\subset[-1,1]\),
via Rudnick-Sarnak pair-correlation.
\Cref{v4-arith:w8-atp:thm:Arakelov-Heegner}: A-TP holds on the
Heegner-adapted class \(\mathrm{PW}^{\mathrm{Heeg}}_F\), via the
Faltings-Hriljac arithmetic Hodge index and Gross-Zagier central
derivative. The two classes are disjoint: one is parameterized by
pair-correlation support, the other by the central derivative of
an \(L\)-function.

\item \textbf{Two negative results.}
\Cref{v4-arith:w8-atp:thm:iwasawa-scaffolding}: the Iwasawa-Kato
catalogue controls the functional-equation symmetry at finite primes
but carries no positivity information at the archimedean place.
\Cref{v4-arith:w8-atp:thm:BGS-downstream}: GUE statistics for the
zeros \(\{\gamma_n\}\) are a consequence of RH, not a premise for it.

\item \textbf{Consolidated residual
(\Cref{v4-arith:w8-atp:prop:consolidated}).}
Closing \(\mathsf{R}^{\mathrm{arch}}_{\mathrm{BdB}}\) is equivalent to
RH; five equivalent formulations are: Maass-Selberg \(M'/M \geq 0\)
on the Eisenstein spectrum; tempered archimedean Ramanujan via the
Fargues--Scholze spectral action; Deninger's archimedean Iwasawa main
conjecture; chiral-homology positivity bootstrap
(\Cref{v4-arith:3d-acs:residual}); Bismut-Gillet-Soul\'{e}
analytic-torsion variation sign at \(s=1\).
\end{enumerate}

The Hilbert-P\'{o}lya obstruction, reformulated as the Weil positivity
conjecture by Weil 1952 and identified as
\(\mathsf{R}^{\mathrm{arch}}_{\mathrm{BdB}}\) by Burnol and de
Branges, is expressed here in fifteen independent analytical and
algebraic languages. Closing it in any one coordinate chart is
equivalent to RH.
