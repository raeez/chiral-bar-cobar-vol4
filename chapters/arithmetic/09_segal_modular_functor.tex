\chapter{The Arithmetic Modular Functor}
\label{v4-arith:modular-functor}

\emph{In the voice of Segal. The CFT axioms of Segal (in
\emph{Topology, geometry and QFT}, CUP~2002, pp.~421--577) take a
Riemann surface \(\Sigma_{g,n}\) with marked points and return a
vector space of conformal blocks; sewing glues conformal blocks
along circles. We lift Segal's functor to the Arakelov curve and
arrive at the arithmetic avatar of every axiom.}

\section{Axioms and Their Translation}
\label{v4-arith:segal:axioms-translation}

Each Segal axiom for a Riemann surface \(\Sigma_{g,n}\) -- finiteness of conformal blocks, sewing along circles, modular invariance, and the partition-function trace -- has a precise arithmetic counterpart on the Arakelov curve \((\overline{\mathrm{Spec}\,\mathcal{O}_{K}},S)\). The correspondence is entry-wise, not analogical: each geometric datum on the left is replaced by the arithmetic datum on the right that plays the same role in the respective cohomology theory.

\begin{convention}[arithmetic Segal dictionary]
\label{v4-arith:conv:arithmetic-segal-dictionary}
\begin{center}
\begin{tabular}{ll}
Geometric Segal & Arithmetic Segal \\
\hline
Riemann surface \(\Sigma_{g,n}\) & Arakelov curve \((\overline{\mathrm{Spec}\,\mathcal{O}_{K}},S)\) \\
Orientation & Trivialization of \(\det R\Gamma\) (Deligne) \\

Marked point & Place (finite or archimedean) \\
Sewing circle & Punctured formal disk \(\mathrm{Spec}\,K_{\mathfrak{p}}\) \\

\end{convention}

\section{Arakelov Genus as Lattice Volume}
\label{v4-arith:segal:genus}

\begin{observation}[why genus \(=\) \(\tfrac{1}{2}\log|d_{K}|\)]
\label{v4-arith:obs:genus-lattice-volume}
The geometric genus satisfies \(g=\tfrac{1}{2}\dim_{\mathbb{R}}H^{1}(\Sigma;\mathbb{R})\); equivalently, the Jacobian \(J(\Sigma)=H^{0}(\Sigma,\Omega^{1})^{*}/H_{1}(\Sigma,\mathbb{Z})\) is a \(g\)-dimensional torus whose Weil height is \(\tfrac{1}{2}\log\det(\mathrm{Im}\,\tau)\) for the period matrix \(\tau\).
The arithmetic analogue is \(\tfrac{1}{2}\log|d_{K}|=\tfrac{1}{2}\log(\mathrm{covol}(\mathcal{O}_{K}\subset K\otimes_{\mathbb{Q}}\mathbb{R}))\).
Both are half-logarithms of lattice covolumes; the functional is the same, the lattices differ.
\end{observation}

\begin{definition}[arithmetic Euler characteristic]
\label{v4-arith:def:arithmetic-euler-characteristic}
\begin{equation}
\chi^{\mathrm{arith}}(\overline{X})\;:=\;-\widehat{\deg}\,\omega_{\overline{X}}+\sum_{v\mid\infty}\log\|\cdot\|_{v}.
\end{equation}
On \(\mathrm{Spec}\,\mathcal{O}_{K}\) with signature \((r_{1},r_{2})\),
\(\chi^{\mathrm{arith}}=\log|d_{K}|-r_{1}\log(\pi/4)-r_{2}\log(\pi/2)\)
by Minkowski--Odlyzko. Additive on disjoint unions of Arakelov
surfaces.
\end{definition}

\section{Weil--Deligne Sewing}
\label{v4-arith:segal:weil-deligne-sewing}

% RECTIFICATION-FLAG: Gate 1 -- sewing tensor (here \otimes) vs A3 line-product (\cdot) inconsistency; the two formulations differ in whether conformal blocks are lines or higher-rank spaces. The tensor form is kept as the primary statement; A3 in Definition~\ref{v4-arith:def:arithmetic-modular-functor} should be aligned to match.
\begin{theorem}[sewing law]
\label{v4-arith:thm:weil-deligne-sewing}
\ClaimStatusConjectured.

Here \(\mathcal{F}^{\mathrm{arith}}\) denotes the arithmetic modular functor
defined in Definition~\ref{v4-arith:def:arithmetic-modular-functor}
(an adelically factorized assignment of meromorphic lines to Arakelov
surfaces with marked places); and \(\mathcal{V}^{\mathrm{prim}}\)
denotes the Hochschild trace class of the identity on the global
arithmetic Iwahori--Hecke category of \(GL_{2}\) (a class in
\(K\)-theory, not a vertex algebra). The \(\ell\)-adic coefficient
system at a non-archimedean prime \(\mathfrak{p}\) is the \'etale
cohomology local system attached to \(\mathcal{V}^{\mathrm{prim}}\)
via the comparison isomorphism \(H^{*}_{\mathrm{\acute{e}t}}(-,\mathbb{Q}_{\ell})\)
for any prime \(\ell\neq p\); this is the coefficient ring of the
Weil--Deligne representation, not an enrichment of \(\mathcal{V}^{\mathrm{prim}}\)
with vertex-algebra structure.
\begin{equation}
\mathcal{F}^{\mathrm{arith}}(\overline{X}_{1}\cup_{p}\overline{X}_{2})
\;=\;\bigoplus_{(\rho,N)}
\mathcal{F}^{\mathrm{arith}}(\overline{X}_{1};\rho,N)\otimes_{\mathrm{End}(\rho)}
\mathcal{F}^{\mathrm{arith}}(\overline{X}_{2};\rho,N)^{*}
\end{equation}
where \((\rho,N)\) runs over isomorphism classes of Weil--Deligne
representations of \(W_{\mathbb{Q}_{p}}\) whose coefficients are
\(\ell\)-adic (\(\ell\neq p\)),
\(\rho\) continuous, \(N\) the monodromy operator.
\end{theorem}

\begin{remark}[labels are Langlands parameters]
Segal's intermediate representation labels are local Galois
representations. At archimedean places, sewing is along Hodge
structures (Weil-group representations of \(W_{\mathbb{R}}\) or
\(W_{\mathbb{C}}\)). The correspondence between WZW representation
labels and automorphic representations under local Langlands is:
\begin{equation*}
\bigl\{\text{irr.~admissible }G(\mathbb{Q}_{p})\text{-reps}\bigr\}\;\leftrightarrow\;\bigl\{\text{Weil--Deligne }L\text{-parameters}\bigr\}.
\end{equation*}
The arithmetic modular functor is the local Langlands correspondence in Segal's axiom system.
\end{remark}

\section{Mapping Class Group Is Galois Plus Adelic}
\label{v4-arith:segal:MCG}

\begin{observation}
\label{v4-arith:obs:MCG-decomposition}
The arithmetic MCG splits into two complementary actions.
The Galois group \(\mathrm{Gal}(\overline{K}/K)\) controls label permutations:
it acts on Weil--Deligne parameters at marked places, the arithmetic
counterpart of the deck transformations on the geometric surface.
The adelic quotient \(\mathrm{GL}_{n}(\mathbb{A}_{K})/\mathrm{GL}_{n}(K)\)
controls conformal blocks via Hecke operators: it is the arithmetic
counterpart of the isotopy-class action on holomorphic sections.
For \(g=1\), the geometric \(\mathrm{MCG}=\mathrm{SL}_{2}(\mathbb{Z})\)
is the arithmetic prototype; it embeds as the \(\mathrm{GL}_{2}\)
factor restricted to principal adeles.
The semidirect product \(\mathrm{Gal}(\overline{K}/K)\ltimes\mathrm{GL}_{n}(\mathbb{A}_{K})/\mathrm{GL}_{n}(K)\)
combines both actions.
\end{observation}

% RECTIFICATION-FLAG: Gate 1 -- MCG semidirect product action not fully specified: the action of Gal(\bar K/K) on GL_n(A_K)/GL_n(K) by conjugation on the adelization (via automorphisms of the coefficient field) is standard but should be written out explicitly. The current statement gives the group but not the cocycle.
\begin{theorem}[arithmetic MCG]
\label{v4-arith:thm:arithmetic-MCG}
\ClaimStatusConjectured.
\begin{equation}
\mathrm{MCG}^{\mathrm{arith}}\;=\;\mathrm{Gal}(\overline{K}/K)\ltimes\mathrm{GL}_{n}(\mathbb{A}_{K})/\mathrm{GL}_{n}(K),
\end{equation}
where \(\mathrm{Gal}(\overline{K}/K)\) acts on \(\mathrm{GL}_{n}(\mathbb{A}_{K})/\mathrm{GL}_{n}(K)\)
via the natural action on the coefficient field of the adele ring.
\(\mathrm{Gal}\) permutes Weil--Deligne parameters at marked places;
\(\mathrm{GL}_{n}(\mathbb{A}_{K})/\mathrm{GL}_{n}(K)\) acts on
conformal blocks via Hecke operators. Frobenius at \(p\) plays the
role of Dehn twist around the \(p\)-th sewing circle: both generate
\(\pi_{1}(\mathrm{Spec}\,k_{\mathfrak{p}}\setminus\{0\})\cong\hat{\mathbb{Z}}\)
and act on conformal blocks by local monodromy.
\end{theorem}

\section{Verlinde Formula = Class Number Formula}
\label{v4-arith:segal:verlinde}

\begin{theorem}[arithmetic Verlinde]
\label{v4-arith:thm:arithmetic-verlinde}
\ClaimStatusProvedElsewhere (Tate's thesis, class number formula).
\(\dim\mathcal{F}^{\mathrm{arith}}(\mathrm{Spec}\,\mathcal{O}_{K},\emptyset)\),
interpreted via zeta-regularisation, equals the residue of the
Dedekind zeta at \(s=1\):
\begin{equation}
\boxed{\;\dim\mathcal{F}^{\mathrm{arith}}(\mathrm{Spec}\,\mathcal{O}_{K},\emptyset)
\;=\;\mathrm{Res}_{s=1}\zeta_{K}(s)\;=\;\frac{2^{r_{1}}(2\pi)^{r_{2}}h_{K}R_{K}}{w_{K}\sqrt{|d_{K}|}}.\;}
\end{equation}
The class number formula \emph{is} the arithmetic Verlinde
formula.
\end{theorem}

\section{Explicit Computation: Conformal Blocks via Tate Integrals}
\label{v4-arith:segal:explicit-computation}

\begin{computation}[conformal block at \((\mathbb{Z},\{p,\infty\},\mathrm{triv})\)]
\label{v4-arith:comp:conformal-block-p-infty}
\(\mathcal{F}^{\mathrm{arith}}(\mathrm{Spec}\,\mathbb{Z},\{p,\infty\},V_{\mathrm{vac}}\otimes\lambda_{\infty})\)
is spanned by the Tate integral
\begin{equation}
\Phi_{p}(s)\;=\;\xi(s)\cdot(1-p^{-s}),
\end{equation}
where \(\xi(s)=\pi^{-s/2}\Gamma(s/2)\zeta(s)\) is the completed
Riemann zeta (Tate's thesis, \S3.4). Removing the \(p\)-th Euler factor
from \(\zeta(s)\) gives \(\zeta(s)(1-p^{-s})\); multiplication by
\(\pi^{-s/2}\Gamma(s/2)\) completes to \(\xi(s)(1-p^{-s})\).
Dimension: \(1\).
\end{computation}

\begin{observation}
\label{v4-arith:obs:vacuum-removes-Euler-factor}
The vacuum conformal block at prime \(p\) is the completed zeta function
with the \(p\)-th Euler factor removed. The sewing axiom at \(p\) reinserts it.
\end{observation}

\begin{computation}[sewing at \(\{2,3\}\)]
\label{v4-arith:comp:sewing-23}
The vacuum conformal block at marked places \(\{2,3,\infty\}\) is
\(\xi(s)(1-2^{-s})(1-3^{-s})\) (the completed zeta with the Euler
factors at \(2,3\) removed; \Cref{v4-arith:obs:vacuum-removes-Euler-factor}).
Sewing at the two finite places with unramified Satake parameters
\(\alpha_{p}=\chi_{p}(p)\in\{\pm 1\}\) reinserts the twisted Euler
factors:
\begin{equation*}
\Phi(s)\;=\;\xi(s)\,
\frac{(1-2^{-s})(1-3^{-s})}{(1-\alpha_{2}\cdot 2^{-s})(1-\alpha_{3}\cdot 3^{-s})},
\end{equation*}
where \(\xi(s)=\pi^{-s/2}\Gamma(s/2)\zeta(s)\). For trivial Satake
\((\alpha_{2},\alpha_{3})=(1,1)\) one recovers \(\Phi(s)=\xi(s)\); for
a non-trivial Dirichlet character \(\chi\) unramified at \(\{2,3\}\)
(conductor coprime to \(6\)) the result is the completed
\(L\)-function \(\Lambda(s,\chi)\) with the two twisted Euler factors
reinserted in place of the trivial ones.
\end{computation}

\section{Functional Equation as Sewing Axiom}
\label{v4-arith:segal:functional-equation}

\begin{theorem}[Tate's global functional equation]
\label{v4-arith:thm:tate-functional-equation}
\ClaimStatusProvedElsewhere.

Let \(\Lambda(s,\chi)=\pi^{-s/2}\Gamma(s/2)L(s,\chi)\) denote the
completed Dirichlet \(L\)-function. Then
\begin{equation}
\Lambda(s,\chi)\;=\;\varepsilon(\chi)\cdot\Lambda(1-s,\chi^{-1}),
\end{equation}
where \(\varepsilon(\chi)\) is the root number, proved by adelic Poisson
summation (Tate's thesis, \S4.4). The Segal sewing axiom is the
arithmetic functional equation.
\end{theorem}

\begin{remark}[finiteness reading]
The content of sewing is not dimension~\(1\) per se; it is the
\(s\leftrightarrow 1-s\) involution on the one-dimensional line
whose invariants are the values of the completed \(L\)-function.
\end{remark}

\section{Five-Axiom Arithmetic Modular Functor}
\label{v4-arith:segal:five-axioms}

\begin{definition}
\label{v4-arith:def:arithmetic-modular-functor}
An \emph{arithmetic modular functor} is a functor
\begin{equation*}
\mathcal{F}\colon\mathcal{C}^{\mathrm{Arak}}\to\mathrm{Hom}(\mathbb{C}_{s},\mathrm{Line})
\end{equation*}
(from Arakelov surfaces to lines depending meromorphically on a
parameter \(s\)) satisfying:

\paragraph{(A1) Finiteness.} \(\mathcal{F}(\overline{X},S,\{\chi_{\mathfrak{p}}\})\)
is a \(1\)-dimensional meromorphic line, with poles at \(s\in\{0,1\}\)
only.

\paragraph{(A2) Factorization.} For independent places,
\(\mathcal{F}(\overline{X},S,\{\chi_{\mathfrak{p}}\})=\bigotimes_{\mathfrak{p}\in S}\mathcal{F}_{\mathfrak{p}}(\chi_{\mathfrak{p}})\).

\paragraph{(A3) Weil--Deligne sewing.} For \(S=S_{1}\sqcup S_{2}\)
along a parameter \((\rho,N)\) at an extra prime \(\mathfrak{q}\),
\begin{equation*}
\mathcal{F}(\overline{X},S\cup\{\mathfrak{q}\})\;=\;\sum_{(\rho,N)}
\mathcal{F}(\overline{X}_{1},S_{1}\cup(\rho,N))\cdot
\mathcal{F}(\overline{X}_{2},S_{2}\cup(\rho,N))^{-1},
\end{equation*}
summed over unramified \((\rho,N)\).

\paragraph{(A4) Modular invariance.}
\(\mathrm{Gal}(\overline{K}/K)\ltimes GL_{n}(\mathbb{A}_{K})/GL_{n}(K)\)
acts on the functor.

\paragraph{(A5) Functional equation.}
\begin{equation*}
\mathcal{F}(\overline{X},S,\chi)(s)\;=\;\varepsilon(\chi)\cdot\mathcal{F}(\overline{X},S,\chi^{-1})(1-s).
\end{equation*}
This is the arithmetic modular invariance: \(\mathrm{SL}_{2}(\mathbb{Z})\)
becomes the \(\mathbb{Z}/2\) flip \(s\leftrightarrow 1-s\).
\end{definition}

\begin{theorem}[arithmetic Segal = Tate]
\label{v4-arith:thm:segal-tate-equivalence}
\ClaimStatusConjectured. The arithmetic modular functor
\(\mathcal{F}^{\mathrm{arith}}\) satisfying (A1)--(A5) exists and is
unique up to isomorphism; it is Tate's automorphic \(L\)-functor.
\end{theorem}

The geometric and arithmetic Segal functors are the same functor
viewed through different cohomology theories: de Rham for \(\Sigma_{g}\)
over \(\mathbb{C}\), and \'etale plus archimedean Hodge for
\(\overline{\mathrm{Spec}\,\mathcal{O}_{K}}\). The equals sign is a
theorem (\Cref{v4-arith:thm:segal-tate-equivalence}).
