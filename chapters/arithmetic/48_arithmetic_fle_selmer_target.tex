\chapter{Arithmetic Fundamental Local Equivalence: Selmer-Derived Target and Open Categorical Equivalences}
\label{v4-arith:fle-completion-table}

\section{Selmer-Derived Global Spectral Stack}
\label{v4-arith:fle-table:spectral-target}

A global spectral target must impose the global Galois relation and
its local conditions simultaneously. The formal deformation functors of
Chapter~\ref{v4-arith:coefficients:chapter} provide a precise pullback
on a common coefficient category. A geometric stack representing that
functor, and a comparison with local parameter stacks, require further
hypotheses.

\begin{definition}[Selmer deformation functor]
\label{v4-arith:fle-table:def:selmer-stack}
Fix a finite field \(k\) of characteristic \(\ell\). Choose a finite
ramification set \(S\) containing the archimedean place and \(\ell\).
Fix a continuous representation
\(\bar\rho:G_{\mathbb Q,S}\to\mathrm{GL}_2(k)\), and a determinant
lift \(\delta:G_{\mathbb Q,S}\to W(k)^\times\).
Use the category \(\mathrm{Art}_{W(k)}\) of augmented simplicial
Artinian local coefficient algebras from
\S\ref{v4-arith:coefficients:sec:deformations}.

Let \(\mathcal D_{G,\delta}^{\mathrm{or}}\) denote the derived
continuous deformation functor with an identification of its determinant
with \(\delta\). Arrows preserve that identification. For each finite
\(v\in S_f\), supply a pointed local functor
\(\mathcal D_v^{\mathcal L}\) and a morphism
\(\mathcal D_v^{\mathcal L}\to\mathcal D_{G_{\mathbb Q_v},\delta|_v}^{\mathrm{or}}\)
on this same coefficient category.

Define
\begin{equation}
\label{v4-arith:fle-table:eq:selmer-stack}
\mathcal X^{\mathrm{Sel}}_{\bar\rho,\delta,\mathcal L}
:=\mathcal D_{G_{\mathbb Q,S},\delta}^{\mathrm{or}}
\times^h_{\prod_{v\in S_f}\mathcal D_{G_{\mathbb Q_v},\delta|_v}^{\mathrm{or}}}
\prod_{v\in S_f}\mathcal D_v^{\mathcal L}.
\end{equation}
This notation initially denotes a functor to spaces. It denotes a
geometric stack only when such a representative has been constructed.
A local Emerton--Gee or Weil-parameter description can be used after
its comparison with the indicated local functor, on the specified
coefficient category, is supplied.
\end{definition}

\begin{proposition}[Selmer tangent complex]
\label{v4-arith:fle-table:prop:selmer-tangent}
At the residual point, suppose the tangent of each
\(\mathcal D_v^{\mathcal L}\) is \(L_v[1]\), with its induced map
\(g_v:L_v\to C^\bullet(G_{\mathbb Q_v},\mathrm{ad}^0\bar\rho)\).
Then
\begin{equation}
\label{v4-arith:fle-table:eq:selmer-tangent}
T\mathcal X^{\mathrm{Sel}}_{\bar\rho,\delta,\mathcal L}
\simeq C^\bullet_{\mathcal L}
       (G_{\mathbb Q,S},\mathrm{ad}^0\bar\rho)[1],
\end{equation}
where \(C^\bullet_{\mathcal L}\) is the mapping fiber of
\[
 C^\bullet(G_{\mathbb Q,S},\mathrm{ad}^0\bar\rho)
 \oplus\prod_{v\in S_f}L_v
 \xrightarrow{\mathrm{res}-g}
 \prod_{v\in S_f}C^\bullet(G_{\mathbb Q_v},\mathrm{ad}^0\bar\rho).
\]
\end{proposition}
\begin{proof}
The tangent functor preserves the homotopy pullback
\eqref{v4-arith:fle-table:eq:selmer-stack}.
Propositions~\ref{v4-arith:coefficients:prop:tangent}
and~\ref{v4-arith:coefficients:prop:determinant} compute its global
and full local tangent complexes. Their fiber is the displayed
Selmer complex, as in
Theorem~\ref{v4-arith:coefficients:thm:selmer-tangent}.
\end{proof}

Its cohomology groups \(H^0_{\mathcal L}\),
\(H^1_{\mathcal L}\), and \(H^2_{\mathcal L}\) occur in tangent
degrees \(-1,0,1\). They record infinitesimal automorphisms,
deformations, and obstructions. If each \(g_v\) is an inclusion of
complexes, a quotient by \(L_v\) gives an equivalent description.
A general morphism of local conditions requires the fiber above.
The trace-zero coefficients rely on preserving a determinant
identification, rather than merely its value.

Neither geometric representability nor Noetherianity follows from
this tangent formula. Quasi-smoothness is also a separate condition
on a geometric representative. For derived schemes it allows a
nonzero tangent group in degree one: the dual-number scheme of
Example~\ref{v4-arith:coefficients:ex:quasismooth} is an explicit
case. Real-place complexes have not been included in
\eqref{v4-arith:fle-table:eq:selmer-stack}. A compact-support
interpretation must specify them, especially for \(\ell=2\).

\section{Local Geometrisation and Its Domain}
\label{v4-arith:fle-table:local-geometrisation}

The following collects what Fargues--Scholze, Emerton--Gee, and
Dotto--Emerton--Gee establish in the local setting. These are the
inputs to the conditional programme of \S\ref{v4-arith:fle-table:conditions}.

\begin{theorem}[local inputs, Fargues--Scholze / Emerton--Gee / Dotto--Emerton--Gee]
\label{v4-arith:fle-table:thm:local-inputs}
\ClaimStatusProvedHere\ (assembling external results).
\begin{enumerate}
\item \emph{Local geometrisation and spectral action}
  (Fargues--Scholze arXiv:2102.13459). For each prime \(p\),
  the Fargues--Scholze spectral action equips
  \(D^{\mathrm{sph}\text{-}\mathrm{fin}}([GL_{2}(\mathbb{Q}_{p})])\)
  with an action of \(\mathrm{IndCoh}_{\mathrm{Nilp}}(\mathrm{LocSys}_{GL_{2}})\)
  and supplies local Langlands geometrisation and Hecke functoriality.

\item \emph{Local \((\varphi,\Gamma)\) parameter stacks}
  (Emerton--Gee arXiv:1908.07185). For each prime \(p\), the
  Emerton--Gee stack
  \(\mathcal{X}^{\mathrm{EG},p}_{GL_{2}}\) parametrises
  two-dimensional \((\varphi,\Gamma)\)-modules over \(\mathbb{Q}_{p}\)
  with a well-behaved moduli structure; local conditions
  \(\mathcal{X}^{\mathcal{L}_{v}}_{v}\hookrightarrow
  \mathcal{X}^{W,\varphi\Gamma}_{GL_{2},v}\) are open or closed
  substacks.

\item \emph{Fully faithful functor at fixed central character}
  (Dotto--Emerton--Gee arXiv:2603.26887). For primes
  \(p \geq 5\), there exists a fully faithful functor
  \[
  D^{\mathrm{sph}\text{-}\mathrm{fin}}_{GL_{2}(\mathbb{Q}_{p}),\chi}
  \longhookrightarrow
  \mathrm{IndCoh}_{\mathrm{Nilp}}\bigl(\mathcal{X}^{\mathrm{EG},p}_{GL_{2},\chi}\bigr)
  \]
  for each fixed central character \(\chi\). Essential surjectivity,
  the cases \(p = 2,3\), and the central-character-family
  statement remain open.

\item \emph{Archimedean OPE confined to archimedean factor}.
  The Hochschild-trace candidate \(\mathcal{V}^{\mathrm{prim}}\)
  carries chiral OPE structure at the archimedean factor
  \(v = \infty\) via the \((\mathfrak{gl}_{2},O(2))\)-module structure;
  finite places carry locally constant sheaf data and Hecke-algebra
  actions via the Satake isomorphism. Chiral OPE structure is an
  archimedean-fibre phenomenon.

\item \emph{Commutative Bethe--Hecke algebra}.
  At each finite prime \(p\), the centre of the spherical Hecke algebra
  \(\mathcal{H}(GL_{2}(\mathbb{Q}_{p}),GL_{2}(\mathbb{Z}_{p}))\)
  is commutative and acts on
  \(\mathcal{V}^{\mathrm{prim}}\) via the Satake isomorphism. This
  commutative Bethe--Hecke structure is the correct Hecke-algebraic
  shadow; higher Hopf or Yangian structures require separate
  verification at each prime.

\item \emph{Classical Kim arithmetic Chern--Simons}.
  Kim's arithmetic Chern--Simons theory arXiv:1510.05818 supplies
  classical arithmetic CS torsors on the arithmetic curve and action
  lines in \(K_{2}\). The quantum boundary category
  \(\partial_{q}\mathrm{ACS}\) of Conjecture~\ref{v4-arith:fle-table:conj:acs-boundary}
  is a distinct object from \(\mathcal{V}^{\mathrm{prim}}\);
  its construction is open.
\end{enumerate}
\end{theorem}

\section{Conditional Categorical Equivalences}
\label{v4-arith:fle-table:conditions}

The local inputs of \S\ref{v4-arith:fle-table:local-geometrisation}
establish the spectral action, the \((\varphi,\Gamma)\) parameter stacks,
and full faithfulness at fixed central character. A complete arithmetic FLE
requires four further categorical equivalences, each an open problem
conditional on a named hypothesis.

\begin{conjecture}[local categorical comparison]
\label{v4-arith:fle-table:conj:local-cat}
\ClaimStatusConjectured. For every prime \(p\) there is an equivalence
\begin{equation}
\label{v4-arith:fle-table:eq:local-cat}
D^{\mathrm{sph\text{-}fin}}\bigl([GL_{2}(\mathbb{Q}_{p})]\bigr)
\simeq
\mathrm{IndCoh}_{\mathrm{Nilp}}\bigl(\mathcal{X}^{\mathrm{EG},p}_{GL_{2}}\bigr),
\end{equation}
compatible with Hecke functors, central-character families, and the
Fargues--Scholze spectral action. The Dotto--Emerton--Gee functor
establishes full faithfulness for \(p \geq 5\) at fixed central
character; the conjecture adds essential surjectivity, the cases
\(p = 2, 3\), and the family statement.
\end{conjecture}

\begin{conjecture}[adelic categorical descent]
\label{v4-arith:fle-table:conj:adelic-descent}
\ClaimStatusConjectured.
Under the local categorical comparison
\eqref{v4-arith:fle-table:eq:local-cat} at all primes, the local
categories glue to the global automorphic category
\(D([GL_{2}])^{\mathrm{sph\text{-}fin}}\) via an equivalence
compatible with Whittaker models, Bernstein centres, the archimedean
\((\mathfrak{g},K)\) factor, and Eisenstein and continuous spectrum.
Representation-level restricted tensor products are classical
(Flath); the categorical descent is the open statement.
\end{conjecture}

\begin{conjecture}[mixed-characteristic Iwahori \(E_{2}\)]
\label{v4-arith:fle-table:conj:iwahori-e2}
\ClaimStatusConjectured. The mixed-characteristic local Iwahori
Hecke category
\(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}\)
carries an \(E_{2}\)-monoidal structure compatible with diamond
Beilinson--Drinfeld fusion and with the Iwahori stratification:
\begin{equation}
\label{v4-arith:fle-table:eq:iwahori-e2}
\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}
\in
\mathrm{Alg}_{E_{2}}\bigl(\mathrm{Cat}_{\infty}\bigr).
\end{equation}
Current sources supply spherical fusion and \(E_{1}\) convolution; the
\(E_{2}\) enhancement is the local input required for the
restricted-tensor \(E_{2}\) transport argument. The restricted-tensor
\(E_{2}\) transport lemma (proved elsewhere in this volume)
is a separate, purely categorical statement; its hypotheses require
\eqref{v4-arith:fle-table:eq:iwahori-e2} as input.
\end{conjecture}

\begin{conjecture}[full-target arithmetic Ben-Zvi--Nadler]
\label{v4-arith:fle-table:conj:full-bzn}
\ClaimStatusConjectured.
For the Selmer-derived full Weil\(/(\varphi,\Gamma)\) target, the
arithmetic Ben-Zvi--Nadler comparison is
\begin{equation}
\label{v4-arith:fle-table:eq:bzn}
\mathrm{HH}_{\bullet}\bigl(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}}\bigr)
\simeq
\mathcal{O}\bigl(\mathrm{LocSys}^{W,\varphi\Gamma}_{GL_{2}}\bigr)^{\mathrm{flat}},
\end{equation}
where \(\mathrm{LocSys}^{W,\varphi\Gamma}_{GL_{2}}\) is the full
Weil\(/(\varphi,\Gamma)\) local system stack and
\({\cdot}^{\mathrm{flat}}\) denotes the flat-locus subfunctor.
The conjecture requires formal-Noetherian Hochschild continuity for
\(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}}\) and
comparison of the Arinkin--Gaitsgory nilpotent flat locus with the
Fargues--Scholze/Emerton--Gee flat locus.
\end{conjecture}

\begin{conjecture}[categorical arithmetic Whittaker theory]
\label{v4-arith:fle-table:conj:whittaker}
\ClaimStatusConjectured.
The arithmetic Whittaker factorisation lifts from
representation-level Whittaker models to a categorical equivalence
compatible with the local comparisons of
\eqref{v4-arith:fle-table:eq:local-cat}, adelic descent, and the
Selmer-derived spectral stack
\eqref{v4-arith:fle-table:eq:selmer-stack}.
Representation-level Tate--Jacquet--Langlands factorisation is proved;
the categorical lift is the open statement.
\end{conjecture}

\begin{conjecture}[quantum ACS boundary category]
\label{v4-arith:fle-table:conj:acs-boundary}
\ClaimStatusConjectured. There exists a quantum boundary category
\(\partial_{q}\mathrm{ACS}\) and an equivalence
\[
\partial_{q}\mathrm{ACS} \simeq \mathcal{V}^{\mathrm{prim}}
\]
compatible with an appropriate quantisation of Kim's arithmetic
Chern--Simons theory arXiv:1510.05818. Kim's theory provides the
classical arithmetic CS torsors and action lines in \(K_{2}\); the
quantisation is the open content of this conjecture.
\end{conjecture}

\section{Proved and Conditional Components of the Arithmetic FLE}
\label{v4-arith:fle-table:summary}

\begin{theorem}[arithmetic FLE: proved and conditional components]
\label{v4-arith:fle-table:thm:summary}
\ClaimStatusProvedHere\ (by enumeration).
The arithmetic programme based on
\(\mathcal{V}^{\mathrm{prim}}\) and the Selmer-derived stack
\(\mathcal{X}^{\mathrm{Sel}}_{\bar\rho,\delta,\mathcal{L}}\)
\eqref{v4-arith:fle-table:eq:selmer-stack} decomposes as follows.
\begin{enumerate}
\item \emph{Constructed as a formal functor.} The Selmer homotopy
  pullback exists for the supplied local functors and maps on
  \(\mathrm{Art}_{W(k)}\)
  (Definition~\ref{v4-arith:fle-table:def:selmer-stack}). Its tangent
  is the stated Selmer complex under the local tangent hypotheses
  (Proposition~\ref{v4-arith:fle-table:prop:selmer-tangent}).
  Geometric representability and comparisons with the local parameter
  stacks remain additional requirements.

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:local-cat}.}
  Local categorical comparison at all primes, including essential
  surjectivity at \(p \geq 5\) and the full statement at \(p = 2, 3\).

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:local-cat} and
  Conjecture~\ref{v4-arith:fle-table:conj:adelic-descent}.}
  Adelic categorical descent to the global automorphic category.

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:iwahori-e2}.}
  Iwahori \(E_{2}\)-monoidal structure at mixed characteristic;
  the restricted-tensor \(E_{2}\) transport lemma applies once this
  local input is supplied.

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:full-bzn}.}
  Full-target arithmetic Ben-Zvi--Nadler comparison
  \eqref{v4-arith:fle-table:eq:bzn}.

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:whittaker}.}
  Categorical arithmetic Whittaker factorisation.

\item \emph{Conditional on
  Conjecture~\ref{v4-arith:fle-table:conj:acs-boundary}.}
  Quantum ACS boundary category equivalent to
  \(\mathcal{V}^{\mathrm{prim}}\).
\end{enumerate}
\(\mathcal{V}^{\mathrm{prim}}\) is a Hochschild trace class throughout;
its chiral OPE structure is confined to the archimedean factor
as established in Theorem~\ref{v4-arith:fle-table:thm:local-inputs}(4).
A vertex algebra requires an OPE on a formal disk defined at every
place; \(\mathcal{V}^{\mathrm{prim}}\) carries this structure only at
\(v = \infty\).
\end{theorem}
