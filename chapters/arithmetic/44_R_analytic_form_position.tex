\chapter{Moment Growth and Arithmetic Spectral Comparisons}
\label{v4-arith:R-analytic-comparison}
Let \(L:\mathbb C[x]\to\mathbb C\) be a specified positive Hamburger
functional with finite moments \(\mu_n=L(x^n)\). Positivity means
\(L(p\overline p)\geq0\) for every polynomial \(p\), where the bar
conjugates coefficients. The analytic condition is
\(\mu_{2n}^{1/(2n)}=O(n)\). Together with positivity it implies
determinacy. Its arithmetic application requires construction of the
regularized functional and its carrier, support and operator comparisons.

\section{Positive Functionals and Growth Estimates}
\label{v4-arith:r-comparison:statement}
\begin{definition}[The analytic moment condition]
\label{v4-arith:r-comparison:claim-1}
For the specified finite positive functional \(L\), write
\(\mathrm R_{\mathrm{analytic}}\) for
\(\limsup_{n\to\infty}\mu_{2n}^{1/(2n)}/n<\infty\).
A two-sided factorial estimate is stronger than this upper bound.
\end{definition}
\begin{remark}[The arithmetic application]
\label{v4-arith:r-comparison:claim-2}
The classical implication concerns determinacy of \(L\). Its arithmetic
application requires the functional, support and operator identifications.
No implication between RH and an unspecified regularization is asserted.
\end{remark}
\begin{proposition}[A logarithmic lower comparison]
\label{v4-arith:r-comparison:claim-3}
If the specified functional satisfies
\(\mu_{2n}\leq(2\pi)^{-4n}(2n)!(\log n)^{4n}(1+o(1))\),
its positive even moments have
\(\mu_{2n}^{-1/(2n)}\geq c/[n(\log n)^2]\) eventually for some
\(c>0\). This comparison series converges.
\end{proposition}
\begin{proof}
Apply Proposition~\ref{v4-arith:r-comparison:prop:BL-carleman-corrected} and
Corollary~\ref{v4-arith:r-comparison:cor:inconclusive} below.
\end{proof}
\begin{remark}[Three requirements]
\label{v4-arith:r-comparison:claim-paraphrase}
Existence and positivity of \(L\), an upper-growth estimate for its
moments, and its arithmetic comparison are separate premises.
No theorem of logical independence between them is asserted.
\end{remark}

\section{Factorial Roots}
\label{v4-arith:r-comparison:stirling-error}
The root \((2n)!^{1/(2n)}\) is asymptotic to \(2n/e\).
Taking the \(2n\)-th root of \((\log n)^{4n}\) gives
\((\log n)^2\), and inversion reverses that exponent.

\section{Li Coefficients and Polynomial Moments}
\label{v4-arith:r-comparison:BL-lambda-not-mu}
The Li summand \(1-(1-1/\rho)^n\) differs from the polynomial
summand \([\rho(1-\rho)]^n\). Their traces require different
summation and domain data. The termwise comparison is made below.

\section{The Conditional Operator Comparison}
\label{v4-arith:r-comparison:chain-position}
Under Hypothesis~\ref{v4-arith:hpvbconv:hyp:operator-quantitative-comparison},
\[
 \mathrm{VB}^{*}\Longleftrightarrow(\mathrm{Reg})
 \Longleftrightarrow(\mathrm{HB+})\Longleftrightarrow(\mathrm{BD+}).
\]
Here (BD+) is the sharp approximation predicate of
Chapter~\ref{v4-arith:HPAr-vbconverse}, not exponential completeness.
(Reg) includes the identified positive functional, determinacy and
spectral support. The classical moment estimate supplies determinacy;
the arithmetic identification, support and operator comparison remain
additional inputs.

\section{A Conditional Logarithmic Estimate}
\label{v4-arith:r-comparison:carleman-BL-explicit}
\subsection{The Assumed Bound}
\label{v4-arith:r-comparison:upper-bound}
Suppose the specified \(L\) satisfies
\begin{equation}
\label{v4-arith:r-comparison:eq:upper-bound}
 \mu_{2n}\leq(2\pi)^{-4n}(2n)!(\log n)^{4n}(1+o(1)).
\end{equation}
This is an assumption on the functional, not a consequence of
zero counting or a bound for Li coefficients.
\subsection{Stirling Evaluation}
\label{v4-arith:r-comparison:corrected-stirling}
\begin{lemma}[Factorial roots]
\label{v4-arith:r-comparison:lem:stirling}
\[
 (2n)!^{1/(2n)}=\frac{2n}{e}(4\pi n)^{1/(4n)}
 (1+O(n^{-2}))\sim\frac{2n}{e}.
\]
\end{lemma}
\begin{proof}
Take the \(2n\)-th root of
\((2n)!=\sqrt{4\pi n}(2n/e)^{2n}(1+O(n^{-1}))\).
The last factor becomes \(1+O(n^{-2})\), while
\((4\pi n)^{1/(4n)}\to1\).
\end{proof}
\begin{proposition}[A conditional lower comparison]
\label{v4-arith:r-comparison:prop:BL-carleman-corrected}
Assume \eqref{v4-arith:r-comparison:eq:upper-bound} and positive even moments. Then
\begin{equation}
\label{v4-arith:r-comparison:eq:mu-power}
 \mu_{2n}^{1/(2n)}\leq
 \frac{2n}{e(2\pi)^2}(\log n)^2(1+o(1)).
\end{equation}
For every \(0<c<e(2\pi)^2/2\), there is \(n_0\) such that
\begin{equation}
\label{v4-arith:r-comparison:eq:mu-inverse-lower}
 \mu_{2n}^{-1/(2n)}\geq\frac{c}{n(\log n)^2}\quad(n\geq n_0),
\end{equation}
so
\begin{equation}
\label{v4-arith:r-comparison:eq:carleman-bound}
 \sum_{n\geq n_0}\mu_{2n}^{-1/(2n)}
 \geq c\sum_{n\geq n_0}\frac1{n(\log n)^2}.
\end{equation}
\end{proposition}
\begin{proof}
Take roots of the assumed bound, apply Stirling and invert the
positive inequality. If instead an even moment of positive order
vanishes, any positive representing measure is supported at zero;
its prescribed mass makes it unique.
\end{proof}
\begin{corollary}[The lower comparison converges]
\label{v4-arith:r-comparison:cor:inconclusive}
\begin{equation}
\label{v4-arith:r-comparison:eq:bertrand}
 \sum_{n\geq2}\frac1{n(\log n)^2}<\infty.
\end{equation}
This convergent lower comparison does not establish divergence of
Carleman's sum.
\end{corollary}
\begin{proof}
The integral test gives
\(\int_2^\infty dx/[x(\log x)^2]=1/\log2\).
\end{proof}
\begin{remark}[Root scaling]
\label{v4-arith:r-comparison:rem:delta-error}
The exponent \(4n\) of the logarithm becomes \(2\) under the
\(2n\)-th root and \(-2\) under inversion. This calculation
uses only the displayed assumed estimate on \(L\).
\end{remark}
\begin{proposition}[A factorial upper bound implies determinacy]
\label{v4-arith:r-comparison:rem:heuristic-saves}
If \(L\) is positive with finite moments and
\(\mu_{2n}\leq A C^n(n!)^2\) eventually for constants \(A,C>0\),
then it is determinate. No matching lower bound is needed.
\end{proposition}
\begin{proof}
For positive even moments,
\(\mu_{2n}^{1/(2n)}\leq A^{1/(2n)}\sqrt C(n!)^{1/n}=O(n)\).
Carleman's sum dominates a positive multiple of the harmonic series.
A vanishing even moment gives the unique measure supported at zero.
\end{proof}

\section{Finite Cutoffs and Regularized Moments}
\label{v4-arith:r-comparison:gap-analysis}
\subsection{Distinct Zero Sums}
\label{v4-arith:r-comparison:can-BL-sharpen}
\begin{proposition}[Li summands and polynomial summands]
\label{v4-arith:r-comparison:prop:BL-vs-mu}
Under RH, for \(\rho=1/2+i\gamma\),
\[
 |1-(1-1/\rho)^n|\leq2,\qquad
 [\rho(1-\rho)]^n=(1/4+\gamma^2)^n.
\]
A bound for the summed Li expression does not, by this termwise
magnitude comparison, give finite unregularized polynomial moments.
\end{proposition}
\begin{proof}
The identity \(|1-1/\rho|=1\) gives the first inequality.
The polynomial identity gives the second, whose cutoff sums diverge
by the next proposition. An identity for the Li trace therefore
requires a separate comparison to a regularized polynomial functional.
\end{proof}
\begin{proposition}[Positive cutoff moments diverge]
\label{v4-arith:r-comparison:prop:closing-gap-is-density-RH}
Put \(S_n(T)=\int_{(0,T]}(1/4+t^2)^n\,dN(t)\). For fixed
integer \(n\geq0\),
\[
 S_n(T)=\frac{T^{2n+1}}{2\pi(2n+1)}\log\frac{T}{2\pi}
 -\frac{T^{2n+1}}{2\pi(2n+1)^2}+O_n(T^{2n}\log T).
\]
The integral over \((0,\infty)\) diverges. Under RH it is the
cutoff shifted-zero sum over positive ordinates.
\end{proposition}
\begin{proof}
Apply the Stieltjes calculation in
Proposition~\ref{v4-arith:w5-i:prop:R-analytic-density}.
The positive leading term tends to infinity.
\end{proof}
\begin{remark}[The leading term and regularization]
\label{v4-arith:r-comparison:rem:form-vs-difficulty}
Divergence already occurs in the leading zero-counting term.
Improving its remainder alone cannot make the positive integral
finite. A specified regularization is needed before its moment
growth can be studied.
\end{remark}
\subsection{Three Mathematical Requirements}
\label{v4-arith:r-comparison:axis-distinction}
One must construct the finite positive functional, prove a sufficient
growth estimate for that same functional, and identify it with the
arithmetic operator and its spectral support. None of these maps is
constructed by renaming a divergent cutoff sum.

\section{Determinacy, \(\mathrm{VB}^{*}\), and RH}
\label{v4-arith:r-comparison:chain-comparison}

The following propositions separate the classical determinacy implication
from the arithmetic operator comparison.

\subsection{Determinacy and the Operator Comparison}
\label{v4-arith:r-comparison:step-1}

\begin{proposition}[The predicate in the operator comparison]
\label{v4-arith:r-comparison:prop:det-converse-not-main}
Under Hypothesis~\ref{v4-arith:hpvbconv:hyp:operator-quantitative-comparison},
VB\(^{*}\) is equivalent to (Reg), not to bare (Det).
In the RH sector this supplies a conditional route to the
VB\(^{*}\) converse when (Reg) is constructed. The forward
arithmetic implication to zero placement retains its separate
P2, P3, determinant--trace and positivity hypotheses.
\end{proposition}
\begin{proof}
Use item~1 of the hypothesis and the definition of (Reg).
Neither its support condition nor its arithmetic moment
identification follows from uniqueness of an abstract measure.
\end{proof}

\subsection{Step 2: Determinacy vs RH}
\label{v4-arith:r-comparison:step-2}

\begin{proposition}[What the distance and determinacy results supply]
\label{v4-arith:r-comparison:prop:det-strictly-finer}
The classical BD criterion is convergence of its finite distances
to zero. The supplied-rate transport of
Corollary~\ref{v4-arith:hpvbconv:cor:rate-transport} transfers a
quantitative estimate only after that estimate and the scaled
isometry are given. Separately, a positive moment functional with
the Carleman growth bound is determinate. These statements alone
supply neither the arithmetic support condition nor an equivalence
between bare (Det) and RH.
\end{proposition}
\begin{proof}
The distance identity multiplies each finite norm by the fixed
positive scale; it does not estimate its dependence on the cutoff.
Carleman's theorem concerns uniqueness for the supplied moment
functional. Applying either result to an arithmetic operator
requires the stated comparison data.
\end{proof}

\subsection{The Implication Diagram}
\label{v4-arith:r-comparison:corrected-chain}
\begin{remark}[The moment and operator steps]
\label{v4-arith:r-comparison:diag:chain}
The classical step is
\[
 \mathrm R_{\mathrm{analytic}}\Longrightarrow
 \text{Carleman criterion}\Longrightarrow\mathrm{Det}.
\]
The arithmetic step, with the identified positive moment functional,
is
\[
 \mathrm{Det}+\text{specified support}
 \Longrightarrow(\mathrm{Reg})
 \xrightarrow{\text{operator comparison}}\mathrm{VB}^{*}.
\]
The conditional forward arithmetic theorem then gives RH with its
remaining hypotheses. The operator comparison is item~1 of
Hypothesis~\ref{v4-arith:hpvbconv:hyp:operator-quantitative-comparison}.
\end{remark}
\begin{remark}[The role of the moment estimate]
\label{v4-arith:r-comparison:rem:closes-converse}
The growth estimate proves the determinacy component. It does not
construct the arithmetic carrier, its support or operator comparison.
Those are additional steps even when the zero-placement statement
is assumed. No independence theorem between RH and this moment
estimate is asserted by this implication diagram.
\end{remark}


\section{Determinacy Consequences}
\label{v4-arith:r-comparison:conclusion}
\begin{theorem}[Moment estimates and arithmetic comparisons]
\label{v4-arith:r-comparison:thm:conclusion}
For a specified positive finite Hamburger functional:
\begin{enumerate}
\item Linear even-moment root growth implies determinacy.
\item A factorial upper bound suffices for that growth.
\item The logarithmic estimate \eqref{v4-arith:r-comparison:eq:upper-bound} gives the
convergent lower comparison \eqref{v4-arith:r-comparison:eq:carleman-bound}.
\item Its arithmetic application retains the functional, support
and operator comparisons of Remark~\ref{v4-arith:r-comparison:diag:chain}.
\end{enumerate}
\end{theorem}
\begin{proof}
The first assertion is Carleman's harmonic comparison; the second
is Proposition~\ref{v4-arith:r-comparison:rem:heuristic-saves}; the third is
Proposition~\ref{v4-arith:r-comparison:prop:BL-carleman-corrected}. The fourth is the
stated domain of the arithmetic comparison.
\end{proof}
\begin{corollary}[The moment estimate and the arithmetic target]
\label{v4-arith:r-comparison:cor:hypothesis-also-wrong}
The implication from \(\mathrm R_{\mathrm{analytic}}\) to
Hamburger determinacy does not by itself establish RH. The stated
route also requires the arithmetic moment identification, spectral
support and operator comparison, together with the hypotheses of
the forward arithmetic theorem. No converse non-implication is
claimed.
\end{corollary}
\begin{proof}
The moment estimate supplies the first line of
Remark~\ref{v4-arith:r-comparison:diag:chain}. The remaining arrows
have the additional hypotheses displayed there.
\end{proof}


\section{Examples and the Arithmetic Construction}
\label{v4-arith:r-comparison:honest-reduction}
\begin{proposition}[Positive functionals at two growth scales]
\label{v4-arith:r-comparison:thm:honest-reduction}
The probability measure \(\frac12\mathbf1_{[-1,1]}dx\) has
\(\mu_{2n}=1/(2n+1)\) and odd moments zero. It satisfies
\(\mathrm R_{\mathrm{analytic}}\), but no eventual lower bound
\(\mu_{2n}\geq C_-^n(n!)^2\) with \(C_->0\).
For \(a>0\), the probability measure \(\frac a2e^{-a|x|}dx\)
has \(\mu_{2n}=(2n)!/a^{2n}\) and odd moments zero. Its even-moment
roots are asymptotic to \(2n/(ae)\), so it is determinate.
\end{proposition}
\begin{proof}
Integrating on \([-1,1]\) gives the first formula. Its \(2n\)-th
root tends to one, whereas that of \(C_-^n(n!)^2\) grows as
\(\sqrt{C_-}n/e\). For the second measure, symmetry and the
substitution \(u=ax\) give
\[
 a\int_0^\infty x^{2n}e^{-ax}dx
 =a^{-2n}\int_0^\infty u^{2n}e^{-u}du=a^{-2n}(2n)!.
\]
Stirling gives linear root growth and hence Carleman determinacy.
Neither example is identified with the arithmetic functional.
\end{proof}
\begin{question}[The arithmetic moment construction]
\label{v4-arith:r-comparison:rem:what-delta-should-say}
Construct a specified regularization of the shifted-zero polynomial
sums, including its counterterms and normalization, defining a finite
positive Hamburger functional \(L\). Prove
\(L(x^{2n})^{1/(2n)}=O(n)\), or another sufficient determinacy
criterion, for this same functional. Identify its representing
measure and support with the arithmetic carrier and construct the
operator comparison. A two-sided factorial asymptotic is a stronger
possible target, not a premise needed for the Carleman implication.
\end{question}
\begin{remark}[Other arithmetic inputs]
\label{v4-arith:r-comparison:rem:not-sharpen}
The moment theorem supplies neither the weight-selection nor the
explicit-formula hypotheses. These remain separate requirements
in the sufficient arithmetic argument.
\end{remark}
\section{Arithmetic Consequences}
\label{v4-arith:r-comparison:summary}
A constructed positive functional and its linear root bound give
(Det). The specified support and arithmetic identification give
(Reg); the supplied operator comparison then transfers positivity.
The full arithmetic conclusion retains the remaining forward-theorem
hypotheses. The finite cutoff calculation, the classical examples and
the conditional estimates do not construct the missing regularization
or its arithmetic comparison.
