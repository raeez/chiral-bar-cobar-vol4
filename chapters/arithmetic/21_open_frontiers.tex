\chapter{Boundary Conjectures for the Six-Window Trace}
\label{v4-arith:open-frontiers}

\emph{The arithmetic trace class is controlled by three boundary
conjectures. They are not consequences of one another. F1 gives the
global spectral category, F2 gives the mixed-characteristic
\(E_{2}\)-monoidal enhancement, and F3 identifies the resulting
Hochschild trace with \(\mathcal{V}^{\mathrm{prim}}\). Local theorems
enter each boundary, but the displayed global equivalences are separate
inputs.}

\emph{Chapter~\ref{v4-arith:closure} resolves the six-window trace into
\(\mathrm{L}_{\mathrm{ACD}}\), F2.c, F3.BZN\(^{\mathrm{Ar}}\),
arith.~Positselski, and Koszul--Pet compatibility. The
Riemann-Hypothesis branch of Chapter~\ref{v4-arith:rh-reformulation}
is orthogonal to this trace construction except where the explicit
spectral positivity hypotheses are imposed.}

\section{F1: Global Arithmetic Arinkin--Gaitsgory}
\label{v4-arith:of:F1}

\begin{conjecture}[global arithmetic AG]
\label{v4-arith:conj:F1-global-AG}
For \(G=GL_{2}\), there is an equivalence of dg-categories
\begin{equation}
\mathrm{D}([GL_{2}])^{\mathrm{sph\text{-}fin}}\;\simeq\;\mathrm{IndCoh}_{\mathrm{Nilp}}\bigl(\mathcal{X}^{\mathrm{Sel}}_{\bar\rho,\det,\mathcal{L}}\bigr),
\end{equation}
where:
\begin{itemize}
\item \([GL_{2}]:=GL_{2}(\mathbb{Q})\backslash GL_{2}(\mathbb{A}_{\mathbb{Q}})/Z(\mathbb{A})K_{\infty}\)
is the global automorphic double coset;
\item \(\mathcal{X}^{\mathrm{Sel}}_{\bar\rho,\det,\mathcal{L}}\) is
the Selmer-derived global deformation stack obtained by imposing a
fixed residual representation \(\bar\rho\), determinant, and local
conditions \(\mathcal{L}_{v}\) on the full Weil/\((\varphi,\Gamma)\)
local parameter stacks; it is not a product or colimit of local
Emerton--Gee stacks;
\item \(\mathrm{IndCoh}_{\mathrm{Nilp}}\) denotes ind-coherent sheaves
with nilpotent singular support (Arinkin--Gaitsgory arXiv:1201.6343).
\end{itemize}
Under the equivalence, \(\mathcal{V}^{\mathrm{prim}}\) corresponds to
the structure sheaf \(\mathcal{O}\) of this Selmer spectral stack.
\end{conjecture}

\begin{remark}[form]
The local theorems do not imply the displayed global equivalence.
Fargues--Scholze arXiv:2102.13459 construct the local geometrisation,
spectral action, and Langlands-parameter space via the
Fargues--Fontaine curve and diamonds. Emerton--Gee arXiv:1908.07185
construct the local \((\varphi,\Gamma)\)-module stacks (with arXiv:2012.12403 as the survey companion). These sources
do not prove the categorical equivalence displayed in F1.
Dotto--Emerton--Gee arXiv:2603.26887 gives a fully faithful
\(GL_{2}(\mathbb{Q}_{p})\) functor for \(p\geq 5\) with fixed central
character; it is not an equivalence and it is not a family theorem
over all central characters.
Caraiani--Scholze arXiv:1511.02418 prove a unitary-Shimura-variety
result in a different global setting.

Thus F1 is a global categorical assertion for \(GL_{2}/\mathbb{Q}\),
not a restatement of the local Fargues--Scholze or Emerton--Gee
theorems. Under F1, \(\mathcal{V}^{\mathrm{prim}}\) is a spectral
object on the Selmer deformation stack, compatible with its
Hochschild-trace presentation of
Chapter~\ref{v4-arith:hochschild-trace}.
\end{remark}

\section{F2: \(E_{2}\)-Enhancement via Scholze Diamonds}
\label{v4-arith:of:F2}

\begin{conjecture}[perfectoid \(E_{2}\)-enhancement]
\label{v4-arith:conj:F2-perfectoid-E2}
The local Iwahori Hecke category \(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}\)
carries a canonical \(E_{2}\)-monoidal structure coming from the
Beilinson--Drinfeld Grassmannian over \(\mathrm{Spec}\,\mathbb{Z}_{p}\)
in Scholze's perfectoid (diamond) setting. Globally, the product
\(\bigotimes'_{p}\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}\) inherits a
well-defined \(E_{2}\)-monoidal structure.
\end{conjecture}

\begin{remark}[trace role]
Dunn additivity \(E_{1}\otimes E_{1}=E_{2}\) applied to the
\(E_{2}\)-monoidal structure gives the \(E_{1}\)-algebra structure on
the Hochschild trace \(\mathrm{HH}_{\bullet}(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}})\);
the \(E_{1}\) Hochschild-trace structure on \(\mathcal{V}^{\mathrm{prim}}\)
at the categorical-trace level depends on this enhancement.
\end{remark}

\begin{remark}[form]
The Beilinson--Drinfeld Grassmannian as
a perfectoid space is constructed in Fargues--Scholze
arXiv:2102.13459. Zhu's Witt-vector Grassmannian
(arXiv:1603.05593; Bhatt--Scholze arXiv:1507.06490) gives the
equal-characteristic analogue. The mixed-characteristic
Iwahori-Hecke \(E_{2}\) enhancement, as opposed to \(E_{1}\) convolution
or spherical fusion, remains conditional.

The global assertion is the Scholze-diamond gluing across primes,
producing a restricted-tensor-product \(E_{2}\)-category.
\end{remark}

\section{F3: Arithmetic Ben-Zvi--Nadler}
\label{v4-arith:of:F3}

\begin{conjecture}[arithmetic Ben-Zvi--Nadler]
\label{v4-arith:conj:F3-arith-BZN}
There is an equivalence of \(E_{1}\)-algebras
\begin{equation}
\mathrm{HH}_{\bullet}\bigl(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}}\bigr)\;\xrightarrow{\sim}\;\mathcal{V}^{\mathrm{prim}}
\end{equation}
realising the Chern--Simons--Hecke-trace-boundary triangle arithmetically:
Kim (arXiv:1510.05818) arithmetic Chern--Simons supplies a classical
boundary shadow; the Hochschild trace targets the full
Weil/\((\varphi,\Gamma)\) spectral category.
\end{conjecture}

\begin{remark}[form]
Ben-Zvi--Nadler
arXiv:0904.1247 establishes
\(\mathrm{HH}_{\bullet}(\mathrm{D}(G/G))=\mathcal{O}(\mathrm{LocSys}_{G}(S^{1}))^{\mathrm{flat}}\)
for complex reductive \(G\).

The arithmetic adaptation requires:
\begin{enumerate}
\item Conjecture F1 (global arithmetic AG);
\item Conjecture F2 (perfectoid \(E_{2}\)-enhancement);
\item Explicit computation of the Hochschild trace on arithmetic
\(\mathrm{Hk}^{\mathrm{Iw}}_{\mathrm{glob}}\) matching \(\mathcal{V}^{\mathrm{prim}}\)'s
\(L\)-function character.
\end{enumerate}

F3 identifies Gaiotto's CS presentation
(\Cref{v4-arith:thm:arith-CS-VA-correspondence}) with
Bezrukavnikov--Bernstein's Hochschild presentation
(\Cref{v4-arith:thm:crowning-placement}) with the bar-cobar
presentation (\Cref{v4-arith:thm:bar-verifications}) via a single
arithmetic Ben-Zvi--Nadler equivalence.
\end{remark}

\section{Further Boundary Conditions F4--F11}
\label{v4-arith:of:further-frontiers}

\subsection{F4: \(\mathsf{B}\)-Row Arithmetic Realisation}
\label{v4-arith:of:F4-B-row}

\begin{conjecture}[arithmetic \(\mathsf{B}\)-row ceiling]
\label{v4-arith:conj:F4-B-row}
There exists an arithmetic avatar of the Vol~III Mukai-enhanced
K3 Heisenberg such that \(\kappa^{\mathrm{Ar}}+(\kappa^{!})^{\mathrm{Ar}}=8\)
arithmetically. The witness is a Bruinier Heegner Chern-class
reciprocity on a \(p\)-adic Shimura variety, at each prime \(p\) of
good reduction.
\end{conjecture}

\begin{remark}
Ingredients available: Kudla--Rapoport arithmetic intersection
numbers on unitary Shimura varieties; Bruinier's singular-theta
lifts; Borcherds products on \(\mathrm{Sp}_{4}(\mathbb{Z})\). Open:
integral assembly into an arithmetic \(\mathsf{B}\)-row theorem.
\end{remark}

\subsection{F5: Arithmetic Chain Homotopy \(h^{\mathrm{Ar}}_{A}\)}
\label{v4-arith:of:F5-chain-homotopy}

\begin{conjecture}[universal arithmetic chain homotopy]
\label{v4-arith:conj:F5-chain-homotopy}
There exists a universal chain homotopy \(h^{\mathrm{Ar}}_{A}\) on
the arithmetic bar complex \(B^{\mathrm{ch,Ar}}(\mathcal{V}^{\mathrm{prim}})\)
satisfying
\begin{equation}
h^{\mathrm{Ar}}_{A}\;=\;\frac{h^{\mathrm{LV}}_{A}}{\kappa+\kappa^{!}}
\end{equation}
(where \(h^{\mathrm{LV}}_{A}\) is the Losev--Vol~III universal
chain homotopy), with obstruction class \(\mathrm{obs}^{(1)}\in H^{2}(\mathfrak{grt}^{\mathrm{ell}},\mathfrak{sp}\otimes\mathfrak{sp})\)
(where \(\mathfrak{grt}^{\mathrm{ell}}\) is the elliptic
Grothendieck--Teichmüller Lie algebra and \(\mathfrak{sp}\) the symplectic
Lie algebra in the relevant rank) vanishing iff \(r_{\max}=2\).
\end{conjecture}

\subsection{F6: Chenevier Determinant at \(p=691\)}
\label{v4-arith:of:F6-chenevier-691}

\begin{conjecture}[Chenevier determinant identity]
\label{v4-arith:conj:F6-chenevier-691}
At \(p=691\) (the prime at which Ramanujan's cusp form \(\Delta_{12}\)
and the Eisenstein series are congruent mod \(691\)),
\begin{equation}
\det_{\mathrm{Chenevier}}\bigl|_{p=691}\;=\;\chi_{\mathrm{cyc}}^{11}\cdot\varepsilon_{B_{12}},
\end{equation}
where \(\chi_{\mathrm{cyc}}\) is the cyclotomic character, \(\varepsilon_{B_{12}}\)
is the mod-\(691\) Kummer character attached to the numerator of the
Bernoulli number \(B_{12}\).
\end{conjecture}

\begin{remark}[connection]
This is the arithmetic avatar of the Chenevier determinant
appearing in the Vol~III universal trace identity. The prime
\(p=691\) is the seed of the arithmetic \(\mathsf{B}\)-row structure
(Chapter~\ref{v4-arith:hochschild-trace},
\Cref{v4-arith:thm:crowning-placement}).
\end{remark}

\subsection{F11: All-Genus Arithmetic Koszul Duality}
\label{v4-arith:of:F11-all-genus}

\begin{conjecture}[all-genus arithmetic Koszul duality]
\label{v4-arith:conj:F11-all-genus}
For every \(g\geq 0\) and every arithmetic genus-\(g\) datum
\(\mathcal{D}_g=(\mathcal{B}_g,\mathcal{E}_g,\varphi_g)\)
(\Cref{v4-arith:ag:defn:datum}), the Heisenberg Koszul self-duality
identity
\[
\kappa^{\mathrm{Ar},g}_{\mathsf{G}}(\mathcal{D}_g)+(\kappa^{!})^{\mathrm{Ar},g}_{\mathsf{G}}(\mathcal{D}_g)=0
\]
is equivalent to the motivic functional equation
\(\Lambda_{\mathcal{D}_g}(s)=\epsilon_{\mathcal{D}_g}\Lambda_{\mathcal{D}_g}(w_{\mathcal{D}_g}-s)\)
with \(w_{\mathcal{D}_g}\in\{1,2,2g+1\}\) the motivic weight center
indexed by window (Dedekind; Hasse--Weil on the Jacobian; Siegel spin
on \(\mathrm{GSp}_{2g}\)).
\end{conjecture}

\begin{remark}[form]
The forward direction is the Euler-product and Poisson-summation
direction in the window where analytic continuation and the functional
equation are already part of the input: Tate 1950 and Weil 1967 for
number fields, Bloch--Kato 1990 for weight-\(1\) motivic factors, and
Arthur 2013 for the automorphic Siegel-spin packet after transfer.
The converse direction requires the 2-dim arithmetic Positselski equivalence with
\(\mu_1=\mu_2=0\) on \(\Lambda_2=\mathbb{Z}_p[[\Gamma_1\times\Gamma_2]]\)
(extending Chapter~\ref{v4-arith:arith-positselski}) plus rank-\(g\)
Koszul--Petersson compatibility (extending Chapter~\ref{v4-arith:kp-compat}).
Ten residual conditions (\Cref{v4-arith:ag:residual}) enumerate the
substantial domain.

The datum \(\mathcal{D}_g\) admits
six convergent presentations (Arakelov fibration; Minkowski torus;
ordinary crystalline Emerton--Gee stack; arithmetic Teichmüller with
Hodge--Arakelov; chiral-bar-cobar interior; Shimura--Siegel
Kudla--Rapoport); see \Cref{v4-arith:ag:windows}.

The \(g=0\) specialisation is
Vol~IV's P1 resolution (\Cref{v4-arith:thm:P1-resolution}), and the
all-genus F-RH corollary (\Cref{v4-arith:ag:cor:all-genus-FRH})
places the Riemann, Hasse--Weil, Dedekind, and Siegel-spin
zero-placement clauses in a single rank-\(g\) form, still conditional
on the corresponding spectral positivity input.
\end{remark}

\section{Dependency Lattice}
\label{v4-arith:of:dependency-graph}

The three trace conjectures are not linearly ordered. The implication
lattice is:
\begin{center}
\begin{tikzcd}[column sep=large]
\text{F1: global spectral category} \ar[dr] & &
\text{F2: mixed-characteristic }E_{2}\text{-structure} \ar[dl] \\
& \text{F3: arithmetic BZN trace} \ar[d] & \\
& \text{six-window trace synthesis} &
\end{tikzcd}
\end{center}
F1 supplies the spectral-side category; F2 supplies the
\(E_{2}\)-enhancement making Hochschild trace \(E_{1}\)-algebraic; F3
identifies that trace with \(\mathcal{V}^{\mathrm{prim}}\). With these
three inputs, the arithmetic-branch synthesis
(\Cref{v4-arith:thm:six-window-placement}) becomes a theorem.

\section{Implication to the Trace Statement}
\label{v4-arith:of:chain-of-implications}

The six-window trace theorem is conditional on the conjunction
\[
\mathrm{F1}\ \wedge\ \mathrm{F2}\ \wedge\ \mathrm{F3}.
\]
The \(\mathsf{G}\)-row functional-equation theorem
\Cref{v4-arith:thm:G-row-Riemann-functional-equation} is an
independent Hochschild-cochain computation. It does not imply F1 or
F2, and F1--F3 do not imply it without the arithmetic
Koszul-resolution calculation.

\section{Boundary Index}
\label{v4-arith:of:summary}

\begin{enumerate}
\item F1: global arithmetic Arinkin--Gaitsgory
(\Cref{v4-arith:conj:F1-global-AG}): local FS/EG inputs exist; the
local categorical comparison and global descent remain open.
\item F2: perfectoid \(E_{2}\)-enhancement
(\Cref{v4-arith:conj:F2-perfectoid-E2}): spherical/equal-characteristic
inputs exist; mixed-characteristic Iwahori \(E_{2}\) remains
conditional.
\item F3: arithmetic Ben-Zvi--Nadler
(\Cref{v4-arith:conj:F3-arith-BZN}): theorem over \(\mathbb{C}\),
conjecture over \(\mathbb{Z}\).
\item F4: arithmetic \(\mathsf{B}\)-row ceiling \(\kappa+\kappa^{!}=8\)
(\Cref{v4-arith:conj:F4-B-row}).
\item F5: universal arithmetic chain homotopy
\(h^{\mathrm{Ar}}_{A}=h^{\mathrm{LV}}_{A}/(\kappa+\kappa^{!})\)
(\Cref{v4-arith:conj:F5-chain-homotopy}).
\item F6: Chenevier determinant at \(p=691\)
(\Cref{v4-arith:conj:F6-chenevier-691}).
\item F11: all-genus arithmetic Koszul duality
(\Cref{v4-arith:conj:F11-all-genus}): the identity
\(\kappa^{\mathrm{Ar},g}+(\kappa^{!})^{\mathrm{Ar},g}=0\Leftrightarrow\Lambda_{\mathcal{D}_g}(s)=\epsilon_{\mathcal{D}_g}\Lambda_{\mathcal{D}_g}(w_{\mathcal{D}_g}-s)\)
with \(w_{\mathcal{D}_g}\in\{1,2,2g+1\}\); the converse direction is
conditional on 2-dim Positselski and rank-\(g\)
Koszul--Petersson. The arithmetic-gravity boundary is
Chapter~\ref{v4-arith:3d-acs-E1bar}.
\item F7--F10: shadow-tower depth, critical-level degeneration,
super Bethe-Hecke chain-level, CY-C conditional bridges.
\end{enumerate}

Each item above is a mathematical boundary condition. The six-window
trace is the object cut out by F1, F2, and F3; the RH branch enters
only through the separately stated spectral-positivity hypotheses.
