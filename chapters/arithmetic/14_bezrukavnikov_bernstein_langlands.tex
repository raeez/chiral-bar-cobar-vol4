\chapter{The Hochschild Trace of the Iwahori Hecke Category}
\label{v4-arith:hochschild-trace}

The primary object is \(\mathcal{V}^{\mathrm{prim}}\), the Hochschild
trace class of the identity on the global arithmetic Iwahori Hecke
category of \(GL_2\). This chapter places \(\mathcal{V}^{\mathrm{prim}}\)
on the arithmetic Langlands diamond via Bezrukavnikov's affine Hecke
equivalences (arXiv:1209.0403), Mirko\-vi\'c--Vilonen geometric Satake
(arXiv:math/0401222), Ben-Zvi--Nadler character theory
(arXiv:0904.1247), Arinkin--Gaitsgory spectral decomposition
(arXiv:1201.6343), the Emerton--Gee \((\varphi,\Gamma)\) stack
(arXiv:1908.07185), and Fargues--Scholze geometrization
(arXiv:2102.13459).

\section{The Naive \(\mathrm{Bun}_{GL_{2}}(\mathrm{Spec}\,\mathbb{Z})\) Is a Point}
\label{v4-arith:bb:bun-is-point}

\begin{theorem}[Minkowski]
\label{v4-arith:thm:minkowski}
\ClaimStatusProvedElsewhere. \(\pi_{1}^{\mathrm{\acute{e}t}}(\mathrm{Spec}\,\mathbb{Z})=1\):
every \'etale cover of \(\mathrm{Spec}\,\mathbb{Z}\) is trivial. Class
number of \(\mathbb{Q}\) is \(1\).
\end{theorem}

\begin{corollary}
\label{v4-arith:cor:Bun-GL2-is-point}
\ClaimStatusProvedHere. \(\mathrm{Bun}_{GL_{2}}(\mathrm{Spec}\,\mathbb{Z})=\{\mathbb{Z}^{2}\}\),
a single isomorphism class of rank-\(2\) projective \(\mathbb{Z}\)-modules.
\(\mathcal{D}\)-modules on this groupoid are vector spaces; the category
is semisimple of rank \(1\) over the coefficient field, with no eigenvalue
decomposition under Hecke operators.
\end{corollary}

\section{The Spherical Hecke Category Collapses the OPE}
\label{v4-arith:bb:spherical-collapses-OPE}

The naive automorphic category for \(\mathrm{Bun}_{GL_2}(\mathrm{Spec}\,\mathbb{Z})\)
being a point forces the question: which local Hecke category at a prime \(p\)
carries enough noncommutative structure for a nontrivial Hochschild trace?
Geometric Satake identifies the spherical category but reveals its commutativity.

\begin{theorem}[Mirkovi\'c--Vilonen]
\label{v4-arith:thm:mirkovic-vilonen}
\ClaimStatusProvedElsewhere. Geometric Satake gives
\(\mathrm{Perv}_{GL_{2}(\mathcal{O})}(\mathrm{Gr}_{GL_{2}})\xrightarrow{\sim}\mathrm{Rep}(\widehat{GL_{2}})=\mathrm{Rep}(GL_{2})\).
\end{theorem}

\begin{observation}[spherical Hecke commutativity]
\label{v4-arith:obs:type-mismatch-perv-VA}
The local spherical Hecke algebra \(\mathcal{H}_{p}=C_{c}(GL_{2}(\mathbb{Z}_{p})\backslash
GL_{2}(\mathbb{Q}_{p})/GL_{2}(\mathbb{Z}_{p}))\) is commutative: this is the content
of the Satake isomorphism \(\mathcal{H}_p \xrightarrow{\sim} \mathbb{C}[GL_2^\vee]^{GL_2^\vee}\).
Its derived enhancement \(\mathcal{H}^{(p)}\), an \(E_{1}\)-algebra in chain complexes
(hence \(E_{2}\) as a graded object under Dunn additivity), and \(\mathrm{Perv}(\mathrm{Gr})\),
a monoidal abelian category under convolution, are related by the Mirković--Vilonen equivalence;
but the commutativity of \(\mathcal{H}_p\) means the resulting Hochschild homology carries no
automorphic spectral content beyond the centre.
\end{observation}

\section{The Iwahori Hecke Category Carries Convolution Data}
\label{v4-arith:bb:iwahori-carries-OPE}

Before invoking Bezrukavnikov's equivalence we record the Bernstein
presentation that gives the chapter its name. The affine Hecke
algebra \(H_{q}=H_{q}(\widetilde{W})\) of the extended affine Weyl
group \(\widetilde{W}=W_{0}\ltimes X^{*}(T)\) (with \(W_{0}\) the
finite Weyl group and \(X^{*}(T)\) the character lattice of the
maximal torus) is generated by:
\begin{itemize}
\item lattice translations \(\theta_{\lambda}\) for
\(\lambda\in X^{*}(T)\), forming the Bernstein subalgebra
\(\mathbb{C}[X^{*}(T)]\subset H_{q}\);
\item finite Hecke generators \(T_{s}\) for each simple reflection
\(s\in W_{0}\), satisfying the braid and quadratic
\((T_{s}-q)(T_{s}+1)=0\) relations.
\end{itemize}
The interaction between the two layers is the \emph{Bernstein
relation}: for a simple reflection \(s=s_{\alpha}\) with simple
coroot \(\alpha^{\vee}\) and \(\lambda\in X^{*}(T)\) orthogonal to
\(\alpha^{\vee}\),
\[
T_{s}\,\theta_{\lambda}\,T_{s}^{-1}\;=\;\theta_{s(\lambda)},
\]
while for general \(\lambda\) the conjugation acquires a correction
term proportional to \((\theta_{\lambda}-\theta_{s(\lambda)})/(1-\theta_{-\alpha})\)
expressing the failure of \(\theta_{\lambda}\) to be \(s\)-invariant
(Lusztig 1989, \emph{Affine Hecke algebras and their graded version},
J.~AMS~2, eq.~(8.4)). The Bezrukavnikov equivalence below sends this
noncommutative algebra structure to a coherent-sheaf convolution on
the Steinberg variety; it is the categorical lift of the Bernstein
presentation, and the chapter's framing "Bezrukavnikov--Bernstein--Langlands"
records the three layers.

The commutative spherical category is replaced by the Iwahori-equivariant
category, whose noncommutative convolution datum is precisely what Bezrukavnikov's
equivalences encode on the coherent side.

\begin{theorem}[Bezrukavnikov arXiv:1209.0403]
\label{v4-arith:thm:bezrukavnikov}
\ClaimStatusProvedElsewhere. Let \(\widetilde{\mathcal{N}}\) denote the Springer resolution
\(\widetilde{\mathcal{N}}=T^{*}(GL_{2}/B)\) of the nilpotent cone, and let
\(\mathrm{St}_{GL_{2}}=\widetilde{\mathcal{N}}\times_{\mathcal{N}}\widetilde{\mathcal{N}}\)
denote the Steinberg variety (the derived fibre product over the nilpotent cone). Equivalences
\begin{equation*}
\mathrm{D}^{b}\mathrm{Coh}^{GL_{2}\times\mathbb{G}_{m}}(\widetilde{\mathcal{N}})\xrightarrow{\sim}\mathrm{D}^{b}\mathrm{Coh}(\mathrm{St}_{GL_{2}})\xrightarrow{\sim}\mathrm{D}^{b}_{\mathrm{mix}}(\mathrm{Perv}_{\mathrm{Iw}}(\mathrm{Fl}_{GL_{2}})).
\end{equation*}
The Iwahori-fixed direction carries the noncommutative convolution
datum needed for the Hochschild trace: the affine flag
variety admits an Iwahori stratification indexed by the extended affine
Weyl group \(\widetilde{W}_{GL_{2}}=X^{*}(T)\rtimes W_{f}\cong\mathbb{Z}^{2}\rtimes S_{2}\),
where \(X^{*}(T)=\mathbb{Z}^{2}\) is the character lattice of the
standard torus of \(GL_{2}\).
\end{theorem}

\begin{definition}[global Iwahori Hecke category]
\label{v4-arith:def:global-iwahori-hecke}
\begin{equation}
\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}}\;:=\;\bigotimes_{p}^{\prime}\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p},\quad\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}=\mathrm{D}^{b}_{\mathrm{mix}}(\mathrm{Perv}_{\mathrm{Iw}}(\mathrm{Fl}_{GL_{2},\mathbb{Q}_{p}})).
\end{equation}
\end{definition}

\section{The Emerton--Gee Spectral Side}
\label{v4-arith:bb:emerton-gee}

The automorphic side is the global Iwahori Hecke category. The spectral
side requires a global \((\varphi,\Gamma)\)-module stack; Emerton--Gee
construct the local building block, and the global Selmer-derived stack
assembles them with fixed arithmetic data.

\begin{theorem}[Emerton--Gee arXiv:1908.07185]
\label{v4-arith:thm:emerton-gee}
\ClaimStatusProvedElsewhere. There is a Noetherian formal algebraic
stack \(\mathcal{X}^{\mathrm{EG}}_{GL_{2}}\) over \(\mathrm{Spec}\,\mathbb{Z}_{p}\)
whose points parametrise \((\varphi,\Gamma)\)-modules of rank~\(2\).
The global spectral object used below is not a product or colimit of
these local stacks; it is the Selmer-derived global deformation stack
with fixed residual representation, determinant, and local
conditions.
\end{theorem}

\begin{theorem}[arithmetic Arinkin--Gaitsgory]
\label{v4-arith:thm:arithmetic-AG}
\ClaimStatusConjectured (theorem-waiting locally, conjectural
globally).
Let \(\mathrm{D}^{\mathrm{sph\text{-}fin}}([GL_{2}])\) denote the spherically-generated
full subcategory of the automorphic \(\infty\)-category \(\mathrm{D}([GL_{2}])\) (functions
on the adelic quotient \(GL_2(\mathbb{A})\)), and let
\(\mathcal{X}^{\mathrm{Sel}}_{\bar\rho,\det,\mathcal{L}}\) denote the Selmer-derived global
deformation stack of a fixed residual representation \(\bar\rho\colon \mathrm{Gal}(\overline{\mathbb{Q}}/\mathbb{Q})\to GL_2\),
with fixed determinant and local conditions \(\mathcal{L}\). Then:
\begin{equation}
\mathrm{D}^{\mathrm{sph\text{-}fin}}([GL_{2}])\;\simeq\;\mathrm{IndCoh}_{\mathrm{Nilp}}(\mathcal{X}^{\mathrm{Sel}}_{\bar\rho,\det,\mathcal{L}}).
\end{equation}
Fargues--Scholze arXiv:2102.13459 supplies local geometrization,
spectral action, and Hecke functoriality; Emerton--Gee
supplies the local \((\varphi,\Gamma)\) stack. The displayed
automorphic-to-\(\mathrm{IndCoh}_{\mathrm{Nilp}}\) equivalence is theorem-waiting
locally and conjectural under global adelic descent.
\end{theorem}

\section{The Hochschild Trace Placement}
\label{v4-arith:bb:hochschild-placement}

The automorphic and spectral sides being assembled, the placement of
\(\mathcal{V}^{\mathrm{prim}}\) is the Hochschild homology of the global
Iwahori Hecke category.

\begin{theorem}[crowning placement]
\label{v4-arith:thm:crowning-placement}
\ClaimStatusConjectured.
\begin{equation}
\boxed{\;\mathcal{V}^{\mathrm{prim}}\;=\;\mathrm{HH}_{\bullet}\bigl(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}}\bigr)\;}
\end{equation}
\(E_{1}\)-algebra from Dunn additivity (\(E_{1}\otimes E_{1}=E_{2}\))
applied to the \(E_{2}\)-monoidal structure on each local factor;
Ben-Zvi--Nadler arXiv:0904.1247 identifies
\(\mathrm{HH}_{\bullet}(\mathrm{D}(G/G))=\mathcal{O}(\mathrm{LocSys}_{G}(S^{1}))^{\mathrm{flat}}\).
\end{theorem}

\begin{remark}[adelic inertia circle]
The inertia circle at \(p\) is the profinite completion \(\widehat{\mathbb{Z}}\subset W_{\mathbb{Q}_{p}}\)
of the inertia group; the local system stack \(\mathrm{LocSys}_{GL_{2}}(S^1_p)\)
(flat \(GL_2\)-bundles on the arithmetic circle \(\mathrm{Spec}\,\mathbb{Z}((t))/\mathrm{Spec}\,\mathbb{Z}_p\))
has functions equal to the Hochschild trace. Ben-Zvi--Nadler arXiv:0904.1247 proves the
complex-group identification \(\mathrm{HH}_\bullet(\mathrm{D}(G/G))\simeq \mathcal{O}(\mathrm{LocSys}_G(S^1))^{\mathrm{flat}}\);
the present arithmetic statement is its \((\varphi,\Gamma)\)-module counterpart.
\end{remark}

\section{Connection to Gaiotto's 3D Chern--Simons Presentation}
\label{v4-arith:bb:connection-gaiotto}

\begin{proposition}[three-way comparison]
\label{v4-arith:prop:three-way-equivalence}
\ClaimStatusConjectured. The three presentations of
\(\mathcal{V}^{\mathrm{prim}}\) are conjecturally equivalent:
\begin{enumerate}
\item (Kim arXiv:1510.05818 / Gaiotto) arithmetic Chern--Simons at the classical
level produces a boundary contribution whose partition function specializes to
\(\mathcal{V}^{\mathrm{prim}}\) on the arithmetic Langlands spectral side;
\item (Bezrukavnikov--Bernstein) \(\mathcal{V}^{\mathrm{prim}}=\mathrm{HH}_{\bullet}(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}})\);
\item (Costello--Gwilliam) \(\mathcal{V}^{\mathrm{prim}}\) is a factorization algebra
\(\mathcal{V}^{\mathrm{prim}}\in\mathrm{FactAlg}(\mathrm{Mod}_{\widehat{\mathbb{Z}},\psi})\)
over the adelic curve, with the Whittaker and factorization conditions of Chapter~20.
\end{enumerate}
These are conditional comparison identifications, not proved equivalences.
Ben-Zvi--Nadler's Chern--Simons/trace/Hecke triangle (arXiv:0904.1247) is the
complex template; the arithmetic gluing requires global adelic descent (Conjecture~F3
of Chapter~21).
\end{proposition}

\section{Hecke Trace = Eichler--Selberg}
\label{v4-arith:bb:hecke-trace}

The Hochschild trace of \(\mathcal{V}^{\mathrm{prim}}\) specializes at
each weight-level pair \((k,N)\) to the trace of a Hecke operator on
the space of cusp forms; this specialization is the Eichler--Selberg
trace formula.

\begin{theorem}[Eichler--Selberg trace formula]
\label{v4-arith:thm:healed-hecke-trace}
\ClaimStatusProvedElsewhere. For each \(N\geq 1\), \(k\geq 2\):
\begin{equation}
\mathrm{tr}\bigl(T_{p}\mid S_{k}(\Gamma_{1}(N))\bigr)\;=\;-\sum_{t^{2}<4p}P_{k}(t,p)H_{w}(t^{2}-4p)
+ \delta_{\mathrm{id}}(p) + \delta_{\mathrm{Eis}}(p,k,N),
\end{equation}
where \(P_{k}(t,p)=(\alpha^{k-1}-\beta^{k-1})/(\alpha-\beta)\),
\(\alpha\beta=p\), \(\alpha+\beta=t\), \(H_{w}\) denotes the Hurwitz
weighted class number of binary quadratic forms of discriminant \(t^2-4p\);
\(\delta_{\mathrm{id}}(p)\) is the identity contribution \((p^{k-1}+1)/12\cdot\mu(N)\)
(for \(p\nmid N\)); and \(\delta_{\mathrm{Eis}}(p,k,N)\) collects the
% RECTIFICATION-FLAG: Gate 1. "boundary terms" named here as identity + Eisenstein
% contributions. The hyperbolic term (t²=4p, parabolic elements) and the full
% Eichler-Selberg decomposition at t²>4p should be verified against Knightly-Li
% "Traces of Hecke Operators" AMS 2006, Ch. 5, and Zagier's trace formula notes.
% The displayed formula covers the elliptic (t²<4p) main term correctly.
Eisenstein boundary contribution.
\end{theorem}

\begin{theorem}[universal trace-class tabulation]
\label{v4-arith:thm:trace-class-tabulation}
\ClaimStatusConjectured. \(\mathcal{V}^{\mathrm{prim}}\) represents
the universal family of Hecke eigensystems on the global automorphic
category \(\mathrm{D}([GL_{2}])^{\mathrm{sph\text{-}fin}}\): it is
the object over the weight-level base \(\mathcal{W}\times\mathrm{Spec}\,\mathbb{Z}[1/N]\)
(where \(\mathcal{W}\) is the weight space parametrizing integers \(k\geq 2\))
whose closed-point fibres are individual Hecke eigensystems.
Restricted to \(S_{k}(\Gamma_{1}(N))\) for fixed \((k,N)\), the fibre of the
universal family recovers the Eichler--Selberg trace formula
coefficient by coefficient; individual Hecke eigenforms are
the closed points of this family.
\end{theorem}

\section{The Arithmetic Langlands Diamond for \(\mathcal{V}^{\mathrm{prim}}\)}
\label{v4-arith:bb:summary}

\begin{enumerate}
\item \(\mathrm{Bun}_{GL_{2}}(\mathrm{Spec}\,\mathbb{Z})=\{\mathbb{Z}^{2}\}\)
is a single isomorphism class (\Cref{v4-arith:cor:Bun-GL2-is-point}); the
automorphic category over this groupoid is semisimple of rank \(1\), forcing
the passage to the Iwahori-equivariant setting.
\item The spherical Hecke algebra is commutative (Satake isomorphism); the Iwahori-equivariant
affine Hecke category \(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},p}\) carries the noncommutative
convolution datum (\Cref{v4-arith:thm:bezrukavnikov}).
\item The spectral side is the Selmer-derived global stack, built
from local Emerton--Gee/\((\varphi,\Gamma)\) stacks with fixed
residual representation, determinant, and local conditions.
\item Arithmetic Arinkin--Gaitsgory: Fargues--Scholze and Emerton--Gee supply local inputs;
the automorphic-to-\(\mathrm{IndCoh}_{\mathrm{Nilp}}\) equivalence is
theorem-waiting locally and conjectural globally
(\Cref{v4-arith:thm:arithmetic-AG}).
\item Crowning placement:
\(\mathcal{V}^{\mathrm{prim}}=\mathrm{HH}_{\bullet}(\mathrm{Hk}^{\mathrm{Iw}}_{GL_{2},\mathrm{glob}})\)
(\Cref{v4-arith:thm:crowning-placement}).
\item Three-way comparison with Kim--Gaiotto arithmetic Chern--Simons and Costello--Gwilliam
factorization (\Cref{v4-arith:prop:three-way-equivalence}).
\item Hecke-trace tabulation recovers the Eichler--Selberg formula
(\Cref{v4-arith:thm:trace-class-tabulation}).
\end{enumerate}

\(\mathcal{V}^{\mathrm{prim}}\) is an arithmetic Hochschild
character. It encodes on the automorphic side the universal trace
of the identity on the global Iwahori Hecke category of \(GL_{2}\);
on the spectral side, through the conjectural arithmetic Langlands
equivalence, the structure sheaf \(\mathcal{O}\) of the global
Galois-deformation stack. The rank-\(2\) Heisenberg presentation
per prime is the Whittaker model of this trace, with the two
generators diagonalising the Satake parameters.
