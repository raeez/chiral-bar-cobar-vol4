\chapter{Arithmetic Chern--Simons Holography}
\label{v4-arith:gaiotto}

Class-\(\mathcal S\), Higgs-branch vertex algebras, and
Costello--Gaiotto boundary categories require a complex curve or a
quantized boundary theory. Kim arithmetic Chern--Simons supplies only
the classical arithmetic action line. Finite state sums and the
conditional boundary category appear in
Chapter~\ref{v4-arith:3d-acs-E1bar}; the profinite comparison with
\(\mathcal V^{\mathrm{prim}}\) remains open.

\section{Class \(\mathcal{S}\) Fails Arithmetically}
\label{v4-arith:ga:class-S-fails}

\begin{theorem}[Gaiotto, arXiv:0904.2715]
\label{v4-arith:thm:gaiotto-class-S}
\ClaimStatusProvedElsewhere. Class-\(\mathcal{S}\) builds
4D \(\mathcal{N}=2\) theories \(T[\Sigma,\mathfrak{g}]\) by
compactifying the 6D \((2,0)\) theory \(\mathfrak{X}[\mathfrak{g}]\) on
a punctured Riemann surface \(\Sigma\).
\end{theorem}

\begin{observation}[three obstructions]
\label{v4-arith:obs:class-S-obstructions}
Arithmetic class-\(\mathcal{S}\) on \(\overline{\mathrm{Spec}\,\mathbb{Z}}\times\mathbb{R}^{4}\)
fails:
\begin{enumerate}
\item 6D \((2,0)\) requires a smooth oriented \(6\)-manifold;
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\times\mathbb{R}^{4}\) is
topologically \(5\)-dimensional (arithmetic \(1\)-dim plus \(4\)-dim
real). Connes' spectral dimension of \(\overline{\mathrm{Spec}\,\mathbb{Z}}\)
is \(1\), not \(2\) (Connes--Consani arXiv:1405.4527).
\item The self-dual \(2\)-form \(B\) on an M5-brane has no Arakelov
analogue (no star on \(2\)-forms on a \(1\)-d arithmetic curve, where
\(\Omega^{2}=0\)).
\item Hitchin-moduli Higgs branches on
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\) are not defined by the
class-\(\mathcal S\) formula: the Arakelov dualizing sheaf is a
metrised arithmetic line, not the holomorphic canonical bundle of a
Riemann surface.
\end{enumerate}
\end{observation}

\begin{corollary}
\label{v4-arith:cor:class-S-falsified}
\ClaimStatusProvedHere. Class-\(\mathcal{S}\) as a generating
principle for arithmetic vertex algebras is obstructed.
\end{corollary}

\section{The Hitchin Base Has No Class-\(\mathcal S\) Arithmetic Specialization}
\label{v4-arith:ga:CH-vanish}

\begin{theorem}[arithmetic Hitchin obstruction]
\label{v4-arith:thm:arith-branch-vanishing}
\ClaimStatusProvedHere. The class-\(\mathcal S\) Coulomb branch
\[
\mathcal C_{\Sigma,\mathfrak g}=\bigoplus_i
H^0(\Sigma,K_\Sigma^{d_i})
\]
uses the holomorphic canonical bundle of a complex curve. For
\(\overline{\mathrm{Spec}\,\mathbb Z}\) the dualizing object is the
Arakelov line with metric at infinity. Its sections belong to
Arakelov intersection theory, not to a Hitchin base. Hence the
class-\(\mathcal S\) Coulomb and Higgs branches do not vanish; rather,
the formula defining them has no canonical arithmetic value.
\end{theorem}

\begin{remark}[why the replacement is three-dimensional]
This obstruction is not a proof that an arithmetic \(4\)-dimensional
theory is topological. It says that the class-\(\mathcal S\) descent
from a \(6\)-dimensional theory along a complex curve has no
arithmetic curve to descend along. Kim's arithmetic Chern--Simons
theory is therefore not the topological remnant of that construction;
it is a separate \(3\)-dimensional arithmetic field theory.
\end{remark}

\section{Kim's Arithmetic Chern--Simons}
\label{v4-arith:ga:kim-ACS}

\begin{definition}[Kim 2015, arXiv:1510.05818]
\label{v4-arith:def:kim-ACS}
For a number field \(F\) and finite Galois module \(M\), arithmetic
Chern--Simons theory assigns to a Galois representation
\(\rho\colon\pi_{1}^{\mathrm{\acute{e}t}}(\overline{\mathrm{Spec}\,\mathcal{O}_{F}})\to G\)
the action
\begin{equation}
\mathrm{CS}_{F}(\rho)\;=\;\langle\rho^{*}c,[\overline{\mathrm{Spec}\,\mathcal{O}_{F}}]\rangle\;\in\;H^{3}(\overline{\mathrm{Spec}\,\mathcal{O}_{F}},\mu_{N})
\end{equation}
built from a class \(c\in H^{3}(BG,\mu_{N})\). This is a genuine 3D
arithmetic Chern--Simons action, not yet an Atiyah--Segal quantum
TQFT. The topology is the Mazur--Morishita dictionary: primes are
knots and \(\overline{\mathrm{Spec}\,\mathbb{Z}}\) is the arithmetic
3-manifold (Morishita arXiv:0904.3399).
\end{definition}

\begin{remark}
Chung--Kim--Kim--Pappas--Park--Yoo arXiv:1706.03336 extends Kim's
construction to higher-dimensional arithmetic manifolds.
Arithmetic Chern--Simons supplies the rigorous classical action
functional at the arithmetic vertex that Witten's analysis
(\Cref{v4-arith:thm:witten-conclusion}) identifies as the ingredient
Deninger's programme requires.
\end{remark}

\section{\(\mathcal{V}^{\mathrm{prim}}\) and the ACS Boundary Shadow}
\label{v4-arith:ga:V-prim-as-boundary}

\begin{theorem}[finite arithmetic CS boundary shadow]
\label{v4-arith:thm:arith-CS-VA-correspondence}
\ClaimStatusConjectured. Let \(G\) be a finite gauge group and
\(c\in H^{3}(BG,\mu_{N})\) a Kim arithmetic Chern--Simons level. Kim's
theory supplies a classical arithmetic CS torsor/action line for this
finite datum. The expression
\(\mathrm{ACS}[\overline{\mathrm{Spec}\,\mathbb{Z}},\mathbb{G}_{m},1]\)
is not an instance of Kim's finite-gauge construction; it is a
separate abelian extension problem. A quantum boundary category
\(\partial_{q}\mathrm{ACS}\), once constructed, has conjectural
archimedean Heisenberg factorization shadow
\begin{equation}
\chi\bigl(\partial_{q}\mathrm{ACS}\bigr)(s)\;=\;\prod_{p}(1-p^{-s})^{-1},
\end{equation}
and is conjecturally compared with the Hochschild-trace candidate
\(\mathcal{V}^{\mathrm{prim}}\) (a Hochschild trace class;
\Cref{v4-arith:thm:heisenberg-bar-zeta}). The quantum boundary category
at finite places is conditional, as in Chapter~\ref{v4-arith:3d-acs-E1bar}.
\end{theorem}

\begin{proof}[Form]
Kim's construction is classical: it assigns Chern--Simons actions
and torsors to Galois representations. Witten's CS/WZW boundary
theorem transports to \(\overline{\mathrm{Spec}\,\mathbb{Z}}\) only
after replacing the complex boundary curve and its OPE by the
arithmetic integral transform of Chapter~\ref{v4-arith:3d-acs-E1bar}.
That chapter defines the finite-level boundary category by
transgression, conditional on the finite arithmetic cobordism input;
the displayed Euler product is then a character-level expectation for
the renormalised profinite boundary comparison with the
Hochschild-trace class \(\mathcal{V}^{\mathrm{prim}}\).
\end{proof}

\begin{corollary}[character = \(\zeta\)]
\label{v4-arith:cor:character-is-zeta}
\ClaimStatusConjectured.
\begin{equation}
\chi(\partial_{q}\mathrm{ACS})(s)\;=\;\prod_{p}(1-p^{-s})^{-1}\;=\;\zeta(s),
\end{equation}
conditional on constructing the quantum boundary category and
identifying its Hochschild trace with the Hochschild-trace candidate.
\end{corollary}

\section{Wilson Lines Are Labeled by Ramified Hecke Characters}
\label{v4-arith:ga:wilson-lines}

\begin{theorem}[Wilson-line labels as local Hecke data]
\label{v4-arith:thm:wilson-hecke}
\ClaimStatusConjectured. Wilson lines in \(\mathrm{ACS}\) wrapped
around the ``knot'' corresponding to a prime \(p\) (Mazur--Morishita
arithmetic topology) are \(1\)-dimensional defects in the arithmetic
\(3\)-fold, each \emph{labeled by} a local Hecke character \(\chi_{p}\)
of \(\mathbb{Q}_{p}^{\times}\) (ramified or unramified). The label
is a closed point of the universal family of local characters,
and conjecturally a module over the quantum boundary category
specialized at \(p\). Inserting a Wilson line with label \(\chi_{p}\)
produces a boundary defect whose local datum at \(p\) is the
ramification datum of \(\chi_{p}\) (conjectural; the categorical
boundary module is part of the conditional boundary-category input of
Chapter~\ref{v4-arith:3d-acs-E1bar}).
Wilson-line labels are \(0\)-dimensional data; the Wilson line itself
is \(1\)-dimensional.
\end{theorem}

\begin{remark}[defect/puncture dictionary]
Class-\(\mathcal{S}\) punctures correspond to 6D codimension-\(2\)
defects in the complex setting; arithmetic ramification corresponds
to Wilson lines in arithmetic Chern--Simons. The structural
dictionary ``punctures \(=\) defects'' becomes
``Wilson lines \(=\) ramification'' in the arithmetic setting,
with the 6D/4D framing replaced by the 3D arithmetic TQFT framing.
\end{remark}

\section{Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees as Adelic Character}
\label{v4-arith:ga:BLLPRvR-adelic}

\begin{theorem}[arithmetic Schur index]
\label{v4-arith:thm:arithmetic-schur-index}
\ClaimStatusConjectured. Let \(\mathcal{H}^{\mathrm{arith}}\) denote
the conjectural arithmetic Hilbert space on which Frobenius operators
\(\mathrm{Frob}_{p}\) act (one per prime \(p\)), and let the Schur
grading at \(p\) be \(t_{p}^{\mathrm{Frob}_{p}}\) (the formal
\(\mathrm{Frob}_{p}\)-graded trace variable). The arithmetic Schur
index is the formal adelic graded trace
\begin{equation}
\mathcal{I}^{\mathrm{arith}}(\{t_{p}\})\;=\;\mathrm{Tr}_{\mathcal{H}^{\mathrm{arith}}}\prod_{p}t_{p}^{\mathrm{Frob}_{p}}.
\end{equation}
Under the Hecke substitution \(t_{p}=p^{-s}\),
\(\mathcal{I}^{\mathrm{arith}}(s)=\prod_{p}(1-p^{-s})^{-1}=\zeta(s)\).
This is an Euler-character computation, not a construction of the
Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees Schur-index
category.
\end{theorem}

\begin{remark}
The Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees correspondence
takes holographic form only after the finite-gauge ACS datum, boundary
category, and trace functor have been constructed. The arithmetic
vertex algebra generating problem, for which class-\(\mathcal{S}\) is obstructed
(\Cref{v4-arith:cor:class-S-falsified}), admits arithmetic CS
holography as a conditional comparison, not as a theorem of the
present finite-gauge formalism.
\end{remark}

\section{Arithmetic CS as the Generating Principle}
\label{v4-arith:ga:summary}

\begin{enumerate}
\item Class-\(\mathcal{S}\) obstructed: three 4D-gauge obstructions
(\Cref{v4-arith:cor:class-S-falsified}).
\item The class-\(\mathcal S\) Hitchin base has no canonical
arithmetic specialization on \(\overline{\mathrm{Spec}\,\mathbb Z}\)
(\Cref{v4-arith:thm:arith-branch-vanishing}).
\item Kim's arithmetic Chern--Simons (arXiv:1510.05818) is the
classical 3D action functional
(\Cref{v4-arith:def:kim-ACS}).
\item Finite state sums are constructed in
Chapter~\ref{v4-arith:3d-acs-E1bar}; the renormalised profinite
boundary category is conjectural, as is its comparison with
\(\mathcal{V}^{\mathrm{prim}}\) and its character \(\zeta(s)\)
(\Cref{v4-arith:thm:arith-CS-VA-correspondence}).
\item Wilson lines are \(1\)-dimensional defects labeled by ramified
Hecke characters (= expected modules of the quantum boundary
category specialized at \(p\), \(0\)-dimensional data)
(\Cref{v4-arith:thm:wilson-hecke}).
\item BLLPRvR adelic character: \(\zeta(s)\) is the conjectural Schur
index character of arithmetic CS
(\Cref{v4-arith:thm:arithmetic-schur-index}).
\end{enumerate}

Kim arithmetic \(\mathbb{G}_{m}\)-Chern--Simons on
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\) at level~\(1\) supplies the
classical arithmetic action. Chapter~\ref{v4-arith:3d-acs-E1bar}
constructs finite state sums and states the boundary category as a
conditional profinite problem; the renormalised comparison with
\(\mathcal{V}^{\mathrm{prim}}\) remains open.
