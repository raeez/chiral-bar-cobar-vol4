\chapter{The Automorphic Vertex of the Langlands--TQFT Triangle}
\label{v4-arith:witten}

\emph{In the voice of Witten. Kapustin--Witten (hep-th/0604151)
realised geometric Langlands as S-duality in \(\mathcal{N}=4\)
SYM on \(\Sigma\times C\). This voice tests whether the
Hartnoll--Yang BKL--primon correspondence supports a genuine
arithmetic TQFT bulk.}

\(\mathcal{V}^{\mathrm{prim}}\) is a Hochschild trace class whose
graded character is \(\zeta(s)\). Three TQFT frameworks (Kapustin--Witten,
Beem--Lemos--Liendo--Peelaers--Rastelli--van Rees (BLLPRvR),
and Chern--Simons/WZW) offer candidate TQFT-bulk structures for the
Langlands triangle. Each fails at a specific structural obstruction,
placing \(\mathcal{V}^{\mathrm{prim}}\) on the automorphic vertex
rather than the TQFT vertex.

\section{The Kapustin--Witten Obstruction: Absence of Complex Structure}
\label{v4-arith:wit:KW-does-not-lift}

\begin{theorem}[Kapustin--Witten setup]
\label{v4-arith:thm:KW-setup}
\ClaimStatusProvedElsewhere. Geometric Langlands for \(C/\mathbb{C}\)
arises as S-duality of 4D \(\mathcal{N}=4\) super Yang--Mills (SYM) on \(\Sigma\times C\)
with \(\Sigma\) topologically twisted and \(C\) holomorphically
twisted. Categories \(D(\mathrm{Bun}_{G})\) and
\(\mathrm{QCoh}(\mathrm{Loc}_{^{L}G})\) appear as boundary
conditions; S-duality \(\tau\mapsto-1/\tau\) exchanges them.
\end{theorem}

\begin{observation}[arithmetic obstruction]
\label{v4-arith:obs:KW-arithmetic-obstruction}
\(\overline{C}=\overline{\mathrm{Spec}\,\mathbb{Z}}\) is an arithmetic
\(1\)-scheme with closed-point residue fields \(\mathbb{F}_{p}\) at each
prime \(p\), plus the archimedean place. The holomorphic twist of
\(\mathcal{N}=4\) super Yang--Mills requires a complex structure on the
compactified direction; \(\overline{\mathrm{Spec}\,\mathbb{Z}}\) carries
no such structure, its closed fibres being spectra of finite fields
rather than complex-analytic curves.
\end{observation}

\begin{corollary}[Kapustin--Witten is undefined over \(\mathbb{Z}\)]
\label{v4-arith:cor:no-arith-4D-SYM}
\ClaimStatusProvedHere. The expression ``\(\mathcal{N}=4\) SYM on
\(\Sigma\times\overline{\mathrm{Spec}\,\mathbb{Z}}\)'' names no
defined theory in 2026's mathematics. The Kapustin--Witten holomorphic
twist requires a complex structure on the compactified direction; the
topological \(\Sigma\)-coupling and the S-duality
\(\tau\mapsto-1/\tau\) are consequences. The arithmetic compactification
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\) admits no such structure.
\end{corollary}

\section{The BLLPRvR Obstruction: Infinite Flavour Rank}
\label{v4-arith:wit:BLLPRvR-fails}

\begin{theorem}[BLLPRvR, arXiv:1312.5344]
\label{v4-arith:thm:BLLPRvR}
\ClaimStatusProvedElsewhere. The Beem--Lemos--Liendo--Peelaers--%
Rastelli--van Rees (BLLPRvR) correspondence assigns a vertex algebra (VA)
\(\chi(\mathcal{T})\) to any 4D \(\mathcal{N}=2\) superconformal field
theory (SCFT) \(\mathcal{T}\), with central charges satisfying
\(c_{2d}=-12c_{4d}\) and \(k_{2d}=-\tfrac{1}{2}k_{4d}\).
\end{theorem}

\begin{observation}
\label{v4-arith:obs:BLLPRvR-rank-mismatch}
The primon construction \(\mathcal{V}^{\mathrm{prim}}\) is generated
by a countable family of Heisenberg generators \(\{a_{p,n}\}_{p\in\mathbb{P},n\in\mathbb{Z}}\)
indexed by primes; its central charge, in the Polyakov-regularised
sense of \Cref{v4-arith:polyakov} (zeta-regularisation of the formal
prime sum), is \(c = 2\cdot\mathrm{Res}_{s=1}\zeta(s) = 2\), and its
flavour symmetry group is \(\prod_{p\in\mathbb{P}} U(1)_{p}\) with
flavour rank \(|\mathbb{P}| = \aleph_{0}\).
No 4D \(\mathcal{N}=2\) SCFT has flavour symmetry of rank \(\aleph_{0}\):
the flavour rank is bounded by the gauge rank plus hypermultiplet content,
both finite for any theory with a Lagrangian description.
\end{observation}

\begin{corollary}
\label{v4-arith:cor:BLLPRvR-fails-arith}
\ClaimStatusProvedHere. The VA \(\chi(\mathcal{T})\) assigned by
BLLPRvR to any 4D \(\mathcal{N}=2\) SCFT \(\mathcal{T}\) has finite
flavour rank; \(\mathcal{V}^{\mathrm{prim}}\) has flavour rank
\(\aleph_{0}\). Hence \(\mathcal{V}^{\mathrm{prim}}\) is not the
BLLPRvR vertex algebra of any 4D \(\mathcal{N}=2\) SCFT.
The class-\(\mathcal{S}\) construction on an infinite-genus or
infinite-puncture arithmetic curve would require
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\) to carry the complex structure
of a Riemann surface, blocked by
\Cref{v4-arith:obs:KW-arithmetic-obstruction}.
\end{corollary}

\section{The Chern--Simons Obstruction: Irrational Levels}
\label{v4-arith:wit:3D-CS-fails}

\begin{observation}[non-integer level]
\label{v4-arith:obs:non-integer-level}
Witten's Chern--Simons/WZW correspondence gives \(CS_{k}(G)\) on
\(\Sigma\times\mathbb{R}\) with boundary vertex operator algebra (VOA)
\(V_{k}(\mathfrak{g})\) at integer level \(k\in\mathbb{Z}\). A
\(\mathcal{V}^{\mathrm{prim}}\)-boundary Chern--Simons theory requires
gauge group \(\prod_{p\in\mathbb{P}}U(1)_{p}\) at levels
\(k_{p}=1/\log p\). These levels are irrational; the Chern--Simons
path integral on non-trivial 3-manifolds requires \(k\in\mathbb{Z}\)
for well-definedness.
\end{observation}

\begin{corollary}
\label{v4-arith:cor:no-CS-VOA-arith}
\ClaimStatusProvedHere. No finite-level abelian CS delivers
\(\mathcal{V}^{\mathrm{prim}}\) as boundary.
\end{corollary}

\section{The Belinski--Khalatnikov--Lifshitz Billiard: Thermodynamic Coarse-Graining}
\label{v4-arith:wit:BKL-coarse-graining}

\begin{observation}[Selberg scattering versus primon partition]
\label{v4-arith:obs:selberg-vs-primon}
The Selberg trace formula on \(\mathrm{SL}(2,\mathbb{Z})\backslash\mathbb{H}^{2}\)
expresses the spectral density of the Laplacian as a sum over Maass
cusp forms plus a continuous Eisenstein series contribution. The
Eisenstein scattering phase is \(\zeta(2it)/\zeta(2it+1)\) on the
critical line, a scattering-matrix entry; the primon partition function
is \(\prod_{p}(1-p^{-s})^{-1}=\zeta(s)\), with pole at \(s=1\). The
two occurrences of \(\zeta\) are structurally distinct: scattering
amplitude versus thermodynamic partition function.
\end{observation}

\begin{theorem}[Hartnoll--Yang: thermodynamic coarse-graining]
\label{v4-arith:thm:HY-coarse-graining}
\ClaimStatusProvedElsewhere (Hartnoll--Yang arXiv:2502.02661,
\S3). The primon--Belinski--Khalatnikov--Lifshitz (BKL) correspondence
is a thermodynamic coarse-graining: at high temperature, the Selberg
continuous spectrum above eigenvalue \(1/4\) is replaced by the
primon oscillator count. The spectral match is asymptotic, not exact.
\end{theorem}

\section{The Automorphic Vertex}
\label{v4-arith:wit:automorphic-vertex}

\begin{theorem}[\(\mathcal{V}^{\mathrm{prim}}\) on the automorphic vertex]
\label{v4-arith:thm:automorphic-vertex}
\ClaimStatusProvedHere. \(\mathcal{V}^{\mathrm{prim}}\) occupies the
automorphic vertex of the Langlands--TQFT triangle:
\begin{equation*}
\underbrace{\bigl\{\text{automorphic reps of }GL_{1}(\mathbb{A}_{\mathbb{Q}})\bigr\}}_{\mathcal{V}^{\mathrm{prim}}}\;\longleftrightarrow\;\underbrace{\bigl\{\text{1-dimensional Galois representations}\bigr\}}_{\text{Artin reciprocity}}\;\longleftrightarrow\;\underbrace{\bigl\{\text{arithmetic TQFT bulk}\bigr\}}_{\text{Deninger (conjectural)}}.
\end{equation*}
The graded character of \(\mathcal{V}^{\mathrm{prim}}\) is
\(\zeta(s)\); its Heisenberg decomposition is the Euler product
\(\prod_{p}(1-p^{-s})^{-1}\). Artin reciprocity identifies the first
two vertices; the third is Deninger's conjectural \(\mathbb{F}_{1}\)-geometric programme.
\end{theorem}

\section{Asymptotic Symmetries at the BKL End-of-Time Boundary}
\label{v4-arith:wit:asymptotic-symmetry}

\begin{proposition}[Damour--Henneaux--Nicolai hep-th/0207267]
\label{v4-arith:prop:DHN-billiard}
\ClaimStatusProvedElsewhere. BKL dynamics for 4D pure gravity on
\(T^{3}\) reduces to null billiards on the Weyl chamber of \(AE_{3}\),
a rank-\(3\) hyperbolic Kac--Moody algebra; the fundamental domain
is a modification of \(\mathrm{SL}_{2}(\mathbb{Z})\backslash\mathbb{H}^{2}\).
\end{proposition}

\begin{observation}[rank obstruction for \(AE_{3}\)]
\label{v4-arith:obs:asymptotic-symmetry-placement}
A soft-mode algebra at the BKL end-of-time boundary decomposes into
Weyl-group representations of \(W(AE_{3})\), where \(AE_{3}\) has
root lattice of rank \(3\). The primon Heisenberg algebra is generated
by \(\{a_{p,n}\}_{p\in\mathbb{P},n\in\mathbb{Z}}\), with generator
index set of cardinality \(\aleph_{0}\). The two index sets are
incommensurable; \(\mathcal{V}^{\mathrm{prim}}\) is the bosonic Fock
quantisation of the Eisenstein continuous sector of the gravitational
superspace, not the full \(W(AE_{3})\) soft-mode algebra. The
Hartnoll--Yang thermodynamic match
(\Cref{v4-arith:thm:HY-coarse-graining}) is a coarse-graining of
the Eisenstein density of states, not a spectral identification.
\end{observation}

\section{The Vol III \(\Phi_{d}\) Functor: Arithmetic Scope Boundary}
\label{v4-arith:wit:vol3-obstruction}

\begin{observation}
\label{v4-arith:obs:vol3-obstruction}
Vol III's functor
\(\Phi_{d}=\mathrm{Sp}^{\mathrm{ch}}_{\Sigma_{d-1},C}\circ\Phi^{\mathrm{FA}}_{d}\)
takes a Calabi--Yau \(d\)-fold over \(\mathbb{C}\) as input.
\(\overline{\mathrm{Spec}\,\mathbb{Z}}\) is an arithmetic \(1\)-scheme
over \(\mathbb{Z}\); it admits no Calabi--Yau structure. The functor
\(\Phi_{d}\) is therefore defined on the \(\mathbb{C}\)-analytic locus
of the programme, with \(\overline{\mathrm{Spec}\,\mathbb{Z}}\) outside
its domain.
\end{observation}

\begin{remark}
When a Calabi--Yau \(3\)-fold carries an \(\mathrm{SL}_{2}(\mathbb{Z})\)
action on its elliptic-fibration moduli (for example, K3\(\times E\)),
the modular-form output of \(\Phi_{3}\) carries arithmetic data.
The Borcherds product \(\Phi_{N}\) (\(N\in\{1,2,3,4,6\}\)) lives on
\(\mathrm{Sp}_{4}(\mathbb{Z})\)-moduli space; it is an automorphic
object, occupying the automorphic vertex together with
\(\mathcal{V}^{\mathrm{prim}}\).
\end{remark}

\section{Automorphic Placement}
\label{v4-arith:wit:verdict-automorphic}

\begin{theorem}[automorphic placement of \(\mathcal{V}^{\mathrm{prim}}\)]
\label{v4-arith:thm:witten-verdict}
\ClaimStatusProvedHere. Each candidate TQFT-bulk framework for
\(\mathcal{V}^{\mathrm{prim}}\) (Kapustin--Witten, BLLPRvR,
holomorphic Chern--Simons, the Costello--Gaiotto holomorphically-twisted
theory, and Vol~III \(\Phi_{d}\)) is blocked by a specific structural
obstruction catalogued in
\S\S\ref{v4-arith:wit:KW-does-not-lift}--\ref{v4-arith:wit:vol3-obstruction}. \(\mathcal{V}^{\mathrm{prim}}\) occupies
the automorphic vertex of the Langlands--TQFT triangle
(\Cref{v4-arith:thm:automorphic-vertex}).
\end{theorem}

\begin{remark}[the Deninger programme]
A rigorous construction of \(\overline{\mathrm{Spec}\,\mathbb{Z}}\)
as a ``curve over \(\mathbb{F}_{1}\)'' with a Frobenius-type
\(\mathbb{R}_{>0}\)-action, Deninger's \(\mathbb{F}_{1}\)-geometric
programme, would supply the missing TQFT-bulk vertex. The arithmetic
site of Connes--Consani (arXiv:1405.4527) provides one candidate
foundation.
\end{remark}

\section{The Automorphic Vertex: Structural Consolidation}
\label{v4-arith:wit:summary}

Each obstruction in \S\S\ref{v4-arith:wit:KW-does-not-lift}--\ref{v4-arith:wit:vol3-obstruction}
is structural: the complex-structure requirement of
Kapustin--Witten (\Cref{v4-arith:cor:no-arith-4D-SYM}), the finite
flavour-rank bound of BLLPRvR (\Cref{v4-arith:cor:BLLPRvR-fails-arith}),
the integrality requirement of Chern--Simons levels
(\Cref{v4-arith:cor:no-CS-VOA-arith}), the thermodynamic rather than
spectral character of the Hartnoll--Yang match
(\Cref{v4-arith:thm:HY-coarse-graining}), and the \(\mathbb{C}\)-analytic
domain of \(\Phi_{d}\) (\Cref{v4-arith:obs:vol3-obstruction}).
The automorphic vertex placement (\Cref{v4-arith:thm:automorphic-vertex})
is the consequence: \(\mathcal{V}^{\mathrm{prim}}\) is identified by
Artin reciprocity with the automorphic side of the Langlands triangle,
with the TQFT-bulk vertex remaining Deninger's open programme.
