\section{Rescaling and the Conformal Vector Constraint}
\label{v4-arith:pol:rescaling-not-conformal}

\begin{observation}[rescaling breaks Virasoro]
\label{v4-arith:obs:rescaling-breaks-virasoro}
For a rescaled stress tensor \(T'=\lambda T\):
\begin{equation*}
T'(z)T'(w)\;=\;\lambda^{2}\Bigl(\frac{c/2}{(z-w)^{4}}+\frac{2T(w)}{(z-w)^{2}}+\frac{\partial T(w)}{z-w}\Bigr)+\cdots
\end{equation*}
For this to be a conformal OPE
\(\tfrac{c'/2}{(z-w)^{4}}+\tfrac{2T'(w)}{(z-w)^{2}}+\cdots\) we need
\(\lambda^{2}c=c'\) but \(\lambda^{2}\cdot 2T=2T'=2\lambda T\), i.e.
\(\lambda^{2}=\lambda\), forcing \(\lambda=1\). Rescaling by
\(\lambda\neq 1\) does not yield a conformal vector.
\end{observation}

\begin{remark}
The conformal vector \(\omega_{p}=\tfrac{1}{2}(\log p)(a^{(p)}_{-1}a^{(p)}_{-1}+b^{(p)}_{-1}b^{(p)}_{-1})|0\rangle\)
(\Cref{v4-arith:def:primon-va-naive}) is the Sugawara vector for
rank-\(2\) Heisenberg with the \(\log p\)-rescaled inner product.
In the rescaled coordinate, \(\omega_{p}\) is the canonical conformal vector of
the rescaled algebra; \(c^{(p)}=2\) because there are two generators.
The factor \(\log p\) enters the eigenvalues of \(L_{0}^{(p)}\) but not the
central term of the OPE.
\end{remark}

\begin{theorem}[per-prime central charge]
\label{v4-arith:thm:per-prime-c}
\ClaimStatusProvedHere. \(c^{(p)}=2\) for every prime \(p\). The
\(\log p\) rescaling enters conformal weights but not the central charge.
\end{theorem}

\begin{proof}
\Cref{v4-arith:obs:rescaling-breaks-virasoro} shows that the
\(\log p\)-rescaling preserves the Virasoro OPE precisely at the
unrescaled central term, and the Sugawara construction on the
rank-\(2\) Heisenberg \(\mathcal{H}^{(p)}\)
(\Cref{v4-arith:def:primon-va-naive}) gives
\(c^{(p)}=\mathrm{rank}=2\).
\end{proof}

\section{The Infinite Tensor Product and the Virasoro Obstruction}
\label{v4-arith:pol:infinite-tensor-not-VOA}

\begin{observation}[infinite tensor fails Virasoro]
\label{v4-arith:obs:infinite-tensor-fails}
Stress tensor \(\omega=\sum_{p}\omega_{p}\) is an infinite sum of
non-zero vectors in the Fock module; not a vector in the module.
On the restricted tensor product, the Virasoro commutator
\([L_{m},L_{-m}]=2m L_{0}+\tfrac{c}{12}m(m^{2}-1)\) has
\(c=\sum_{p}c^{(p)}=\sum_{p}2=+\infty\). The central term is an
infinite scalar multiplying every state.
\end{observation}

\begin{corollary}
\label{v4-arith:cor:no-virasoro}
\ClaimStatusProvedHere. The Witt algebra (classical conformal,
no central extension) acts algebraically on
\(\mathcal{V}^{\mathrm{prim}}\) via \(L_m = \sum_p L_m^{(p)}\).
The Virasoro central extension is absent: the central term
\(\tfrac{c}{12}m(m^2-1)\) of \([L_m, L_{-m}]\) is the divergent
sum \(\sum_p 2 = +\infty\), which is not a scalar endomorphism.
Accordingly, \(\mathcal{V}^{\mathrm{prim}}\) carries the structure
of a Witt module but not a VOA.
\end{corollary}

\section{The Natural Boundary of the Prime Zeta Function}
\label{v4-arith:pol:prime-zeta-obstruction}

\begin{theorem}[natural boundary of prime zeta]
\label{v4-arith:thm:prime-zeta-natural-boundary}
\ClaimStatusProvedElsewhere (Landau--Walfisz 1920; Fr\"oberg 1968).
The prime zeta function \(\zeta_{\mathbb{P}}(s)=\sum_{p}p^{-s}\) has
a natural boundary on \(\mathrm{Re}(s)=0\): logarithmic singularities
at \(s=\rho/n\) for every nontrivial zero \(\rho\) of \(\zeta\) and every
\(n\geq 1\) are dense on \(\mathrm{Re}(s)=0\).
\(\zeta_{\mathbb{P}}\) cannot be analytically continued past the
imaginary axis.
\end{theorem}

\begin{corollary}[\(\zeta_{\mathbb{P}}(0)\) is not defined]
\label{v4-arith:cor:zeta-P-0-undefined}
\ClaimStatusProvedHere. The value \(\zeta_{\mathbb{P}}(0)\) lies on
the natural boundary \(\mathrm{Re}(s)=0\) and is not defined by
analytic continuation. The regularisation \(c=2\zeta_{\mathbb{P}}(0)\)
is therefore not a finite value; the proposal \(c=-1\) via this route
is ill-formed.
\end{corollary}

\begin{remark}[what \(\zeta(0)\) regularises]
The value \(\zeta(0)=-1/2\) is the zeta-regularised sum
\(\sum_{n\geq 1}1\) over all positive integers.
The prime sum \(\sum_p 1\) is over a density-zero subset of
\(\mathbb{Z}_{>0}\); its zeta regularisation requires \(\zeta_{\mathbb{P}}\),
not \(\zeta\). The proposal \(c=2\zeta(0)=-1\) regularises the wrong
index set.
\end{remark}

\section{Central Charge as Residue}
\label{v4-arith:pol:c-as-residue}

The per-prime charges \(c^{(p)}=2\) (\Cref{v4-arith:thm:per-prime-c}) diverge
under naive summation; the prime zeta obstruction (\Cref{v4-arith:cor:zeta-P-0-undefined})
forecloses the \(s=0\) route.
The surviving finite regularisation is the residue at the pole
of the Euler product of the per-prime partition traces.

\begin{theorem}[regularised global central charge]
\label{v4-arith:thm:regularised-c}
\ClaimStatusProvedHere. A finite regularisation of the global
central charge compatible with the functional equation \(\xi(s)=\xi(1-s)\)
is the residue at the pole of \(\zeta\):
\begin{equation}
c_{\mathrm{global}}\;=\;2\cdot\mathrm{Res}_{s=1}\zeta(s)\;=\;2.
\end{equation}
Uniqueness of this regularisation among all functional-equation-compatible
procedures is pending.
\end{theorem}

\begin{proof}[Sketch]
The per-prime partition trace \(\mathrm{Tr}_{\mathcal{H}^{(p)}}p^{-sL_{0}}\)
(\Cref{v4-arith:def:primon-va-naive}) assembles via Euler product into
\(\xi(s)\) after archimedean completion (archimedean fibre contributes
\(\pi^{-s/2}\Gamma(s/2)\)).
Per-prime \(c^{(p)}=2\) (\Cref{v4-arith:thm:per-prime-c}).
The trace diverges at \(s=1\) with a simple pole; the residue extracts
the leading coefficient: \(2\cdot\mathrm{Res}_{s=1}\zeta=2\cdot 1=2\).
\end{proof}

\section{Witt versus Virasoro}
\label{v4-arith:pol:witt-virasoro}

\begin{theorem}[Witt acts algebraically; Virasoro central term diverges]
\label{v4-arith:thm:witt-virasoro}
\ClaimStatusProvedHere. The operators \(L_{m}=\sum_{p}L_{m}^{(p)}\)
are well-defined on every restricted vector (the sum is finite by the
restricted-tensor-product condition) and satisfy the Witt algebra
relations with no central term. The Virasoro central term
\(\tfrac{c}{12}m(m^2-1)\) is the divergent sum \(\sum_p 2\),
which is not a scalar.

% The statement is plausible (the central charge is a topological invariant
% of the Heisenberg representation), but a complete argument ruling out
% all normal-ordering procedures should be supplied.
\end{theorem}

\begin{remark}[soft-algebra interpretation]
This matches the Hartnoll--Yang reading of
\(\mathcal{V}^{\mathrm{prim}}\) as a partition function on the
gravitational superspace: asymptotic symmetries are Witt-level
(reparametrisation invariance); quantum anomalies live on the
genuine Virasoro-level. The arithmetic version has Witt
asymptotic symmetry but no honest Virasoro lift.
\end{remark}

\section{The Functional Equation as Arithmetic Modular Invariance}
\label{v4-arith:pol:functional-equation-modular}

\begin{theorem}[arithmetic modular invariance]
\label{v4-arith:thm:arithmetic-modular-invariance}
\ClaimStatusProvedHere. The partition function of
\(\mathcal{V}^{\mathrm{prim}}\) is the completed Riemann zeta
\(\xi(s)=\tfrac{1}{2}s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)\).
Its modular invariance is the functional equation \(\xi(s)=\xi(1-s)\):
the arithmetic analogue of \(\tau\to-1/\tau\).
The genus-\(1\) modular-form trace \(Z(\tau)=\mathrm{tr}\,q^{L_{0}-c/24}\)
is the complex-archimedean specialisation;
the global arithmetic object is \(\xi(s)\).
\end{theorem}

\section{The Arakelov Pairing Replaces the Weyl Anomaly}
\label{v4-arith:pol:arakelov-pairing}

\begin{proposition}[Deligne pairing = Weyl anomaly analogue]
\label{v4-arith:prop:deligne-pairing}
\ClaimStatusProvedElsewhere. Under the Deligne pairing on Arakelov
line bundles, \(\langle\omega_{p},\omega_{p}\rangle_{\mathrm{Arakelov}}=\log p\).
Summed over primes this reproduces (up to regularisation) the
logarithmic derivative of \(\zeta\).
\end{proposition}

\begin{remark}[Weyl anomaly versus height-theoretic anomaly]
On \(\mathbb{C}\), the Weyl anomaly is \(\langle T^{\mu}_{\mu}\rangle=-cR/24\pi\);
the Polyakov--Liouville action realises this as the central observable.
On \(\overline{C}=\overline{\mathrm{Spec}\,\mathbb{Z}}\), the role of
the Weyl anomaly is played by the Arakelov metric contribution to the
height pairing, \(\log p\) per prime, which is height-theoretic in origin.
The two anomalies are analogous; they are distinct because the geometric
context differs (complex curve versus arithmetic compactification).
\end{remark}

\section{The Arakelov--\(\zeta\) Structure of the Central Charge}
\label{v4-arith:pol:arakelov-zeta-structure}

The central charge of \(\mathcal{V}^{\mathrm{prim}}\) is finite and equals \(2\):
a residue at the pole of \(\zeta\), not a sum over primes.
The per-prime value \(c^{(p)}=2\) (\Cref{v4-arith:thm:per-prime-c})
is stable under \(\log p\)-rescaling; the divergence of the naive sum
\(\sum_p c^{(p)}\) is the obstruction to the Virasoro action
(\Cref{v4-arith:cor:no-virasoro}).
The prime zeta obstruction (\Cref{v4-arith:cor:zeta-P-0-undefined})
eliminates the \(c=-1\) proposals;
the residue regularisation (\Cref{v4-arith:thm:regularised-c})
is the unique procedure compatible with the functional equation.
The Witt action on each restricted vector is algebraically exact
(\Cref{v4-arith:thm:witt-virasoro});
its arithmetic analogue of modular invariance is the functional equation
\(\xi(s)=\xi(1-s)\) (\Cref{v4-arith:thm:arithmetic-modular-invariance}),
and the Weyl anomaly is replaced by the Arakelov--Deligne height pairing
\(\langle\omega_p,\omega_p\rangle=\log p\) (\Cref{v4-arith:prop:deligne-pairing}).
