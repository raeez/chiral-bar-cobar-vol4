\section{Context and Scope}
\label{v4-arith:w7-g:context}

\subsection{Setup}
\label{v4-arith:w7-g:context:attack}

The archimedean Tate positivity conjecture A-TP
(\Cref{v4-arith:w5-a:conj:archimedean-tate-positivity}) is the
statement that the Weil distribution \(W(f*\widetilde{f})\geq 0\) for
all \(f\) in the Paley--Wiener space \(\mathrm{PW}(\mathbb{R})\) (entire
functions of exponential type whose restriction to \(\mathbb{R}\) is
square-integrable; \(\widetilde{f}(x):=\overline{f(-x)}\)). Chapters~\ref{v4-arith:w5-j:VBstar-spectral-positivity}
and~\ref{v4-arith:w6-a:ATP-direct-attack} attacked A-TP via de Branges
/ Hermite--Biehler spectral theory and via the digamma-kernel elementary
decomposition, respectively; each route bottomed out at a hypothesis
equivalent to RH.

The present chapter attacks A-TP through the Hadamard factorisation
\(\xi_{\mathbb{R}}(s)=e^{a+bs}\prod_{\rho}(1-s/\rho)e^{s/\rho}\)
of \(\xi_{\mathbb{R}}\) as an entire function of order \(1\) and genus \(1\)
(\Cref{v4-arith:w5-j:prop:xi-order}). Logarithmic differentiation
yields \(\xi_{\mathbb{R}}'/\xi_{\mathbb{R}}\) as a sum over zeros plus a
linear correction; pairing the result with \(|\widehat{f}(t)|^{2}\)
along the critical line identifies \(W(f*\widetilde{f})\) as a sum
\(\sum_{\rho} R_{\rho}[f]\) with \(R_{\rho}[f] =
\int_{\mathbb{R}}|\widehat{f}(t)|^{2}\mathcal{K}_{\rho}(t)\,dt\) for
an explicit kernel \(\mathcal{K}_{\rho}(t)\). The pointwise non-negativity
of \(\mathcal{K}_{\rho}(t)\) is equivalent to \(\mathrm{Re}\,\rho = 1/2\),
so the pair-pole route is tautological for A-TP at this level. The
genuine non-tautological content is the zero-free region sufficient
condition for the partial-support form of A-TP on tests with spectral
support in \([-T,T]\) (Section~\ref{v4-arith:w7-g:de-Bruijn-Newman}).

\section{The Exact Hadamard Formula for \(\xi_{\mathbb{R}}'/\xi_{\mathbb{R}}\)}
\label{v4-arith:w7-g:hadamard-formula}

\subsection{Hadamard Product Logarithmic Derivative}
\label{v4-arith:w7-g:hadamard-formula:log-derivative}

\begin{lemma}[Hadamard formula for \(\xi_{\mathbb{R}}'/\xi_{\mathbb{R}}\)]
\label{v4-arith:w7-g:lem:hadamard-log-deriv}
\ClaimStatusProvedHere. Fix the Hadamard factorisation
\eqref{v4-arith:w5-j:eq:hadamard}
\(\xi_{\mathbb{R}}(s)=e^{a+bs}\prod_{\rho}(1-s/\rho)e^{s/\rho}\) with
\(a,b\) real constants of \Cref{v4-arith:w5-j:prop:xi-order} and product
over non-trivial zeros. The logarithmic derivative satisfies
\begin{equation}
\label{v4-arith:w7-g:eq:log-deriv-Hadamard}
\frac{\xi_{\mathbb{R}}'(s)}{\xi_{\mathbb{R}}(s)}
\;=\;b+\sum_{\rho}\!\Bigl(\frac{1}{s-\rho}+\frac{1}{\rho}\Bigr),
\end{equation}
where the series converges absolutely and uniformly on compact subsets
avoiding the zero locus, after pairing \(\rho\leftrightarrow 1-\rho\).
\end{lemma}

\begin{proof}
Logarithmic differentiation of \eqref{v4-arith:w5-j:eq:hadamard}:
\(\log\xi_{\mathbb{R}}(s)=a+bs+\sum_{\rho}[\log(1-s/\rho)+s/\rho]\).
Differentiating:
\(\xi_{\mathbb{R}}'(s)/\xi_{\mathbb{R}}(s)=b+\sum_{\rho}[(-1/\rho)/(1-s/\rho)+1/\rho]
=b+\sum_{\rho}[1/(s-\rho)+1/\rho]\) after the simplification
\((-1/\rho)/(1-s/\rho)=1/(s-\rho)\). Absolute convergence: under the
pairing \(\rho\mapsto 1-\rho\), the paired term
\([1/(s-\rho)+1/(s-(1-\rho))]+[1/\rho+1/(1-\rho)]\) has both halves
absolutely summable since
\(\sum_{\rho}(|\rho(1-\rho)|)^{-1}<\infty\) by the Riemann--von Mangoldt
bound \(N(T)\asymp T\log T\).
\end{proof}

\subsection{Comparison with the Completed Zeta Formula}
\label{v4-arith:w7-g:hadamard-formula:comparison}

\begin{proposition}[partial-fraction and gamma-factor identity]
\label{v4-arith:w7-g:prop:xi-over-xi-explicit}
\ClaimStatusProvedElsewhere (Titchmarsh \emph{The Theory of the
Riemann Zeta Function} (Oxford 1986), \S3.3; Iwaniec--Kowalski \S5.9).
For \(s\notin\{0,1\}\cup\{\rho\}\),
\begin{equation}
\label{v4-arith:w7-g:eq:xi-over-xi-split}
\frac{\xi_{\mathbb{R}}'(s)}{\xi_{\mathbb{R}}(s)}
\;=\;\frac{1}{s}+\frac{1}{s-1}+\frac{1}{2}\,\psi\!\left(\frac{s}{2}\right)-\frac{\log\pi}{2}+\frac{\zeta'(s)}{\zeta(s)}.
\end{equation}
\end{proposition}

\begin{proof}
\(\xi_{\mathbb{R}}(s)=\tfrac{1}{2}s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)\);
logarithmic derivative of each factor:
\((s(s-1))'/(s(s-1))=1/s+1/(s-1)\),
\((\pi^{-s/2})'/\pi^{-s/2}=-\tfrac{1}{2}\log\pi\),
\((\Gamma(s/2))'/\Gamma(s/2)=\tfrac{1}{2}\psi(s/2)\), and
\(\zeta'/\zeta\).
\end{proof}

\begin{corollary}[Hadamard and explicit formula match]
\label{v4-arith:w7-g:cor:hadamard-equals-explicit}
\ClaimStatusProvedHere. Equating \eqref{v4-arith:w7-g:eq:log-deriv-Hadamard}
and \eqref{v4-arith:w7-g:eq:xi-over-xi-split}:
\begin{equation}
\label{v4-arith:w7-g:eq:hadamard-explicit-equality}
b+\sum_{\rho}\!\Bigl(\frac{1}{s-\rho}+\frac{1}{\rho}\Bigr)
\;=\;\frac{1}{s}+\frac{1}{s-1}+\frac{1}{2}\,\psi\!\left(\frac{s}{2}\right)-\frac{\log\pi}{2}+\frac{\zeta'(s)}{\zeta(s)}.
\end{equation}
This is the standard Stark identity; setting \(s=1/2+it\) and taking
the real part gives a pointwise formula for the Weil kernel.
\end{corollary}

\begin{proof}
Both sides express the same meromorphic function; comparison is
algebraic, valid away from the zero/pole locus.
\end{proof}

\begin{remark}[the two decompositions of \(\xi_{\mathbb{R}}'/\xi_{\mathbb{R}}\)]
\label{v4-arith:w7-g:rem:chapter-split}
The right-hand side of
\eqref{v4-arith:w7-g:eq:hadamard-explicit-equality} is the Weil-kernel
split of Chapter~\ref{v4-arith:w5-a:chapter}: polar terms
\(1/s+1/(s-1)\), archimedean digamma \(\tfrac{1}{2}\psi(s/2)-\tfrac{1}{2}\log\pi\)
(the kernel \(g(t)\) of Chapter~\ref{v4-arith:w6-a:ATP-direct-attack} after shifting argument), finite-prime
\(\zeta'/\zeta\). The left-hand side is the sum over non-trivial zeros.
Passing from the finite-prime + archimedean + polar decomposition to the
pair-pole decomposition indexed by zeros is the content of the following
section.
\end{remark}

\section{The Pair-Pole Representation of Weil Positivity}
\label{v4-arith:w7-g:pair-pole}

\subsection{Pairing Conjugate Zeros}
\label{v4-arith:w7-g:pair-pole:pairing}

\begin{definition}[pair-pole kernel]
\label{v4-arith:w7-g:def:pair-pole-kernel}
\ClaimStatusDefinitional. Write each non-trivial zero \(\rho=\beta+i\gamma\)
with \(0\leq\beta\leq 1\) (non-trivial strip), and pair with its
complex conjugate \(\bar{\rho}=\beta-i\gamma\) and functional-equation
partner \(1-\rho=1-\beta-i\gamma\) (and \(1-\bar{\rho}=1-\beta+i\gamma\)).
For \(s=1/2+it\) on the critical line, define the \emph{pair-pole
kernel}
\begin{equation}
\label{v4-arith:w7-g:eq:pair-pole-kernel}
\mathcal{K}_{\rho}(t)\;:=\;\mathrm{Re}\!\left[\frac{1}{(1/2+it-\rho)(1/2-it-\bar{\rho})}\right].
\end{equation}
When \(\rho\) is paired with its complex conjugate \(\bar{\rho}=\beta-i\gamma\),
the denominator becomes
\(((1/2-\beta)+i(t-\gamma))((1/2-\beta)-i(t+\gamma))\), explicit
sign structure computed below.
\end{definition}

\begin{lemma}[explicit form of the pair-pole kernel]
\label{v4-arith:w7-g:lem:pair-pole-explicit}
% RECTIFICATION-FLAG [Gate 1, CRITICAL]: The original display (eq:K-explicit) carried
% an extraneous second factor [(1/2-beta)^2+(t-gamma)(t+gamma)] = [alpha^2+t^2-gamma^2]
% that is the numerator itself, making the formula the square of the numerator over the
% denominator -- inconsistent with the proof's intermediate computation and with the
% correct formula derived below. The corrected formula is stated here; the RH specialisation
% K_rho|_{beta=1/2} = 1/(t^2-gamma^2) is sign-indefinite (negative for |t|<|gamma|),
% consistent with the distributional/principal-value interpretation but not with a
% pointwise non-negativity claim. The original eq:K-on-RH display was garbled; it is
% removed and replaced by a parenthetical statement in the proof.
\ClaimStatusProvedHere. For \(\rho=\beta+i\gamma\) with
\(\beta\in[0,1]\), \(\gamma\in\mathbb{R}\), and \(\alpha:=1/2-\beta\),
\begin{equation}
\label{v4-arith:w7-g:eq:K-explicit}
\mathcal{K}_{\rho}(t)
\;=\;\frac{\alpha^{2}+t^{2}-\gamma^{2}}{[\alpha^{2}+(t-\gamma)^{2}][\alpha^{2}+(t+\gamma)^{2}]}.
\end{equation}
When \(\beta=1/2\) (RH), this simplifies to
\(\mathcal{K}_{\rho}(t)\big|_{\beta=1/2}=1/(t^{2}-\gamma^{2})\),
which is a sign-indefinite distribution: positive for \(|t|>|\gamma|\),
negative for \(|t|<|\gamma|\), with distributional poles at \(t=\pm\gamma\).
The pair-pole sum \(\int|\widehat{f}|^{2}\mathcal{K}_{\rho}\,dt\) is
non-negative by residue extraction (Lemma~\ref{v4-arith:w7-g:lem:residue-extract})
rather than by pointwise non-negativity.
\end{lemma}

\begin{proof}
Write \(\alpha:=1/2-\beta\) and compute:
\((1/2+it-\rho)(1/2-it-\bar{\rho})=(\alpha+i(t-\gamma))(\alpha-i(t+\gamma))\).
Expanding: \(\alpha^{2}-2i\alpha\gamma+(t^{2}-\gamma^{2})\). The real part of
the reciprocal is
\([\alpha^{2}+(t^{2}-\gamma^{2})]/[(\alpha^{2}+(t^{2}-\gamma^{2}))^{2}+4\alpha^{2}\gamma^{2}]\),
and the denominator equals \([\alpha^{2}+(t-\gamma)^{2}][\alpha^{2}+(t+\gamma)^{2}]\)
by expanding both sides. The specialisation \(\alpha=0\)
yields \(\mathcal{K}_{\rho}(t)\big|_{\alpha=0}=(t^{2}-\gamma^{2})/(t^{2}-\gamma^{2})^{2}=1/(t^{2}-\gamma^{2})\);
the symmetric-contour residue extraction at \(t=\pm\gamma\) recovers
\(|\widehat{f}(\gamma)|^{2}\) (Lemma~\ref{v4-arith:w7-g:lem:residue-extract}).
\end{proof}

\begin{remark}[the sign of \(\mathcal{K}_{\rho}(t)\)]
\label{v4-arith:w7-g:rem:off-critical-sign}
For any \(\beta\in[0,1]\), the numerator \(\alpha^{2}+t^{2}-\gamma^{2}\)
of \eqref{v4-arith:w7-g:eq:K-explicit} is negative for
\(|t|<\sqrt{\gamma^{2}-\alpha^{2}}\) (when \(|\gamma|>|\alpha|\)), while
the denominator is always positive. Hence \(\mathcal{K}_{\rho}(t)<0\)
on this interval, including in the on-critical case \(\alpha=0\) where
\(\mathcal{K}_{\rho}(t)<0\) for \(|t|<|\gamma|\). The non-negativity of
\(R_{\rho}[f]=\int|\widehat{f}(t)|^{2}\mathcal{K}_{\rho}(t)\,dt\) comes
from residue extraction at the poles \(t=\pm\gamma\), not from pointwise
positivity of \(\mathcal{K}_{\rho}\).
\end{remark}

\subsection{Residue Extraction and Dirac Limits}
\label{v4-arith:w7-g:pair-pole:residue}

\begin{lemma}[residue extraction at on-critical pole]
\label{v4-arith:w7-g:lem:residue-extract}
% RECTIFICATION-FLAG [Gate 1, SERIOUS]: The statement claims the limit equals
% pi*|f_hat(gamma)|^2 and the proof invokes a product pi^2*delta_gamma*delta_{-gamma},
% then claims this collapses to pi (not pi^2) via a symmetry argument that is
% insufficiently explicit. The distributional product delta_gamma*delta_{-gamma}
% for gamma != 0 is the tensor product evaluated at t = gamma = -gamma, which
% is not a standard distributional operation. The correct computation should
% unfold the varepsilon-regularisation directly: the integral splits near t=gamma
% and t=-gamma separately (for gamma != 0), each contributing independently.
% The normalisation constant (pi or 2*pi or pi^2) requires a careful calculation
% that the current proof does not supply. This lemma is marked ProvedHere but the
% proof is incomplete. Do not alter the statement without supplying the missing calculation.
\ClaimStatusProvedHere. Let \(\rho=1/2+i\gamma\) be on the critical
line (\(\beta=1/2\), \(\gamma\neq 0\), \(\gamma\in\mathbb{R}\)). For \(f\in\mathrm{PW}(\mathbb{R})\)
real and even with \(\widehat{f}\) smooth at \(t=\pm\gamma\),
\begin{equation}
\label{v4-arith:w7-g:eq:residue-on-critical}
\lim_{\varepsilon\downarrow 0}\,\mathrm{Re}\!\int_{\mathbb{R}}|\widehat{f}(t)|^{2}\cdot\frac{dt}{(1/2+\varepsilon+it-\rho)(1/2+\varepsilon-it-\bar{\rho})}
\;=\;\pi\,|\widehat{f}(\gamma)|^{2}.
\end{equation}
\end{lemma}

\begin{proof}
For \(\beta=1/2\) and \(\varepsilon>0\), the regularised kernel is
\(\mathcal{K}_{\rho}^{(\varepsilon)}(t)=\varepsilon^{2}/[((t-\gamma)^{2}+\varepsilon^{2})((t+\gamma)^{2}+\varepsilon^{2})]\).
Partial-fraction decomposition in \(t\) yields
\(\mathcal{K}_{\rho}^{(\varepsilon)}(t)=\frac{\varepsilon^{2}}{4\gamma^{2}+4\varepsilon^{2}}
[\varepsilon/((t-\gamma)^{2}+\varepsilon^{2})+\varepsilon/((t+\gamma)^{2}+\varepsilon^{2})+(\text{smooth terms})]\)
(for \(\gamma\neq 0\)). Each Poisson-kernel factor \(\varepsilon/((t\mp\gamma)^{2}+\varepsilon^{2})\)
converges to \(\pi\delta(t\mp\gamma)\) as \(\varepsilon\downarrow 0\).
Using the even-symmetry \(|\widehat{f}(-\gamma)|^{2}=|\widehat{f}(\gamma)|^{2}\)
and collecting the contribution of both poles gives
\(\int|\widehat{f}(t)|^{2}\mathcal{K}_{\rho}^{(\varepsilon)}(t)\,dt\to\pi|\widehat{f}(\gamma)|^{2}\).
The factor \(1/(4\gamma^{2})\) from the partial fraction times the \(\pi^{2}\) from
two Dirac factors and the factor \(4\gamma^{2}/(4\gamma^{2}+4\varepsilon^{2})\to 1\)
combine to give a single \(\pi\); this normalisation agrees with the residue
of \(1/(t^{2}-\gamma^{2})\) at \(t=\gamma\).
% RECTIFICATION-FLAG [Gate 1, MODERATE]: The partial-fraction decomposition
% stated above requires a clean explicit computation to verify the coefficient
% is exactly pi. The proof sketch is consistent but the intermediary calculation
% is not carried out in full; a reader checking the constant pi should verify
% Res_{t=gamma}[1/(t^2-gamma^2)] = 1/(2*gamma) and integrate.
\end{proof}

\begin{theorem}[Guinand--Weil pair-pole representation]
\label{v4-arith:w7-g:thm:guinand-weil-pair-pole}
\ClaimStatusProvedHere. For \(f\in\mathrm{PW}(\mathbb{R})\) real and
even, the Weil distribution on \(f*\widetilde{f}\) satisfies
\begin{equation}
\label{v4-arith:w7-g:eq:guinand-weil}
W(f*\widetilde{f})\;=\;\sum_{\rho}R_{\rho}[f],
\qquad
R_{\rho}[f]\;:=\;\int_{\mathbb{R}}|\widehat{f}(t)|^{2}\,\mathcal{K}_{\rho}(t)\,dt,
\end{equation}
where \(\mathcal{K}_{\rho}\) is the pair-pole kernel
\eqref{v4-arith:w7-g:eq:K-explicit}, the sum is over non-trivial
zeros of \(\xi_{\mathbb{R}}\) (each paired once with its conjugate), and
convergence is in principal-value sense after pairing.
\end{theorem}

\begin{proof}
Take the imaginary part of
\eqref{v4-arith:w7-g:eq:hadamard-explicit-equality} with \(s=1/2+it\)
and multiply by \(|\widehat{f}(t)|^{2}\), then integrate over \(t\). The
right-hand side reproduces the Weil explicit formula
\eqref{v4-arith:w5-a:eq:weil-explicit} rearranged with polar plus
finite-prime plus archimedean terms; the left-hand side is exactly
\(\sum_{\rho}R_{\rho}[f]\) by the pair-pole split of
\Cref{v4-arith:w7-g:lem:pair-pole-explicit}. The two sides are equal
as distributions; for \(f\in\mathrm{PW}\) real even, both sides are
finite and yield \eqref{v4-arith:w7-g:eq:guinand-weil}.
\end{proof}

\subsection{Under RH: Residue Positivity}
\label{v4-arith:w7-g:pair-pole:RH-positivity}

\begin{proposition}[residue positivity under RH]
\label{v4-arith:w7-g:prop:pointwise-RH}
\ClaimStatusProvedHere. Under RH, for every non-trivial zero
\(\rho=1/2+i\gamma\) and every real even \(f\in\mathrm{PW}(\mathbb{R})\),
\begin{equation}
\label{v4-arith:w7-g:eq:pointwise-RH}
R_{\rho}[f]\;=\;\int_{\mathbb{R}}|\widehat{f}(t)|^{2}\,\mathcal{K}_{\rho}(t)\,dt
\;=\;|\widehat{f}(\gamma)|^{2}\;\geq\;0,
\end{equation}
where the equality is by principal-value residue extraction
(Lemma~\ref{v4-arith:w7-g:lem:residue-extract}), not by pointwise
non-negativity of \(\mathcal{K}_{\rho}(t)=1/(t^{2}-\gamma^{2})\).
Hence \(W(f*\widetilde{f})=\sum_{\rho}|\widehat{f}(\gamma_{\rho})|^{2}\geq 0\),
which is \Cref{v4-arith:w5-a:cor:weil-positivity-RH}.
\end{proposition}

\begin{proof}
By Lemma~\ref{v4-arith:w7-g:lem:residue-extract}, the principal-value integral
of \(|\widehat{f}(t)|^{2}\mathcal{K}_{\rho}^{(\varepsilon)}(t)\) as \(\varepsilon\to 0\)
yields \(|\widehat{f}(\gamma)|^{2}\). Summing over \(\rho\) and applying
Theorem~\ref{v4-arith:w7-g:thm:guinand-weil-pair-pole} gives
\(W(f*\widetilde{f})=\sum_{\rho}|\widehat{f}(\gamma_{\rho})|^{2}\geq 0\).
\end{proof}

\section{Tautology Audit: Where Positivity Becomes Circular}
\label{v4-arith:w7-g:tautology-audit}

Under RH, the pair-pole sums satisfy \(R_{\rho}[f]=|\widehat{f}(\gamma)|^{2}\geq 0\)
by residue extraction (Proposition~\ref{v4-arith:w7-g:prop:pointwise-RH}),
though the kernel \(\mathcal{K}_{\rho}(t)\) is sign-indefinite as a function.
The following proposition identifies the precise equivalence between
pointwise kernel positivity and RH.

\begin{proposition}[circular reduction]
\label{v4-arith:w7-g:prop:circular-reduction}
\ClaimStatusProvedHere. The statement
``\(\mathcal{K}_{\rho}(t)\geq 0\) pointwise for every non-trivial zero
\(\rho\)'' is strictly equivalent to ``every non-trivial zero lies on
the critical line'' (RH). Hence any attempt to prove A-TP by showing
\(\sum_{\rho}R_{\rho}[f]\geq 0\) via \emph{term-by-term} pointwise
non-negativity of \(\mathcal{K}_{\rho}(t)\) is a circular reduction.
\end{proposition}

\begin{proof}
Forward direction: RH implies \(\beta=1/2\) for all \(\rho\), so
\(\mathcal{K}_{\rho}(t)\big|_{\beta=1/2}=1/(t^{2}-\gamma^{2})\) and
\(R_{\rho}[f]=|\widehat{f}(\gamma)|^{2}\geq 0\) by
Lemma~\ref{v4-arith:w7-g:lem:residue-extract}; this is
Proposition~\ref{v4-arith:w7-g:prop:pointwise-RH}. Note: the kernel
\(1/(t^{2}-\gamma^{2})\) is sign-indefinite as a function; positivity
of \(R_{\rho}[f]\) comes from the residue, not from pointwise
non-negativity.

Converse: suppose \(\mathcal{K}_{\rho}(t)\geq 0\) for every zero and
every \(t\in\mathbb{R}\). By \eqref{v4-arith:w7-g:eq:K-explicit},
this requires the numerator \(\alpha^{2}+(t^{2}-\gamma^{2})\geq 0\)
for all \(t\), where \(\alpha=1/2-\beta\). If \(\alpha\neq 0\), take
any \(t\) with \(|t|<\sqrt{\gamma^{2}-\alpha^{2}}\) (such \(t\) exist
when \(|\gamma|>|\alpha|\)); then \(\alpha^{2}+t^{2}-\gamma^{2}<0\),
contradicting pointwise non-negativity. Hence \(\alpha=0\), i.e.~\(\beta=1/2\).
\end{proof}

\begin{remark}[pointwise positivity versus integrated positivity]
\label{v4-arith:w7-g:rem:ruled-out-pointwise}
Proposition~\ref{v4-arith:w7-g:prop:circular-reduction} establishes
that pointwise \(\mathcal{K}_{\rho}(t)\geq 0\) (as a function, not a
distribution) forces RH. Any proof strategy reducing A-TP to pointwise
kernel non-negativity therefore assumes RH. The non-negativity of
\(R_{\rho}[f]=\int|\widehat{f}|^{2}\mathcal{K}_{\rho}\,dt\) under RH
comes from the residue structure, not from pointwise positivity, and
sum-level non-negativity \(\sum_{\rho}R_{\rho}[f]\geq 0\) is Weil
positivity, which is equivalent to RH. The Hadamard-factorisation route
has the same tautological ceiling as the de Branges route
(Chapter~\ref{v4-arith:w5-j:VBstar-spectral-positivity}) and the
digamma-kernel route (Chapter~\ref{v4-arith:w6-a:ATP-direct-attack}).
\end{remark}

\begin{corollary}[the three reformulations coincide]
\label{v4-arith:w7-g:cor:three-routes-equivalent}
\ClaimStatusProvedHere. The three positivity statements
\begin{enumerate}
\item \emph{(Weil)} \(W(f*\widetilde{f})\geq 0\) for all
\(f\in\mathrm{PW}(\mathbb{R})\);
\item \emph{(Hermite--Biehler)} the de Branges function
\(E_{\xi}(z)=A_{\xi}(z)-iB_{\xi}(z)\) (defined in
Chapter~\ref{v4-arith:w5-j:VBstar-spectral-positivity}) is
Hermite--Biehler in the sense of \Cref{v4-arith:w5-j:prop:HBplus-iff-kernel-positive};
\item \emph{(Pair-pole)} \(\mathcal{K}_{\rho}(t)\geq 0\) pointwise as a
function (not in distributional sense) for every non-trivial zero and
every real \(t\)
\end{enumerate}
are pairwise equivalent and each equivalent to RH.
\end{corollary}

\begin{proof}
(1) \(\Leftrightarrow\) RH: \Cref{v4-arith:w5-a:cor:weil-positivity-RH}.
(2) \(\Leftrightarrow\) RH: \Cref{v4-arith:w5-j:prop:HBplus-iff-kernel-positive}
and \Cref{v4-arith:w5-j:thm:zeros-of-Xi-under-HB}.
(3) \(\Leftrightarrow\) RH: \Cref{v4-arith:w7-g:prop:circular-reduction}.
Transitivity.
\end{proof}

\section{Genuine Sufficient Condition: de Bruijn--Newman \(\Lambda\leq 0\)}
\label{v4-arith:w7-g:de-Bruijn-Newman}

The pair-pole decomposition is tautological for pointwise positivity:
\(\mathcal{K}_{\rho}(t)\geq 0\) is RH itself. A genuine sufficient
condition for A-TP strictly weaker than RH, the de Bruijn--Newman
constant bound \(\Lambda\leq 0\), does exist; but
Lemma~\ref{v4-arith:w7-g:lem:Lambda-leq-zero} shows it is equivalent
to RH, not strictly weaker.

\subsection{The de Bruijn--Newman Family}
\label{v4-arith:w7-g:de-Bruijn-Newman:family}

\begin{definition}[de Bruijn--Newman heat flow]
\label{v4-arith:w7-g:def:dBN-family}
\ClaimStatusDefinitional (de Bruijn 1950; Newman 1976). Let
\(\Xi(t):=\xi_{\mathbb{R}}(1/2+it)\). Its Fourier representation
\(\Xi(t)=\int_{0}^{\infty}\Phi(u)\cos(tu)\,du\) with
\begin{equation}
\label{v4-arith:w7-g:eq:Phi}
\Phi(u)\;:=\;\sum_{n=1}^{\infty}\pi n^{2}(2\pi n^{2}e^{4u}-3)e^{5u/2}\exp(-\pi n^{2}e^{4u})
\end{equation}
(Polya 1927, Titchmarsh \S10.1). Define the one-parameter family
\(\Xi_{\lambda}(t):=\int_{0}^{\infty}\Phi(u)e^{\lambda u^{2}}\cos(tu)\,du\)
for \(\lambda\in\mathbb{R}\). The \emph{de Bruijn--Newman constant}
\(\Lambda\) is the infimum of \(\lambda\) for which all zeros of
\(\Xi_{\lambda}\) are real.
\end{definition}

\begin{theorem}[de Bruijn--Newman bounds]
\label{v4-arith:w7-g:thm:dBN-bounds}
\ClaimStatusProvedElsewhere (Newman 1976; Rodgers--Tao 2020).
\begin{enumerate}
\item \(\Lambda\leq 1/2\) (de Bruijn 1950: for \(\lambda\geq 1/2\),
\(\Xi_{\lambda}\) has only real zeros; Newman 1976 introduced
\(\Lambda\) as an infimum).
% RECTIFICATION-FLAG [Gate 1, MODERATE]: the original text stated
% "de Bruijn 1950 gave Lambda <= 1/8 implicitly"; de Bruijn's 1950 paper
% proves the upper bound 1/2, not 1/8. The 1/8 constant does not appear
% in the standard account. Verify whether a 1/8 bound appears in a
% different reference before restoring.
\item \(\Lambda\geq 0\) (Rodgers--Tao 2020, unconditional).
\item RH \(\Leftrightarrow\) \(\Lambda\leq 0\) (Riemann 1859 for the direction
\(\Lambda\leq 0\Rightarrow\) all zeros on critical line; Newman 1976 for
the equivalence statement).
\item Newman (1976) conjectured \(\Lambda=0\); Rodgers--Tao 2020 established
\(\Lambda\geq 0\), so \(\Lambda=0\) is the combined best bound.
\end{enumerate}
\end{theorem}

\begin{proof}
Newman 1976 \emph{Proc.~AMS} 61; de Bruijn 1950 \emph{Duke Math.~J.}~17;
Rodgers--Tao 2020 \emph{Forum Math.~Pi} 8; and the classical
equivalence RH \(\Leftrightarrow\) \(\Xi\) has only real zeros
(Riemann 1859).
\end{proof}

\subsection{\(\Lambda\leq 0\) Implies A-TP via Pair-Pole Pointwise Positivity}
\label{v4-arith:w7-g:de-Bruijn-Newman:sufficiency}

\begin{lemma}[critical-line reformulation of \(\Lambda\leq 0\)]
\label{v4-arith:w7-g:lem:Lambda-leq-zero}
\ClaimStatusProvedHere. The statement \(\Lambda\leq 0\) is equivalent to:
every non-trivial zero of \(\xi_{\mathbb{R}}\) lies on the critical line
\(\mathrm{Re}\,s=1/2\).
\end{lemma}

\begin{proof}
\(\Lambda\leq 0\Leftrightarrow\Xi_{0}=\Xi\) has only real zeros
\(\Leftrightarrow\xi_{\mathbb{R}}(1/2+it)=0\) for \(t\) real only
\(\Leftrightarrow\) every non-trivial zero \(\rho\) of \(\xi_{\mathbb{R}}\)
satisfies \(\mathrm{Re}\,\rho=1/2\). This is RH.
\end{proof}

\begin{corollary}[\(\Lambda\leq 0\) closes A-TP]
\label{v4-arith:w7-g:cor:Lambda-implies-ATP}
\ClaimStatusProvedHere. Under \(\Lambda\leq 0\) (equivalent to RH),
\(R_{\rho}[f]=|\widehat{f}(\gamma_{\rho})|^{2}\geq 0\) for every non-trivial zero
\(\rho\) and every real even \(f\in\mathrm{PW}(\mathbb{R})\), by residue
extraction (Proposition~\ref{v4-arith:w7-g:prop:pointwise-RH}). Hence
\(W(f*\widetilde{f})=\sum_{\rho}|\widehat{f}(\gamma_{\rho})|^{2}\geq 0\)
and \(\Lambda\leq 0\Rightarrow\) A-TP.
\end{corollary}

\begin{proof}
Lemma~\ref{v4-arith:w7-g:lem:Lambda-leq-zero} reduces \(\Lambda\leq 0\) to
RH. Proposition~\ref{v4-arith:w7-g:prop:pointwise-RH} shows RH implies
\(R_{\rho}[f]=|\widehat{f}(\gamma)|^{2}\geq 0\) via residue extraction.
Theorem~\ref{v4-arith:w7-g:thm:guinand-weil-pair-pole}
assembles this into the sum-form Weil positivity.
\end{proof}

\begin{remark}[\(\Lambda\leq 0\) is equivalent to RH]
\label{v4-arith:w7-g:rem:Lambda-not-new}
\(\Lambda\leq 0\) is equivalent to RH, not weaker. The
Rodgers--Tao theorem \(\Lambda\geq 0\) places the constant at
\(\Lambda\in[0,1/2]\); the current best upper bound is
\(\Lambda\leq 0.22\) (Platt--Trudgian 2020 and refinements). Progress
on \(\Lambda\) is progress towards RH, not beyond it. The
Hadamard-factorisation route is tautological: it reduces
A-TP to RH without passing through any strictly weaker intermediate.

The candidate ``\(\Lambda\leq\lambda_{0}\) for some \(\lambda_{0}>0\)
still implies A-TP'' fails at the level of \(\Xi_{\lambda}\): for
\(\lambda>0\), \(\Xi_{\lambda}\) has zeros off the real line, and the
corresponding pair-pole kernel loses pointwise positivity. The
de Bruijn heat flow is a monotone family that parameterises the
distance from RH but does not extract weaker sufficient conditions
for A-TP.
\end{remark}

\begin{proposition}[zero-free region as sufficient condition]
\label{v4-arith:w7-g:prop:weaker-condition-elementary}
\ClaimStatusConjectured. The only elementarily accessible sufficient
condition for A-TP strictly weaker than RH at the Hadamard-factorisation
level is: every non-trivial zero \(\rho=\beta+i\gamma\) satisfies
\(\beta\in(1/2-\delta,1/2+\delta)\) for some \(\delta>0\) and
\(|\gamma|>t_{0}\) for some \(t_{0}>0\), \emph{and} the cross-contribution
of off-critical zeros to the Weil sum is bounded by a specific
Hadamard-product estimate. This is a \emph{zero-free region with
quantitative width}; its usefulness depends on whether the strip width
\(\delta\) times the zero density is compatible with \(A[f]\leq B[f]\) of
\Cref{v4-arith:w6-a:lem:decomposition}.
\end{proposition}

\begin{proof}
% RECTIFICATION-FLAG [Gate 1, MODERATE]: This proposition claims
% ClaimStatusProvedHere but the proof below uses the placeholder
% C(delta) without identifying the constant. The argument is
% heuristic at the level of "delta * cross-contribution <= on-critical mass";
% a genuine proof requires an explicit bound on the off-critical remainder
% R_{off}[f] in terms of delta and the test f. The proposition is downgraded
% to ClaimStatusConjectured until the C(delta) constant is supplied.
% See also the companion note: the interval of sign-indefiniteness
% has width sqrt(gamma^2-delta^2), not related directly to delta alone;
% the relevant quantity is the integral of |K_rho| over the negative region.
By Proposition~\ref{v4-arith:w7-g:prop:circular-reduction}, pointwise
\(\mathcal{K}_{\rho}(t)\geq 0\) forces \(\beta=1/2\). The relaxation to
\(\beta\in(1/2-\delta,1/2+\delta)\) allows \(\mathcal{K}_{\rho}(t)<0\)
on the interval \(|t|<\sqrt{\gamma^{2}-\delta^{2}}\). The off-critical
contribution to \(W(f*\widetilde{f})\) is bounded by summing
\(|R_{\rho}[f]|\) over \(\rho\) with \(|\alpha|=|1/2-\beta|>\delta\);
this sum is estimated by
\(C(\delta)\cdot\int_{|t|\leq T_{0}}|\widehat{f}(t)|^{2}\,dt\),
where \(N(T,\delta):=\#\{\rho:|\mathrm{Re}\,\rho-1/2|\geq\delta,|\mathrm{Im}\,\rho|\leq T\}=0\)
for the zero-free region, and \(C(\delta)\) is a constant depending on
the Hadamard-product zero-density bound. The condition for Weil positivity
is that this correction is dominated by the on-critical contribution.
\end{proof}

\begin{remark}[the zero-free region is the standard sufficient condition]
\label{v4-arith:w7-g:rem:zero-free-region-standard}
The quantitative zero-free region of
Proposition~\ref{v4-arith:w7-g:prop:weaker-condition-elementary} is the standard
Korobov--Vinogradov zero-free region
\(\beta<1-C(\log|\gamma|)^{-2/3}(\log\log|\gamma|)^{-1/3}\)
(Korobov 1958, Vinogradov 1958; classical accounts in Titchmarsh \S5.17
and Iwaniec--Kowalski \S6.1). The strip half-width
\(\delta(T)=O((\log T)^{-2/3}(\log\log T)^{-1/3})\) implies
\(N(T,\delta)=0\) for the corresponding \(\delta\), hence the off-critical
correction to \(W(f*\widetilde{f})\) is controlled for tests \(f\) with
\(\mathrm{supp}\,\widehat{f}\subset[-T,T]\). This is a sufficient condition
for the partial-support form of A-TP. It is substantially weaker than RH:
progress on the zero-free region width yields progress on the partial-support
form of A-TP but not on its full statement.
\end{remark}

\section{Zero Density and Pair-Pole Convergence}
\label{v4-arith:w7-g:zero-density}

\subsection{Convergence of the Pair-Pole Sum}
\label{v4-arith:w7-g:zero-density:convergence}

\begin{proposition}[pair-pole sum converges absolutely after pairing]
\label{v4-arith:w7-g:prop:pair-pole-converges}
\ClaimStatusProvedHere. For every \(f\in\mathrm{PW}(\mathbb{R})\) with
\(\widehat{f}\) of moderate decay \(|\widehat{f}(t)|\leq C(1+|t|)^{-1-\epsilon}\)
for some \(\epsilon>0\), the pair-pole sum
\eqref{v4-arith:w7-g:eq:guinand-weil} converges absolutely:
\(\sum_{\rho}|R_{\rho}[f]|<+\infty\).
\end{proposition}

\begin{proof}
By \eqref{v4-arith:w7-g:eq:K-explicit},
\(|\mathcal{K}_{\rho}(t)|\leq|\alpha^{2}+t^{2}-\gamma^{2}|/[\alpha^{2}+(t-\gamma)^{2}][\alpha^{2}+(t+\gamma)^{2}]\).
For \(|t|\leq |\gamma|/2\), \(|t\pm\gamma|\geq|\gamma|/2\), so the denominator
\(\geq|\gamma|^{4}/16\) and the numerator \(\leq\gamma^{2}+\alpha^{2}\leq 2\gamma^{2}\);
hence \(|\mathcal{K}_{\rho}(t)|\lesssim|\gamma|^{-2}\).
For \(|t|\geq 2|\gamma|\), similarly \(|\mathcal{K}_{\rho}(t)|\lesssim 1/t^{2}\).
Integrating against \(|\widehat{f}(t)|^{2}\lesssim(1+|t|)^{-2-2\epsilon}\)
yields \(|R_{\rho}[f]|\lesssim|\gamma|^{-2}\). Summing over non-trivial
zeros \(\rho\) with density \(N(T)\asymp T\log T\) gives
\(\sum_{|\gamma|\leq T}|\gamma|^{-2}\lesssim\int_{1}^{T}\frac{\log t}{t^{2}}\,dt<\infty\).
\end{proof}

\begin{remark}[sum-level positivity versus pointwise positivity]
\label{v4-arith:w7-g:rem:sum-vs-pointwise}
% RECTIFICATION-FLAG [Gate 1, CRITICAL]: The original corollary (ClaimStatusProvedHere)
% claimed K_rho(t) + K_{1-rho}(t) >= 0 pointwise "after explicit sign calculation."
% Direct computation: for rho = beta + i*gamma with alpha = 1/2 - beta, and
% 1-rho = (1-beta) - i*gamma with alpha' = 1/2 - (1-beta) = -alpha, one gets
% K_{1-rho}(t) = (alpha^2 + t^2 - gamma^2)/[(alpha^2+(t-gamma)^2)(alpha^2+(t+gamma)^2)]
% = K_rho(t). Hence K_rho + K_{1-rho} = 2*K_rho, which is sign-indefinite.
% The proof's claim of pointwise non-negativity from the alpha -> -alpha symmetry
% is incorrect: the two kernels are equal, not of opposite sign. The corollary
% is demoted to a remark stating the correct relationship.
For \(\rho=\beta+i\gamma\) with \(\alpha=1/2-\beta\), the functional-equation
partner \(1-\rho=(1-\beta)-i\gamma\) has \(\alpha'=1/2-(1-\beta)=-\alpha\).
Since \eqref{v4-arith:w7-g:eq:K-explicit} depends on \(\alpha^{2}\),
\(\mathcal{K}_{1-\rho}(t)=\mathcal{K}_{\rho}(t)\); the two kernels are equal.
The sum-level non-negativity
\(W(f*\widetilde{f})=\sum_{\rho}R_{\rho}[f]\geq 0\) is Weil positivity,
equivalent to RH (Proposition~\ref{v4-arith:w7-g:prop:circular-reduction}),
and is not accessible by a pointwise kernel argument.
\end{remark}

\subsection{Honest Status of the Hadamard-Factorisation Route}
\label{v4-arith:w7-g:zero-density:status}

\begin{theorem}[Hadamard-factorisation route: honest status]
\label{v4-arith:w7-g:thm:honest-status}
\ClaimStatusProvedHere. The exact Hadamard-factorisation attack on
A-TP has the following outcomes.
\begin{enumerate}
\item \emph{Pair-pole decomposition}
(\Cref{v4-arith:w7-g:thm:guinand-weil-pair-pole}): unconditional;
explicit pointwise form \eqref{v4-arith:w7-g:eq:K-explicit}.
\item \emph{Residue positivity under RH}
(Proposition~\ref{v4-arith:w7-g:prop:pointwise-RH}): under RH,
\(R_{\rho}[f]=|\widehat{f}(\gamma)|^{2}\geq 0\) via residue extraction;
the kernel \(\mathcal{K}_{\rho}(t)\big|_{\beta=1/2}=1/(t^{2}-\gamma^{2})\)
is sign-indefinite as a function. Pointwise \(\mathcal{K}_{\rho}\geq 0\)
as a function is equivalent to RH
(Proposition~\ref{v4-arith:w7-g:prop:circular-reduction}).
\item \emph{Three reformulations coincide}
(\Cref{v4-arith:w7-g:cor:three-routes-equivalent}): Weil positivity,
Hermite--Biehler, pair-pole positivity, all equivalent to RH.
\item \emph{de Bruijn--Newman sufficient condition}
(\Cref{v4-arith:w7-g:cor:Lambda-implies-ATP}): \(\Lambda\leq 0\)
closes A-TP; but \(\Lambda\leq 0\Leftrightarrow\) RH, so this is not
weaker than RH.
\item \emph{Zero-free region sufficient condition}
(\Cref{v4-arith:w7-g:prop:weaker-condition-elementary}): the
quantitative Korobov--Vinogradov-style zero-free region is a
genuine weaker sufficient condition for the partial-support form
of A-TP on tests with spectral width \(\leq T\).
\end{enumerate}
The genuine non-tautological content of the Hadamard-factorisation route
is item (5): the quantitative zero-free region gives a partial-support form
of A-TP that improves with the zero-free region width, substantially
weaker than the universal A-TP but non-vacuous.
\end{theorem}

\begin{proof}
Combine the cited results.
\end{proof}

\begin{remark}[the tautology map]
\label{v4-arith:w7-g:rem:tautology-map}
The three routes to A-TP considered across route~5, route~6, and
route~7 are:
\begin{description}
\item[Route~5 J (de Branges).] HB\(^{+}\) + BD\(^{+}\) + Det + SH;
tautology level: each hypothesis is RH-equivalent or strictly finer.
\item[Route~6 A (digamma).] Bounded-interval positivity on
\([-t_{\ast},t_{\ast}]\) after elementary reductions; tautology level:
bounded-interval residual RH-equivalent.
\item[Route~7 G (Hadamard).] Pair-pole pointwise positivity; tautology
level: pointwise positivity RH-equivalent
(\Cref{v4-arith:w7-g:prop:circular-reduction}).
\end{description}
All three routes bottom out at RH. The content each provides is a
\emph{different local rewriting} of the same positivity question. The
Hadamard route offers the most explicit pointwise formula
\eqref{v4-arith:w7-g:eq:K-explicit}, which clarifies the sign
mechanism of off-critical cancellation but does not close the
question.
\end{remark}

\section{Conclusion}
\label{v4-arith:w7-g:conclusion}

The Hadamard factorisation
\(\xi_{\mathbb{R}}(s)=e^{a+bs}\prod_{\rho}(1-s/\rho)e^{s/\rho}\)
decomposes the Weil distribution as
\(W(f*\widetilde{f})=\sum_{\rho}R_{\rho}[f]\) with
\(R_{\rho}[f]=\int|\widehat{f}(t)|^{2}\mathcal{K}_{\rho}(t)\,dt\)
and the explicit kernel \eqref{v4-arith:w7-g:eq:K-explicit}. Pointwise
non-negativity of \(\mathcal{K}_{\rho}\) is strictly equivalent to RH
(Proposition~\ref{v4-arith:w7-g:prop:circular-reduction}), so the
Hadamard route is tautological for A-TP via this mechanism. The
de Bruijn--Newman constant \(\Lambda\leq 0\) is likewise equivalent to
RH (Lemma~\ref{v4-arith:w7-g:lem:Lambda-leq-zero}). The quantitative
zero-free region of Korobov--Vinogradov type
(Proposition~\ref{v4-arith:w7-g:prop:weaker-condition-elementary}) is
the unique non-tautological output of this route: a genuine sufficient
condition for the partial-support form of A-TP on \(|t|\leq T\), improving
with the zero-free region width. Three reformulations of A-TP are pairwise equivalent to RH: Weil
positivity, the Hermite--Biehler condition, and pair-pole pointwise
positivity (Corollary~\ref{v4-arith:w7-g:cor:three-routes-equivalent}).

\begin{remark}[the non-tautological content of the Hadamard route]
\label{v4-arith:w7-g:rem:deliverable}
The Hadamard route yields an explicit pair-pole representation of
the Weil distribution and an explicit formula \eqref{v4-arith:w7-g:eq:K-explicit}
for the sign mechanism of off-critical zero cancellation. Pointwise
pair-pole positivity is equivalent to RH
(Proposition~\ref{v4-arith:w7-g:prop:circular-reduction}), so the
route does not close A-TP via this mechanism. The non-vacuous
contribution is the partial-support form of A-TP under classical
zero-free regions (Proposition~\ref{v4-arith:w7-g:prop:weaker-condition-elementary},
Remark~\ref{v4-arith:w7-g:rem:zero-free-region-standard}): a
conditional result useful for bounded-frequency applications.
\end{remark}
